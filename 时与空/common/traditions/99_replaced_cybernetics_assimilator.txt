tr_cybernetics_assimilator_adopt = {
	possible = {
		if = {
			limit = {
				has_origin = origin_reassembler
				has_civic = civic_machine_assimilator
			}
			has_technology = tech_integrated_cybernetics
		}
		else_if = {
			limit = { auto_cybernetics_ascension_machine = yes }
			has_technology = tech_integrated_cybernetics
		}
		else = {
			has_ascension_perk = ap_organo_machine_interfacing_assimilator
			has_technology = tech_integrated_cybernetics
		}
	}
	unlocks_agenda = agenda_cybernetic_advantage_assimilator
	
	on_enabled = {
		custom_tooltip = tr_cybernetics_assimilator_adopt_desc
		if = {
			limit = {
				has_origin = origin_reassembler
				has_civic = civic_machine_assimilator
			}
			add_research_option = tech_neuro_quantum_links
			add_research_option = tech_gene_tailoring
		}
		if = {
			limit = {
				has_machine_age_dlc = yes
				has_civic = civic_machine_assimilator
			}
			hidden_effect = {
				country_event = {
					id = cyber.2010
					days = 2
				}
			}
		}
	}
}
