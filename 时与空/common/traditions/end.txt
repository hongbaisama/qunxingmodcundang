		has_tradition = tr_time_finish
		can_adopt_tradition_end = yes

	modifier = {
		country_physics_research_produces_mult = 0.15
		country_society_research_produces_mult = 0.15
		country_engineering_research_produces_mult = 0.15
		pop_amenities_usage_mult = 0.3
	}
	triggered_modifier = {
		potential = {
			# 奇观拓展-坠星天矛
			has_country_flag = AOW_flag_CNTR_policy_starfall_spear_has_been_set
		}
		starfall_spear_cooldown_mult = -0.2
	}
	on_enabled = {
		if = {
			limit = {
				OR = {
					has_ascension_perk = ap_beyond_infinity
					is_defender_of_the_galaxy_country = no
				}
			}
			upgrade_r_time = { LEVEL = 5 }
		}
		if = {
			limit = {
				ap_beyond_infinity_basic_potential = yes
				has_beyond_infinity_civic = no
			}
			custom_tooltip = unlock_ap_beyond_infinity
		}
	}
}
