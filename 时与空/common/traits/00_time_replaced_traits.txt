	cost = 0

		"trait_sedentary"
		"trait_docile"
		"trait_unruly"
		"trait_vat_grown"
		"trait_nascent_stage"
	}
	initial = no
	randomized = no
	advanced_trait = no
	sapient = yes
	inline_script = "trait/allowed_archetypes/time_archetypes_no_robotic"
	tags = { organic special }
	custom_tooltip_with_modifiers = trait_wilderness_effect
	modifier = {
		species_empire_size_mult = -2
		pc_city_habitability = -2
		pc_machine_habitability = -2
		pc_relic_habitability = -2
		pc_hive_habitability = -2
		pc_ringworld_habitable_habitability = -2
		pc_shattered_ring_habitable_habitability = -2
		pc_nanotech_habitability = -2
		pc_habitat_habitability = -2
	}
	triggered_pop_group_modifier = {
		potential = {
			exists = owner
			owner = { is_wilderness_empire = yes }
			colony = { is_habitable_for_wilderness = yes }
		}
		pop_bonus_workforce_mult = 0.125
		mult = colony.value:trait_wilderness_workforce_value
	}
	triggered_planet_growth_habitability_modifier = {
		potential = {
			exists = fromfrom
			fromfrom = { is_wilderness_empire = yes }
			is_habitable_for_wilderness = yes
		}
		logistic_growth_mult = 0.25
		bonus_pop_growth_mult = 0.25
		mult = value:trait_wilderness_growth_value
	}
	species_potential_add = {
		always = no
	}
	species_possible_remove = {
		always = no
	}
	species_possible_merge_add = {
		always = no
	}
}

	inline_script = "trait/allowed_archetypes/time_archetypes"
	initial = no
	species_potential_add = {
		always = no
	}
	species_possible_merge_add = {
		always = yes
	}
	sorting_priority = 30
	modifier = {
		pc_gaia_habitability = 1.0
	}
	ai_weight = {
		weight = 0
	}
	custom_tooltip_with_modifiers = pc_gaia_preference_terraforming_trait_effect
	triggered_pop_group_modifier = {
		potential = {
			exists = colony
			colony = { is_planet_class = pc_gaia }
		}
		pop_bonus_workforce_mult = 0.1
	}
	triggered_pop_group_modifier = {
		potential = {
			exists = colony
			colony = { is_planet_class = pc_gaia }
			exists = owner
			owner = {
				if = {
					limit = {
						has_valid_civic = civic_origin_life_seeded
					}
					any_owned_leader = {
						is_councilor_type = councilor_origin_life_seeded
						NOT = { gender = female }
					}
				}
				else_if = {
					limit = {
						has_valid_civic = civic_origin_life_seeded_corporate
					}
					any_owned_leader = {
						is_councilor_type = councilor_origin_life_seeded_corporate
						NOT = { gender = female }
					}
				}
				else = {
					always = no
				}
			}
		}
		key = councilor_origin_life_seeded
		pop_bonus_workforce_mult = 1
		mult = owner.modifier:councilor_origin_life_seeded_modifier
	}
	triggered_pop_group_modifier = {
		potential = {
			exists = colony
			colony = { is_planet_class = pc_gaia }
			exists = owner
			owner = {
				if = {
					limit = {
						has_valid_civic = civic_origin_life_seeded
					}
					any_owned_leader = {
						is_councilor_type = councilor_origin_life_seeded
						gender = female
					}
				}
				else_if = {
					limit = {
						has_valid_civic = civic_origin_life_seeded_corporate
					}
					any_owned_leader = {
						is_councilor_type = councilor_origin_life_seeded_corporate
						gender = female
					}
				}
				else = {
					always = no
				}
			}
		}
		key = councilor_origin_life_seeded_female
		pop_bonus_workforce_mult = 1
