# 圆环之子
leader_trait_ruler_time = {
	immortal_leaders = yes
	replace_traits = { "leader_trait_time_soul" }
	# 不受监管人/帝皇增强
	inline_script = {
		script = trait/time_trait_icon_for_origin_time
		CLASS = leader
		ICON = "GFX_leader_trait_ruler_time"
		RARITY = common
		TIER = none
		PRIORITY = 1000
	}
	inline_script = {
		script = traits/ruler_time_trait_modifiers
		BASE = 1
	}
	custom_tooltip_with_modifiers = leader_trait_ruler_time_tooltip
	on_gained_effect = {
		if = {
			limit = {
				is_origin_time_ruler = yes
				owner = { has_origin_time = yes }
			}
			set_leader_flag = immune_to_negative_traits
			set_leader_flag = leader_death_events_blocked
			change_background_job = origin_time_ruler
			change_background_ethic = ethic_materialist
		}
		else = {
			remove_trait = leader_trait_ruler_time
		}
	}
	leader_class = all
	initial = no
	randomized = no
}
