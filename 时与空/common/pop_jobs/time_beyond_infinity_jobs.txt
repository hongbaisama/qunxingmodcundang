# 时空旅行者

time_traveler = {
	category = ruler
	can_be_automated = no
	
	swappable_data = {
		default = {
			condition_string = RULER_JOB_TRIGGER
			building_icon = building_time_machine
		}
		swap_type = {
			trigger = {
				exists = carrier
				colony = { has_building = building_infinity_time_machine }
			}
			building_icon = building_infinity_time_machine
			weight = 5
		}
	}

	tags = { research }

	possible_pre_triggers = {
		has_owner = yes
		is_enslaved = no
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_ruler_job
	
	inline_script = "jobs/beyond_infinity/time_traveler"

	weight = {
		weight = @ruler_job_weight
		mult = 10000
		modifier = {
			factor = 0.2
			has_citizenship_rights = no
			NOT = { has_trait = trait_mechanical }
		}
	}
	
	inline_script = "jobs/automodding_priority_research"
}
