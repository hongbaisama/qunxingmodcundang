gas_technician = {
	category = worker
	
	swappable_data = {
		default = {
			condition_string = WORKER_JOB_TRIGGER
			building_icon = building_energy_grid
			icon = technician
		}
		swap_type = {
			trigger = {
				gas_colony_job_is_civic_origin_gas_colony_job_trigger = yes
			}
			name = civic_origin_gas_colony_technician
			desc = civic_origin_gas_colony_technician
			icon = civic_origin_gas_colony_technician
			weight = 5
		}
	}

	tags = { energy }

	possible_pre_triggers = {
		has_owner = yes
		is_being_purged = no
		is_being_assimilated = no
		is_sapient = yes
	}

	possible_precalc = can_fill_worker_job
	possible = {
		hidden_trigger = { exists = owner }
		owner = {
			is_gestalt = no
		}
		has_trait = trait_origin_gas_colony
	}

	resources = {
		category = planet_technician
		produces = {
			energy = 6
			volatile_motes = 0.5
		}
		produces = {
			trigger = {
				colony = {
					has_carrier_flag = has_energy_requisitorium
				}
			}
			energy = @requisitorium_energy_reduction
		}
		produces = {
			trigger = {
				gas_colony_job_is_civic_origin_gas_colony_job_trigger = yes
			}
			energy = -1.5
		}
	}
	
	overlord_resources = {
		category = planet_requisitioned_energy
		produces = {
			trigger = {
				colony = {
					has_carrier_flag = has_energy_requisitorium
				}
			}
			energy = @requisitorium_energy_overlord
		}
	}
	
	triggered_country_modifier = {
		potential = {
			gas_colony_job_is_civic_origin_gas_colony_job_trigger = yes
		}
		ship_shield_mult = 0.01
	}
	
	# 技工部分
	
	inline_script = "jobs/vanilla/vanilla_technician_modifiers"
	
	inline_script = {
		script = jobs/vanilla_weight/vanilla_technician_weight
	}
	
	inline_script = "jobs/automodding_priority_energy"
}
