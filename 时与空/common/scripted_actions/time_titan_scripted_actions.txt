time_titan_end_shelter = {
	user_scope = fleet
	scope = planet
	
	possible = {
		is_colony = yes
		custom_tooltip = {
			fail_text = is_not_hostile_or_primitive_or_ours
			OR = {
				AND = {
					exists = owner
					owner = { is_at_war_with = from.controller }
					if = {
						limit = { exists = controller }
						controller = { is_at_war_with = from.controller }
					}
				}
				AND = {
					exists = owner
					owner = { is_country_type = primitive }
					solar_system = { is_owned_by = from.controller }
				}
				is_owned_by = from.controller
			}
		}
		custom_tooltip = {
			fail_text = recently_time_titan_end_shelter_good
			NOT = { has_modifier = time_titan_end_shelter_good }
		}
		custom_tooltip = {
			fail_text = recently_time_titan_end_shelter_bad
			NOT = { has_modifier = time_titan_end_shelter_bad }
		}
		custom_tooltip = {
			fail_text = time_titan_shelter_require_syzygy_admiral
			from = { time_titan_is_fine = yes }
		}
	}
	
	button_clickable = {
		custom_tooltip = {
			fail_text = time_titan_shelter_require_syzygy_admiral
			time_titan_is_fine = yes
		}
	}
	
	button_visible = {
		owner = { has_country_flag = upgrade_to_time_titan_finished }
	}
	
	on_completed = on_time_titan_end_shelter_completed
	on_progress_start = on_time_titan_end_shelter_start
	on_cancelled = on_time_titan_end_shelter_cancelled
	
	context_menu_name = time_titan_end_shelter
	tooltip = time_titan_end_shelter_tooltip
	activity_key = time_titan_end_shelter_activity_key
	progress_activity_key = time_titan_end_shelter_progress_activity_key
	required_progress = 90
	cooldown = 360
	icon = GFX_fleet_action_time_titan_end_shelter
	icon_selected = GFX_fleet_action_time_titan_end_shelter
	
	progress_anim_state = "wind_up"
	
	finish_anim_state = "wind_down"
	
	shortcut = "CTRL+W"
	slot = 4
}
