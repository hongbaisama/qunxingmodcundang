col_$TYPE$_capital = {
	icon = GFX_col_$TYPE$_capital

	potential = {
		unlock_col_$TYPE$_capital = yes
	}

	triggered_planet_modifier = {
		planet_stability_add = 2.5
		pop_happiness = 0.05
		pop_bonus_workforce_mult = 0.05
		pop_government_ethic_attraction = 0.5
		district_psionic_modifier_mult = 0.25
		mult = value:num_col_psionic_value|BASE|$LEVEL$|
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner_or_space_owner
			owner_or_space_owner = {
				OR = {
					has_tradition = tr_psionics_adopt
					has_tradition = tr_psionics_shroud_adopt
				}
			}
		}
		psionic_pop_bonus_workforce_mult = 0.05
		mult = value:num_col_psionic_value|BASE|$LEVEL$|
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner_or_space_owner.federation
			owner_or_space_owner.federation = { has_federation_perk = all_the_priests_1 }
		}
		job_bureaucrat_per_pop = 0.04
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner_or_space_owner
			owner_or_space_owner = { has_tradition = tr_adaptability_finish }
		}
		pop_bonus_workforce_mult = 0.025
	}

	ai_building_set_affinity = {
		trade = 2.0
		unity = 1.5
	}

	weight_modifier = {
		base = 20000
	}
}
