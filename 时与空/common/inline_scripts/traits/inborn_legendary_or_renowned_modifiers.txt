# 帝国代表
triggered_galcom_modifier = {
	potential = {
		has_trait = subclass_official_delegate
		is_ruler = no
	}
	diplo_weight_mult = @inborn_$TYPE$_modifier
	country_trust_growth = @inborn_$TYPE$_modifier
	mult = value:leader_trait_inborn_legendary_or_renowned_value
}
# 产业专员（默认）
triggered_planet_modifier = {
	potential = {
		if = {
			limit = {
				leader_class = official
			}
			NOR = {
				has_trait = subclass_official_delegate
				has_trait = subclass_official_diplomacy_councilor
				has_trait = subclass_official_economy_councilor
				is_ruler = yes
			}
		}
		else = {
			always = no
		}
	}
	planet_jobs_specialist_produces_mult = @inborn_$TYPE$_modifier
	planet_building_build_speed_mult = @inborn_$TYPE$_modifier
	mult = value:leader_trait_inborn_legendary_or_renowned_value
}

# 国务顾问
triggered_councilor_modifier = {
	potential = {
		if = {
			limit = {
				leader_class = official
			}
			OR = {
				has_trait = subclass_official_economy_councilor
				AND = {
					is_ruler = yes
					NOT = { has_trait = subclass_official_diplomacy_councilor }
				}
			}
		}
		else = {
			always = no
		}
	}
	country_trade_produces_mult = @inborn_$TYPE$_modifier_council
	council_agenda_progress_speed = @inborn_$TYPE$_modifier_council
	mult = value:leader_trait_inborn_legendary_or_renowned_value
}
