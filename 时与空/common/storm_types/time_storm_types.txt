# 用于概念显示与命运最终战
dimensional_horror_crisis_storm = {
	inline_script = {
		script = "cosmic_storms/dimensional_horror_crisis_storm"
		TYPE = destiny
	}
	
	affect_hp_regen = yes
	affect_shield_regen = yes
	affect_armor_regen = yes
	
	cosmic_storm_texture_color_path = "gfx/map/storms/dimensional_horror_crisis_levo_storm_color.dds"
	
	# 不用triggered_ship_modifier是为了减少卡顿
	triggered_fleet_modifier = {
		potential = {
			immune_to_dimensional_horror_crisis_storm = no
			is_dimensional_horror_crisis_inactive_system = no
		}
		ship_speed_reduction = 0.5
		ship_shield_reduction = 0.5
		ship_armor_reduction = 0.5
		ship_shield_regen_mult = -0.5
		ship_armor_regen_mult = -0.5
		ship_hull_regen_mult = -0.5
		weapon_type_strike_craft_speed_mult = -0.5
		ship_evasion_mult = -0.5
		ship_evasion_add = -50
		ship_disengage_chance_reduction = 1
		ship_emergency_ftl_mult = 1
		ship_emergency_ftl_min_days_mult = 1
		ship_disengage_opportunities_add = -5
		mult = value:dimensional_horror_crisis_storm_fleet_modifier
		custom_tooltip = dimensional_horror_crisis_storm_fleet_tt
		show_only_custom_tooltip = no
	}
}
