	length = 1
	icon = "GFX_trait_origin_black_hole"
	resources = {
		category = edicts_unscaled
		cost = {
			sr_dark_matter = 500
		}
	}

	modifier = {
	}

	potential = {
		if = {
			limit = {
				has_technology = tech_end_trait
			}
			ruler = {
				has_pass_black_hole = no
				OR = {
					has_any_subclass = yes
					is_dimension_traveler = yes
				}
			}
		}
		else = {
			always = no
		}
	}
	
	allow = {
		custom_tooltip = {
			fail_text = "requires_black_hole_system_within_border"
			any_system_within_border = {
				is_time_black_hole_star_class = yes
			}
		}
	}
	
	effect = {
		ruler = {
			leader_pass_black_hole = yes
		}
		if = {
			limit = {
				is_gestalt = yes
			}
			every_owned_leader = {
				limit = {
					is_gestalt_node = yes
					has_pass_black_hole = no
				}
				leader_pass_black_hole = yes
			}
		}
	}
	
	ai_weight = {
		weight = 10
	}
}
