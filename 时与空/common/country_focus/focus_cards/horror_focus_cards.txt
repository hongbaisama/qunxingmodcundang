horror_focus_2 = {
	category = exploration
	tier = 2
	background = "GFX_focus_card_background_exploration_tier_2"

	name = "horror.2.name"
	desc = "horror.2.desc"
	hint = "horror.2.hint"
	lore = "horror.2.lore"
	
	possible = {
		has_country_flag = horror_focus_1_completed
	}

	condition_for_achievement = {
		custom_tooltip = {
			fail_text = "horror.2.fail"
			success_text = "horror.2.success"
			if = {
				limit = {
					check_variable = {
						which = horror_focus_2_finished
						value >= 3
					}
				}
				custom_progress = {
					check_variable = {
						which = horror_focus_2_finished
						value >= 3
					}
					current_val_coeff = 3
					final_val_coeff = 3
				}
			}
			else = {
				custom_progress = {
					check_variable = {
						which = horror_focus_2_finished
						value >= 3
					}
					mode = clamped
				}
			}
		}
	}

	on_success = {
		create_horror_focus_completed_message = {
			CATEGORY = EXPLORATION
			ID = 2
		}
		add_focus_progress = { category = exploration amount = 25 }
		set_country_flag = horror_focus_2_completed
	}
}
