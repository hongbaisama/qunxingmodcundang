	icon = GFX_district_specialization_energy
	base_buildtime = @zone_buildtime
	resources = {
		category = planet_zones
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = alloys
			COST = @zone_cost
			NOMADIC_COST_MULT = 0.5
		}
	}

	potential = { # planet scope
		is_planet_class = pc_nanites
	}

	unlock = { # planet scope
	}
	
	zone_sets = {
		zone_nanites_energy
	}

	included_building_sets = {
		generator
		fallen_empire_generator
		generator_automation
	}

	triggered_district_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = no }
		}
		job_technician_add = @scaling_district_1_job
		mult = value:district_nanites_value
	}

	triggered_district_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = yes }
		}
		job_technician_drone_add = @scaling_district_1_job
		mult = value:district_nanites_value
	}

	planet_modifier = {
		zone_building_slots_add = @time_mod_zone_max_buildings
	}

	convert_to = {
	}
	
	ai_priority = 5
}
