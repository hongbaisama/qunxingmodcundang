utility_component_template = {
	key = "SMALL_TIME_SHIELD"
	size = small
	icon = "GFX_ship_part_time_shield"
	icon_frame = 1
	power = @power_S7
	resources = {
		category = ship_components
		cost = {
			alloys = @shield_s_t7_cost
			sr_dark_matter = 0.5
		}
		inline_script = {
			script = component_templates/time_component_template_basic_upkeep
			ENERGY = @shield_s_t7_upkeep_energy
			ALLOYS = @shield_s_t7_upkeep_alloys
		}
	}
	
	custom_tooltip = small_time_shield_effect
	
	modifier = {
		ship_shield_add = @shield_S7
		ship_shield_regen_add_static = @regen_S7 #Regeneration per day
	}
	
	ship_modifier = {
		ship_shield_hardening_add = @time_extra_modifier_S7
		weapon_confuser_protect_chance = @time_extra_modifier_S7
		weapon_terrify_protect_chance = @time_extra_modifier_S7
	}
	
	triggered_ship_modifier = {
		ship_shield_regen_add_static = @regen_S7 #Regeneration per day
		mult = value:num_time_modifier|BASE|0|MODIFIER|ship_shield_mult|
	}
	
	triggered_ship_modifier = {
		potential = {
			active_ap_horror_bonus = { SCOPE = owner }
		}
		ship_shield_hardening_add = @time_extra_modifier_S7
	}
	
	triggered_ship_design_modifier = {
		potential = {
			active_ap_time_bonus = { SCOPE = from }
		}
		ship_shield_mult = @time_extra_modifier_S7
	}
	
	prerequisites = { "tech_time_shield" }
	component_set = "TIME_SHIELD"
	upgrades_to = "SMALL_END_SHIELD"
	
	potential = {
		from = { has_tradition = tr_time_1 }
	}

	ai_weight = {
		weight = 100
		modifier = {
			factor = 0.0
			no_resource_for_component = { RESOURCE = sr_dark_matter }
			is_ai = yes
		}
	}
}
