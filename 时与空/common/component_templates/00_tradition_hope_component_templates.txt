	modifier = {
		ship_hull_add = @hull_S7
	}

		ship_damage_reduction_add = 3

		potential = {
			active_ap_horror_bonus = { SCOPE = owner }
		}
		ship_damage_reduction_add = 3

		potential = {
			active_ap_time_bonus = { SCOPE = from }
		}

	}

	component_set = "HOPE_HULL"
	
	potential = {
		from = { has_tradition = tr_hope_4 }
	}
	
	resources = {
		category = ship_components
		cost = {
			alloys = @s_t7_cost
			sr_dark_matter = 1
		}
		inline_script = {
			script = component_templates/time_component_template_basic_upkeep
			ENERGY = @s_t7_upkeep_energy
			ALLOYS = @s_t7_upkeep_alloys
		}
	}
	
	ai_weight = {
		weight = 100
		modifier = {
			factor = 0.0
			no_resource_for_component = { RESOURCE = sr_dark_matter }
			is_ai = yes
		}
	}
}
