	key = "SMALL_END_SHIELD"
	size = small
	icon = "GFX_ship_part_end_shield"
	icon_frame = 1
	power = @time_power_S8
	resources = {
		category = ship_components
		cost = {
			alloys = @time_shield_s_t8_cost
			sr_dark_matter = 1
		}
		inline_script = {
			script = component_templates/time_component_template_basic_upkeep
			ENERGY = @time_shield_s_t8_upkeep_energy
			ALLOYS = @time_shield_s_t8_upkeep_alloys
		}
	}
	
	custom_tooltip = small_end_shield_effect
	
	modifier = {
		ship_shield_add = @time_shield_S8
		ship_shield_regen_add_static = @time_regen_S8 #Regeneration per day
	}
	
	ship_modifier = {
		ship_shield_hardening_add = @time_extra_modifier_S8
		weapon_confuser_protect_chance = @time_extra_modifier_S8
		weapon_terrify_protect_chance = @time_extra_modifier_S8
	}
	
	triggered_ship_modifier = {
		ship_shield_regen_add_static = @time_regen_S8 #Regeneration per day
		mult = value:num_time_modifier|BASE|0|MODIFIER|ship_shield_mult|
	}
	
	triggered_ship_modifier = {
		potential = {
			active_ap_horror_bonus = { SCOPE = owner }
		}
		ship_shield_hardening_add = @time_extra_modifier_S8
	}
	
	triggered_ship_design_modifier = {
		potential = {
			active_ap_time_bonus = { SCOPE = from }
		}
		ship_shield_mult = @time_extra_modifier_S8
	}
	
	prerequisites = { "tech_end_shield" }
	component_set = "END_SHIELD"
	
	potential = {
		from = { has_tradition = tr_end_1 }
	}

	ai_weight = {
		weight = 100
		modifier = {
			factor = 0.0
			no_resource_for_component = { RESOURCE = sr_dark_matter }
			is_ai = yes
		}
	}
}

weapon_component_template = {
	key = "DIMENSION_END_BEAM"
	size = titanic
	type = instant
	prio_projectile = yes
	hidden = yes

	icon = "GFX_ship_part_dimension_end_beam"
	icon_frame = 1
	projectile_gfx = "end_beam"
	
	ship_limit = 1
	
	power = -500

	firing_arc = 30.0
	damage = { min = 1.00 max = 9999.00 }
	windup = { min = 20.00 max = 80.00 }
	hull_damage = 2.50
	shield_damage = 5.00
	shield_penetration = 2.00
	armor_damage = 5.00
	armor_penetration = 2.00
	total_fire_time = 100.00
	range = 250.00
	accuracy = 1.00
	tracking = 1.00
	size_damage_factor = 0.2
	military_power_multiplier = 5
	collateral_damage = @dimension_end_beam_aoe_damage
	collateral_range = @dimension_end_beam_aoe_range
	
	on_hit = {
		horror_weapon_on_hit_aoe_effect = {
			TYPE = dimension_end
			DEBUFF = terrify
			BIO_SHIPS = $BIO_SHIPS$
		}
	}
	
	injected_modifier = {
		modifier = {
			modifier = dimension_end_beam_modifier
			days = @end_beam_injected_modifier_days_no
		}
		stack_info = {			#A ship can only be affected by one injected modifier of a certain id. If multiple modifiers have the same ID the one with the highest priority will be applied.
			id = end_beam_modifier
			priority = 2
		}
	}

	tags = { weapon_type_dimensional_horror weapon_type_energy weapon_type_kinetic weapon_role_artillery t_slot }
	ai_tags = { weapon_role_anti_hull weapon_role_artillery artillery }
	
	inline_script = {
		script = component_templates/time_target_weights_focus_large
	}
	
	custom_tooltip = dimension_end_beam_effect
	
	triggered_ship_modifier = {
		potential = {
			leader = { is_syzygy_admiral = yes }
		}
		ship_hull_damage_mult = 0.05
		ship_shield_damage_mult = 0.05
		ship_armor_damage_mult = 0.05
		ship_shield_penetration_add = 0.05
		ship_shield_penetration_mult = 0.05
		ship_armor_penetration_add = 0.05
		ship_armor_penetration_mult = 0.05
		mult = leader.value:syzygy_admiral_level
	}
	
	triggered_ship_modifier = {
		potential = {
			is_special_dimensional_horror = yes
		}
		ship_hull_damage_mult = 1
		ship_shield_damage_mult = 1
		ship_armor_damage_mult = 1
		ship_shield_penetration_add = 1
		ship_shield_penetration_mult = 1
		ship_armor_penetration_add = 1
		ship_armor_penetration_mult = 1
		mult = value:dimensional_horror_ship_size_bonus_difficulty_value
	}
	
	component_set = "DIMENSION_END_BEAM"

	ai_weight = {
		weight = 0
	}
}
