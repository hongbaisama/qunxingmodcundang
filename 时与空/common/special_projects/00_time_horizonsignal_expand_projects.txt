	key = "TIME_WANDERER_PROJECT"
	cost = 0
	days_to_research = 36000
	tech_department = physics_technology
	picture = GFX_evt_black_hole
	icon = "gfx/interface/icons/situation_log/situation_log_extradimensional.dds"
	
	event_scope = ship_event
	
	event_chain = TIME_WANDERER_CHAIN
	
	requirements = {
		shipclass_science_ship = 1
		leader = scientist
		has_trait = leader_trait_dimension_traveler_brief
	}
	
	abort_trigger = {
		event_target:dimension_traveler_owner = {
			dimension_traveler_is_disappointed = yes
		}
	}
	
	on_start = {
		# 这里总是会丢域，干脆直接检索事件目标
		event_target:dimension_traveler_owner = {
			country_event = { id = time_wanderer.110 }
		}
	}
	
	on_cancel = {
		event_target:dimension_traveler_owner = {
			country_event = { id = time_wanderer.115 }
		}
	}
	
	on_success = {
		event_target:dimension_traveler_owner = {
			country_event = { id = time_wanderer.120 }
		}
	}
	
	on_fail = {
	}
}
