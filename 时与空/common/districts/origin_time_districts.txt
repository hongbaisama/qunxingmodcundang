district_time = {
	overlay_icon = GFX_district_government
	base_buildtime = 360
	exempt_from_ai_planet_specialization = yes
	
	custom_gui = planet_district_time_entry

	zone_slots = {
		slot_time_0
		slot_time_1
		slot_time_2
	}
	
	show_on_uncolonized = {
		OR = {
			is_planet_class = pc_time
			is_planet_class = pc_time_replace # 以防万一
		}
	}
	
	potential = {
		OR = {
			is_planet_class = pc_time
			is_planet_class = pc_time_replace # 以防万一
		}
		NAND = {
			is_capital = yes
			owner = {
				has_menace_perk = menp_behemoth_mind_meld
			}
		}
	}

	conversion_ratio = 1
	convert_to = {
		district_endless_time
		district_mindlink
		district_ark_city
	}
	
	resources = {
		category = planet_districts_time
		cost = {
			energy = 500
			minerals = 500
			food = 500
			consumer_goods = 250
			alloys = 250
			sr_dark_matter = 25
		}
		upkeep = {
			rare_crystals = 1
			exotic_gases = 1
			volatile_motes = 1
		}
	}
	
	# 通用
	planet_modifier = {
		building_time_mult = -0.01
	}
	
	# 主区划
	triggered_planet_modifier = {
		planet_housing_add = @district_time_base_housing
		district_time_usage_cap = @district_time_base_usage_cap
		mult = value:district_time_level_value
	}
	
	triggered_desc = {
		text = district_time_effect
	}
	
	prerequisites = {
		tech_mine_dark_matter
	}
	
	show_tech_unlock_if = {
		any_owned_colony = {
			is_planet_class = pc_time
		}
	}
	
	inline_script = {
		script = districts/district_time_triggered_name
	}
	
	inline_script = {
		script = districts/ai_urban_district_extra_weighting
	}
}
