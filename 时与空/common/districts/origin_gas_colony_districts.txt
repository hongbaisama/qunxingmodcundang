district_gas_city = {
	overlay_icon = GFX_district_government
	base_buildtime = 480
	exempt_from_ai_planet_specialization = yes

	zone_slots = {
		slot_city_government
		slot_city_01
		slot_city_02
	}

	show_on_uncolonized = {
		exists = from
		from = { is_regular_empire = yes }
		is_planet_class = pc_gas_colony
		is_special_colony_type = no
	}

	potential = {
		exists = owner
		owner = {
			OR = {
				is_regular_empire = yes
				AND = {
					is_country_type = primitive
					is_hive_empire = no
				}
			}
		}
		is_planet_class = pc_gas_colony
		is_special_colony_type = no
		NAND = {
			is_capital = yes
			owner = {
				has_menace_perk = menp_behemoth_mind_meld
			}
		}
	}

	allow = {
		is_special_colony_type = no
	}

	conversion_ratio = 1
	convert_to = {
		district_gas_nexus
		district_gas_hive
		district_crashed_slaver_ship
		district_arcology_housing
		district_rw_city
		district_city
		district_nexus
		district_hive
		district_prison
		district_craglands
		district_mindlink
		district_crashed_slaver_ship
		district_resort
		district_slave
		district_ark_city
	}

	resources = {
		category = planet_districts_cities
		cost = {
			minerals = @city_cost
			volatile_motes = 25
			rare_crystals = 25
			exotic_gases = 25
			trigger = {
				NOT = { has_modifier = wooden_planet }
			}
		}
		cost = {
			food = @city_cost
			volatile_motes = 25
			rare_crystals = 25
			exotic_gases = 25
			trigger = {
				has_modifier = wooden_planet
			}
		}
		upkeep = {
			energy = 2
			trigger = {
				NOT = { has_modifier = wooden_planet }
			}
		}
		upkeep = {
			food = 2
			trigger = {
				has_modifier = wooden_planet
			}
		}
	}

	inline_script = {
		script = districts/time_district_city_modifier
	}
	
	inline_script = {
		script = districts/district_triggered_name_urban
	}
	
	inline_script = {
		script = districts/ai_urban_district_extra_weighting
	}
}
