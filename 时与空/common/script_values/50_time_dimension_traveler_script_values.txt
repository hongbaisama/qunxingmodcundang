dimension_traveler_horror_modifier_chance = {
	base = 1
	multiply = value:dimension_traveler_modifier_mult_value
	modifier = {
		multiply = 2
		# 行政官+长久和平
		if = {
			limit = {
				leader_class = official
			}
			owner = {
				years_of_peace = {
					value > 9
				}
			}
		}
		else = {
			always = no
		}
	}
	modifier = {
		multiply = 2
		# 科学家+游戏中期
		if = {
			limit = {
				leader_class = scientist
			}
			mid_game_years_passed >= 0
		}
		else = {
			always = no
		}
	}
	modifier = {
		multiply = 2
		# 指挥官+战争状态
		if = {
			limit = {
				leader_class = commander
			}
			owner = {
				OR = {
					is_at_war = yes
					any_country = {
						NOT = { is_same_value = prev }
						OR = {
							has_crisis_level = crisis_level_5
							is_crisis_faction = yes
						}
					}
				}
			}
		}
		else = {
			always = no
		}
	}
}

	}
}

}
