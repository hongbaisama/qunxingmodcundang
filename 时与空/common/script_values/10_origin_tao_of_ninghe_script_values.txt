# num_buildings里，使用了category条目后，disabled条目似乎会失效，无论建筑是否禁用都会被纳入计数
num_yin_value = {
	base = 0
	add = value:num_time_unity_building|MULT|5|
	add = value:num_time_trade_building|MULT|5|
	inline_script = { script = script_values/num_category_buildings	CATEGORY = amenity			MULT = 5	MODE = add }
	inline_script = { script = script_values/num_category_buildings	CATEGORY = manufacturing	MULT = 5	MODE = add }
	# 铸造类建筑的category为manufacturing，不应计入
	subtract = value:num_time_foundry_building|MULT|5|
	# 区划特化
	add = value:num_time_unity_zone|MULT|10|
	add = value:num_time_trade_zone|MULT|10|
	add = value:num_time_factory_zone|MULT|10|
	add = value:num_time_amenity_zone|MULT|10|
	multiply = value:num_yin_mult_value
}

}
