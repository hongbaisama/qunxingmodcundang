# 彼岸之门每次转变总人口数百分之$PERCENTAGE$的人口，此处计算理应转变人口数占可转变人口数的百分比

	base = $PERCENTAGE$
	complex_trigger_modifier = {
		trigger = count_owned_pop_amount
		parameters = {
			limit = { always = yes }
		}
		mode = mult
	}
	complex_trigger_modifier = {
		trigger = count_owned_pop_amount
		parameters = {
			limit = {
				has_akx_waiting_world = yes
				has_akx_messenger = no
			}
		}
		mode = divide
		min_value = 1
	}
	min = 0
	max = 1
}
