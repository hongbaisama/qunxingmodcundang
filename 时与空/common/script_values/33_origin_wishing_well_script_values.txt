d_wishing_well_produce_value = {
	base = 2
	pow = value:d_wishing_well_pow_value
	multiply = value:d_wishing_well_ascension_perks_value
	multiply = value:d_wishing_well_upkeep_afford_percentage
}

d_wishing_well_upkeep_value = {
	base = 2
	pow = value:d_wishing_well_pow_value
	multiply = value:d_wishing_well_ascension_perks_value
}

d_wishing_well_pow_value = {
	base = 0
	add = trigger:years_passed
	divide = @d_wishing_well_double_interval_year
	modifier = {
		divide = @d_wishing_well_double_interval_origin_factor
		exists = owner
		owner = { has_origin = origin_wishing_well }
	}
	modifier = {
		divide = value:d_wishing_well_pow_ai_value # 根据AI难度削弱
		exists = owner
		owner = { is_ai = yes }
	}
	min = 0
	max = @d_wishing_well_max_pow # 防止太大导致数据溢出
	modifier = {
		max = @d_wishing_well_max_ai_pow # AI上限削弱
		exists = owner
		owner = { is_ai = yes }
	}
}

d_wishing_well_pow_ai_value = {
	base = @d_wishing_well_double_interval_difficulty_default_ai_factor
	modifier = {
		set = @d_wishing_well_double_interval_difficulty_6_ai_factor
		is_difficulty = 6
	}
	modifier = {
		set = @d_wishing_well_double_interval_difficulty_5_ai_factor
		is_difficulty = 5
	}
	modifier = {
		set = @d_wishing_well_double_interval_difficulty_4_ai_factor
		is_difficulty = 4
	}
	modifier = {
		set = @d_wishing_well_double_interval_difficulty_3_ai_factor
		is_difficulty = 3
	}
	modifier = {
		multiply = @d_wishing_well_double_interval_subject_ai_factor
		owner = { is_subject = yes }
	}
}

d_wishing_well_ascension_perks_value = {
	base = 0
	add = owner.trigger:num_ascension_perks
	multiply = @d_wishing_well_resource_ascension_perks_factor
	add = 1
}

d_wishing_well_modifier_value = {
	base = 50
	subtract = trigger:years_passed
	# 50年前，可支付的许愿井维护费越少，增益越少
	modifier = {
		multiply = value:d_wishing_well_upkeep_afford_percentage
		years_passed <= 50
	}
	# 50年后，无法支付的许愿井维护费越多，减益越多
	modifier = {
		multiply = value:d_wishing_well_upkeep_deficit_percentage|BASE|2|
		years_passed > 50
	}
}

d_wishing_well_modifier_special_value = {
	base = 50
	subtract = trigger:years_passed
	max = -10
	# 无法支付的许愿井维护费越多，减益越多
	modifier = {
		multiply = value:d_wishing_well_upkeep_deficit_percentage|BASE|1|
		years_passed <= 50
	}
	modifier = {
		multiply = value:d_wishing_well_upkeep_deficit_percentage|BASE|2|
		years_passed > 50
	}
}

lost_wishing_well_modifier_value = {
	base = 50
	subtract = trigger:years_passed
	max = -10
}

# 可支付的许愿井维护费占比
d_wishing_well_upkeep_afford_percentage = {
	base = 0
	complex_trigger_modifier = {
		trigger = has_country_resource
		trigger_scope = owner
		parameters = {
			type = trade
		}
		mode = add
	}
	complex_trigger_modifier = {
		trigger = resource_income_compare
		trigger_scope = owner
		parameters = {
			resource = trade
		}
		mode = add
	}
	divide = @d_wishing_well_trade_upkeep # 许愿井的基础贸易维护费
	divide = value:d_wishing_well_upkeep_value
	add = 1 # resource_income_compare中已经计算过一次许愿井的维护费了
	max = 1
	min = 0
}

# 无法支付的许愿井维护费占比
d_wishing_well_upkeep_deficit_percentage = {
	base = $BASE|1$
	subtract = value:d_wishing_well_upkeep_afford_percentage
}
