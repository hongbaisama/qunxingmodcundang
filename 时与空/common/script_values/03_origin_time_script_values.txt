# 每超出1%航道容量导致0.9%惩罚，超出100%航道容量达最高（90%）
# 值=1-0.9(航道占用-航道容量)/航道容量=1.9-0.9航道占用/航道容量
num_district_time_usage_punishment = {
	base = 0
	subtract = colony.modifier:district_time_usage
	modifier = {
		divide = colony.value:num_district_time_usage_cap
		colony = {
			check_modifier_value = {
				modifier = district_time_usage_cap
				value > 0
			}
		}
	}
	modifier = {
		set = -2
		colony = {
			check_modifier_value = {
				modifier = district_time_usage_cap
				value <= 0
			}
		}
	}
	multiply = 0.9
	add = 1.9
	max = 1
	min = 0.1
}
