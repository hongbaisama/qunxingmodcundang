# 晋升著名/传奇数量限制
promotes_tier_mult_value = {
	base = -1
	multiply = value:promotes_mult_value_check_tier_num|TIER|$TIER$|
	divide = value:promotes_mult_value_check_tier_cap|TIER|$TIER$|
	add = 1
	max = 1
	min = 0
}

promotes_tier_mult_pool_value = {
	base = -1
	multiply = value:promotes_mult_value_check_tier_num|TIER|$TIER$|POOL|yes|
	divide = value:promotes_mult_value_check_tier_cap|TIER|$TIER$|
	add = 1
	max = 1
	min = 0
	multiply = 0.5 # 领袖池中领袖晋升概率更低
}

promotes_mult_value_check_tier_num = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_leader
		trigger_scope = owner
		parameters = {
			limit = {
				is_leader_tier = leader_tier_$TIER$
				has_leader_flag = has_tried_to_promote_to_$TIER$_leader
				NOT = { has_trait = leader_trait_inborn_$TIER$ }
			}
		}
		mult = 1
		mode = add
	}
	[[POOL]
		complex_trigger_modifier = {
			trigger = count_pool_leader
			trigger_scope = owner
			parameters = {
				limit = {
					is_leader_tier = leader_tier_$TIER$
					has_leader_flag = has_tried_to_promote_to_$TIER$_leader
					NOT = { has_trait = leader_trait_inborn_$TIER$ }
				}
			}
			mult = 1
			mode = add
		}
	]
}

promotes_mult_value_check_tier_cap = {
	base = 0
	add = owner.modifier:promotes_$TIER$_cap
	floor = yes
	min = 1 # 别给狡猾人减到分母为0或负数了
}

# 超过领袖容量会降低晋升概率（达到领袖上限的2倍时概率归零）
promotes_mult_value = {
	base = 0
	modifier = {
		add = value:promotes_mult_value_check_leader_class|class|official|
		leader_class = official
	}
	modifier = {
		add = value:promotes_mult_value_check_leader_class|class|scientist|
		leader_class = scientist
	}
	modifier = {
		add = value:promotes_mult_value_check_leader_class|class|commander|
		leader_class = commander
	}
	max = 1
	min = 0
}

promotes_mult_value_check_leader_class = {
	base = -1
	multiply = value:promotes_mult_value_check_leader_num|class|$class$|
	divide = value:promotes_mult_value_check_leader_cap|class|$class$|
	add = 2
}

promotes_mult_value_check_leader_num = {
	base = 0
	complex_trigger_modifier = {
		trigger = count_owned_leader
		trigger_scope = owner
		parameters = {
			limit = {
				leader_class = $class$
				# 参见game_rules的should_count_towards_leader_cap
				NOR = {
					is_hidden = yes
					has_leader_flag = hired_admiral_merc_leader
					is_leader_tier = leader_tier_legendary
					has_trait = leader_trait_rift_warped
					AND = {
						has_trait_tier1or2 = { TRAIT = leader_trait_eager }
						has_base_skill < 4
					}
				}
			}
		}
		mult = 1
		mode = add
	}
}

promotes_mult_value_check_leader_cap = {
	base = 3 # 基础3容量
	complex_trigger_modifier = {
		trigger = check_modifier_value
		trigger_scope = owner
		parameters = {

	complex_trigger_modifier = {
		trigger = check_modifier_value
		trigger_scope = owner
		parameters = {
			modifier = country_leader_cap_add
		}
		mult = 1
		mode = add
	}
	floor = yes
	min = 1 # 别给狡猾人减到分母为0或负数了
}
