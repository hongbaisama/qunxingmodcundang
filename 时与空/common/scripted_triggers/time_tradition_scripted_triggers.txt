decision_time_trophy_trigger = {
	if = {
		limit = {
			has_modifier = $TROPHY$
		}
		event_target:time_trophy_target@owner = {
			NOT = { has_modifier = $TROPHY$ }
		}
	}
	else = {
		always = no
	}
}

		always = no
	}
}

	}

		}
