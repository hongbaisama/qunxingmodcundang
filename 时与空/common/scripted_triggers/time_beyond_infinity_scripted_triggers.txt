can_finish_crisobj_beyond_infinity_produced = {
	# 时间饱和容量为下一等级所需时间的2倍
	if = {
		limit = {
			has_beyond_infinity_level = { LEVEL = 3 }
		}
		always = yes
	}
	else_if = {
		limit = {
			has_beyond_infinity_level = { LEVEL = 2 }
		}
		check_variable_arithmetic = {
			which = value:crisobj_beyond_infinity_produced_cap_check_value
			value <= 17280
		}
	}
	else_if = {
		limit = {
			has_beyond_infinity_level = { LEVEL = 1 }
		}
		check_variable_arithmetic = {
			which = value:crisobj_beyond_infinity_produced_cap_check_value
			value <= 1440
		}
	}
	else = {
		check_variable_arithmetic = {
			which = value:crisobj_beyond_infinity_produced_cap_check_value
			value <= 48
		}
	}
}
