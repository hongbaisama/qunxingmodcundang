forbid_genetics_ascension = {
	OR = {
		is_robot_empire = yes
		is_individual_machine = yes
		is_natural_design_empire = yes
		has_country_flag = forbid_genetics_ascension # MOD接口
		check_modifier_value = {
			modifier = block_ap_engineered_evolution
			value > 0
		}
		if = {
			limit = {
				NOT = { has_global_flag = time_ascension_paths }
			}
			is_cyborg_empire = yes #锁义飞
		}
		else = {
			always = no
		}
	}
}

forbid_psionics_ascension = {
	OR = {
		if = {
			limit = {
				has_shroud_dlc = no
			}
			NAND = {
				is_gestalt = no
				OR = {
					is_mechanical_empire = no
					has_country_flag = halleberry_out # 魔女之旅
				}
				OR = {
					is_cyborg_empire = no
					AND = {
						has_origin = origin_unplugged
						NOT = { has_country_flag = is_cyberized }
					}
				}
				is_individual_machine = no
			}
		}
		else = {
			always = no
		}
		if = {
			limit = { has_origin = origin_shroud_forged }
			has_country_flag = finish_shroud_forged_control_flag
		}
		else = {
			always = no
		}
		is_natural_design_empire = yes
		has_origin = origin_mindwardens # 禁灵飞
		has_origin = origin_abandoned_by_god # 禁灵飞
		has_country_flag = forbid_psionics_ascension # MOD接口
		check_modifier_value = {
			modifier = block_ap_mind_over_matter
			value > 0
		}
	}
}
