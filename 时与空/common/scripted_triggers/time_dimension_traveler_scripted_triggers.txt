dimension_traveler_is_by_our_side = {
	if = {
		limit = {
			is_dimension_traveler_owner = yes
		}
		if = {
			limit = {
				OR = {
					has_country_flag = dimension_traveler_dance_with_death
					has_country_flag = dimension_traveler_returning
				}
			}
			always = no
		}
		else_if = {
			limit = {
				OR = {
					has_country_flag = dimension_traveler_official_active
					has_country_flag = dimension_traveler_scientist_active
					has_country_flag = dimension_traveler_commander_active
				}
			}
			dimension_traveler_is_our_leader = yes
		}
		else = {
			always = yes
		}
	}
	else = {
		always = no
	}
}

can_change_dimension_traveler_class = {
	custom_tooltip = {
		dimension_traveler_is_disappointed = no
		fail_text = dimension_traveler_is_disappointed_in_you
	}
	custom_tooltip = {
		dimension_traveler_is_by_our_side = yes
		fail_text = dimension_traveler_is_returning
	}
	custom_tooltip = {
		if = {
			limit = {
				exists = event_target:dimension_traveler_leader
			}
			event_target:dimension_traveler_leader = {
				is_exploring_astral_rift = no
			}
		}
		fail_text = dimension_traveler_is_exploring_astral_rift
	}
	custom_tooltip = {
		if = {
			limit = {
				exists = event_target:dimension_traveler_leader
			}
			dimension_traveler_is_researching_important_special_project = no
		}
		fail_text = dimension_traveler_is_researching_important_special_project
	}
	custom_tooltip = {
		if = {
			limit = {
				exists = event_target:dimension_traveler_leader
			}
			event_target:dimension_traveler_leader = {
				is_ruler = no
			}
		}
		fail_text = dimension_traveler_is_ruler
	}

can_print_dimension_traveler_hidden_log = {
	if = {
		limit = {
			is_gestalt = no
		}
		# 非格式塔需担任首席科学家
		exists = event_target:dimension_traveler_leader
		event_target:dimension_traveler_leader = {
			is_head_of_research = yes
		}
	}
	else = {
		# 格式塔需管理首都
		exists = event_target:dimension_traveler_leader
		event_target:dimension_traveler_leader = {
			exists = colony
			colony = { is_capital = yes }
		}
	}
}
