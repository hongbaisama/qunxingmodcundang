requires_origin_time_authority = {
	if = {
		limit = {
			has_country_flag = unlock_auth_time_democratic
		}
		custom_tooltip = {
			fail_text = requires_auth_time_democratic
			is_auth_time_democratic = yes
		}
	}
	else_if = {
		limit = {
			has_country_flag = unlock_auth_time_imperial
		}
		custom_tooltip = {
			fail_text = requires_auth_time_imperial
			is_auth_time_imperial = yes
		}
	}
	else_if = {
		limit = {
			has_country_flag = unlock_auth_time_corporate
		}
		custom_tooltip = {
			fail_text = requires_auth_time_corporate
			is_auth_time_corporate = yes
		}
	}
	else = {
		always = no
	}
}
