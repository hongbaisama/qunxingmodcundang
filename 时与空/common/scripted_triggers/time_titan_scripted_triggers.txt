	time_titan_is_upgrading = no
	time_titan_is_in_combat = no
	time_titan_is_not_mia = yes
	time_titan_is_in_time_titan_end_shelter_progress = no

		has_country_resource = {
			type = sr_dark_matter

		}
	}

# 经测试，只要舰船处于战斗状态或者失踪状态，无论判定can_go_mia = yes还是no，均返回false，否则返回true
# 任意（舰船未交战且未失踪，该条件返回true，很神奇吧？）（弃用）
time_titan_is_mia = {
	event_target:time_titan_fleet = {
		is_in_combat = no
		can_go_mia = no
	}
}

# 任意（舰船未交战且未失踪，该条件也返回true，很神奇吧？）（实际使用）
time_titan_is_not_mia = {
	event_target:time_titan_fleet = {
		OR = {
			is_in_combat = yes
			can_go_mia = yes
		}
	}
}

time_titan_exist_syzygy_admiral = {
	event_target:time_titan_fleet = {
		owner = {
			OR = {
				any_owned_leader = {
					is_syzygy_admiral = yes # 明日船长存活
				}
				has_country_flag = dimension_traveler_trait_foredoomed_to_a_rendezvous
				has_country_flag = dimension_traveler_trait_foredoomed_to_a_rendezvous_end
			}
		}
	}
}

	}
