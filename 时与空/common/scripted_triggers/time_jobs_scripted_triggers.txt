	owner = {
		has_origin_time = yes
		OR = {
			has_valid_civic = civic_technocracy
			has_valid_civic = civic_exalted_priesthood
			has_valid_civic = civic_aristocratic_elite
		}
	}
}

	colony? = {
		exists = owner
		is_homeworld = yes
		has_carrier_flag = original_gas_colony
		owner = {
			OR = {
				has_valid_civic = civic_origin_gas_colony
				has_valid_civic = civic_origin_gas_colony_corporate
				has_valid_civic = civic_origin_gas_colony_gestalt
			}
		}
	}
}
