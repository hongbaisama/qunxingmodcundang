TIME_RULE_trgr_spcs_CAN_live_on_planet = {
	custom_tooltip = {
		fail_text = "can_only_live_on_pc_hive"
		if = {
			limit = {
				has_trait = trait_hive_mind
				root.owner_or_space_owner = { has_origin = origin_pc_hive }
			}
			root = { uses_district_set = hive_world }
		}
	}
	custom_tooltip = {
		fail_text = "can_not_live_on_pc_time"
		if = {
			limit = {
				root = { is_origin_time_planet_class = yes }
			}
			is_origin_time_species = yes
		}
	}
	custom_tooltip = {
		fail_text = "can_only_live_on_pc_time"
		if = {
			limit = {
				has_trait = trait_time_soul_virtuality
			}
			root = { can_time_soul_virtuality_live_on = yes }
		}
	}
	custom_tooltip = {
		fail_text = "can_not_live_on_pc_gas_colony"
		if = {
			limit = {
				root = { is_planet_class = pc_gas_colony }
			}
			has_trait = trait_origin_gas_colony
		}
	}
	custom_tooltip = {
		fail_text = "can_only_live_on_pc_gas_colony"
		if = {
			limit = {
				has_trait = trait_origin_gas_colony
			}
			root = { is_planet_class = pc_gas_colony }
		}
	}
	custom_tooltip = {
		fail_text = "can_not_live_on_pc_psionic"
		if = {
			limit = {
				root = { is_planet_class = pc_psionic }
			}
			has_trait = trait_origin_psionic
		}
	}
	custom_tooltip = {
		fail_text = "can_only_live_on_pc_psionic"
		if = {
			limit = {
				has_trait = trait_origin_psionic
			}
			root = { is_planet_class = pc_psionic }
		}
	}
	custom_tooltip = {
		fail_text = "can_not_live_on_pc_nanites"
		if = {
			limit = {
				root = { is_planet_class = pc_nanites }
			}
			has_trait = trait_origin_nanites
		}
	}
	custom_tooltip = {
		fail_text = "can_only_live_on_pc_nanites"
		if = {
			limit = {
				has_trait = trait_origin_nanites
			}
			root = {
				OR = {
					is_planet_class = pc_nanites
					is_planet_class = pc_nanotech
				}
			}
		}
	}
	custom_tooltip = {
		fail_text = "can_only_live_on_pc_volcanic"
		if = {
			limit = {
				has_trait = trait_wonder_country_96
			}
			root = { is_planet_class = pc_volcanic }
		}
	}
	custom_tooltip = {
		fail_text = "can_not_live_on_pc_time_notofthisworld"
		if = {
			limit = {
				root = { is_planet_class = pc_time_notofthisworld }
			}
			has_trait_notofthisworld = yes
		}
	}
	custom_tooltip = {
		fail_text = "building_blessed_by_god_reject_robot"
		if = {
			limit = {
				root = { has_building = building_blessed_by_god }
			}
			is_robotic = no
		}
	}
	custom_tooltip = {
		fail_text = "circle_nexus_is_defective"
		if = {
			limit = {
				OR = {
					has_global_flag = greatest_dimensional_horror_crisis
					has_global_flag = circle_nexus_is_defective
				}
			}
			if = {
				limit = {
					has_trait = trait_time_1
				}
				root = {
					OR = {
						is_homeworld = yes
						has_modifier = time_1_planet_modifier
					}
				}
			}
			else_if = {
				limit = {
					has_trait = trait_time_2
				}
				root = {
					OR = {
						is_homeworld = yes
						has_modifier = time_2_planet_modifier
					}
				}
			}
			else = {
				root = { is_homeworld = yes }
			}
		}
	}
	# 以防万一
	custom_tooltip = {
		fail_text = "can_not_live_on_any_planet"
		if = {
			limit = {
				has_trait = trait_dimension_traveler
			}
			always = no
		}
	}
}

species_rights_robot_full_rights_check = {
	exists = from
	if = {
		limit = {
			from = { has_origin = origin_adeptus_mechanicus }
		}
		OR = {
			has_citizenship_type = { type = citizenship_adeptus_mechanicus country = from }
			from = { has_policy_flag = ai_full_rights }
		}
	}
	else = {
		from = {
			OR = {
				has_policy_flag = ai_full_rights
				is_mechanical_empire = yes
				has_origin = origin_synthetic_fertility
			}
		}
	}
}

is_free_robot = { # The pop is a free robot with citizen rights
	has_trait = trait_mechanical
	exists = owner
	if = {
		limit = {
			owner = { has_origin = origin_adeptus_mechanicus }
		}
		OR = {
			has_citizenship_type = { type = citizenship_adeptus_mechanicus country = owner }
			owner = {
				has_technology = tech_synthetic_workers
				has_policy_flag = ai_full_rights
			}
		}
	}
	else = {
		owner = {
			has_technology = tech_synthetic_workers
			OR = {
				has_policy_flag = ai_full_rights
				is_mechanical_empire = yes
			}
		}
	}
}
