cb_dark_forest = {
	show_in_diplomacy = no
	show_notification = no
	potential = {
		has_global_flag = dark_forest
	}
	is_valid = {
		NOT = { is_overlord_to = FROM }
		OR = {
			is_subject = no
			overlord = { NOT = { is_same_value = from } }
		}
	}
	proxy_war_resources = {
		category = proxy_war
		cost = {
			influence = 200
		}
	}
	on_proxy_war_start = {
		pay_proxy_war_additional_resources = { VALUE = -50 }
	}
}

cb_bright_pearl = {
	show_in_diplomacy = yes
	show_notification = yes
	potential = {
		is_default_or_player = yes
		NOT = { is_in_federation_with = from }
		exists = event_target:bright_pearl_system
	}
	is_valid = {
		if = {
			limit = {
				exists = event_target:bright_pearl_system
			}
			event_target:bright_pearl_system = {
				owner = {
					OR = {
						is_same_value = FROM
						has_overlord = FROM
					}
				}
			}
			has_total_war_cb = no
			NOT = { is_overlord_to = FROM }
			OR = {
				is_subject = no
				overlord = { NOT = { is_same_value = from } }
			}
		}
		else = {
			always = no
		}
	}
	proxy_war_resources = {
		category = proxy_war
		cost = {
			influence = 200
		}
	}
	on_proxy_war_start = {
		pay_proxy_war_additional_resources = { VALUE = -50 }
	}
}

	is_valid = {
		has_total_war_cb = no

		NOT = { is_overlord_to = FROM }

			is_subject = no
			overlord = { NOT = { is_same_value = from } }
		}
		from = {
			if = {
				limit = {
					NOT = { has_technology = tech_colossus }
				}
				event_target:origin_colossus_fleet = {
					is_controlled_by = prev
				}
			}
			else = {
				event_target:origin_colossus_fleet = {
					is_ship_size = star_eater
					is_controlled_by = prev
				}
			}
		}
		NOT = {
			any_controlled_fleet = {
				is_ship_class = shipclass_military_special
				any_controlled_ship = { is_ship_size = colossus }
			}
		}
	}
	proxy_war_resources = {
		category = proxy_war
		cost = {
			influence = 200
		}
	}
	on_proxy_war_start = {
		pay_proxy_war_additional_resources = { VALUE = -50 }
	}
}

cb_time_name_of_hope = {
	show_in_diplomacy = yes
	show_notification = no
	potential = {
		if = {
			limit = {
				has_country_flag = upgrade_to_time_titan_finished
			}
			if = {
				limit = {
					# 拥有希望使者
					event_target:time_titan_fleet = { is_controlled_by = prev }
				}
				# 希望之名科技
				has_technology = tech_hope_time_titan
				# 非附庸
				is_subject = no
				# 非同联邦
				exists = from
				NOT = { is_in_federation_with = from }
			}
			else = {
				always = no
			}
		}
		else = {
			always = no
		}
	}
	is_valid = {
		from = { is_default_or_player = yes }
		# 非格式塔
		from = { NOT = { has_ethic = ethic_gestalt_consciousness } }
		NOT = { has_ethic = ethic_gestalt_consciousness }
		has_total_war_cb = no
		OR = {
			AND = {
				is_xenophile = yes
				from = { is_xenophile = no }
			}
			AND = {
				is_xenophobe = yes
				from = { is_xenophobe = no }
			}
			AND = {
				is_authoritarian = yes
				from = { is_authoritarian = no }
			}
			AND = {
				is_egalitarian = yes
				from = { is_egalitarian = no }
			}
			AND = {
				is_spiritualist = yes
				from = { is_spiritualist = no }
			}
			AND = {
				is_materialist = yes
				from = { is_materialist = no }
			}
			AND = {
				is_militarist = yes
				from = { is_militarist = no }
			}
			AND = {
				is_pacifist = yes
				from = { is_pacifist = no }
			}
		}
		NOT = { is_overlord_to = FROM }
	}
	proxy_war_resources = {
		category = proxy_war
		cost = {
			influence = 100
		}
	}
	on_proxy_war_start = {
		pay_proxy_war_additional_resources = { VALUE = -25 }
	}
}
