# 教化所星球
col_adeptus_mechanicus = {
	icon = GFX_colony_type_bureaucratic

	potential = {
		exists = space_owner
		space_owner = { has_origin = origin_adeptus_mechanicus }
		NOT = { uses_district_set = habitat }
		is_special_colony_type = no
		is_capital = no
	}

	planet_modifier = {
		pop_adeptus_mechanicus_bonus_workforce_mult = 0.10
		planet_pop_assembly_mult = 0.1
	}

	triggered_planet_modifier = {
		potential = {
			exists = owner_or_space_owner
			owner_or_space_owner = { has_tradition = tr_adaptability_finish }
		}
		pop_adeptus_mechanicus_bonus_workforce_mult = 0.05
	}

	ai_building_set_affinity = {
		adeptus_mechanicus = 2.0
	}

	weight_modifier = {
		modifier = {
			add = 10
			has_designation = col_adeptus_mechanicus
		}
		inline_script = {
			script = colony_types/shared_weights_adeptus_mechanicus
		}
	}
}
