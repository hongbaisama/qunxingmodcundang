decision_care_economy = {
	owned_planets_only = yes
	sound = event_construction
	icon = decision_luxuries
	
	prerequisites = {
	}
	
	enactment_time = 0
	
	potential = {
		exists = leader
		leader = {
			has_trait = leader_trait_planetary_mechanocalibrator
			NOT = { has_leader_flag = care_economy }
		}
	}
	
	effect = {
		custom_tooltip = decision_care_tooltip
		custom_tooltip = care_economy_effect
		hidden_effect = {
			leader = {
				clear_care_flag = yes
				set_leader_flag = care_economy
			}
		}
	}
	
	ai_weight = {
		weight = 100
	}
}
