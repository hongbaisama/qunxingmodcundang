agenda_tradition_origin_pure_brain = {
	agenda_cost = 7000
	allow = {
		has_tradition = tr_origin_pure_brain_adopt
	}
	potential = {
		OR = {
			has_origin = origin_pure_brain
			has_origin = origin_pure_brain_true
		}
	}
	modifier = {
		planet_stability_add = 2.5
	}
	finish_modifier = agenda_tradition_origin_pure_brain_finish
	effect = {
		custom_tooltip = council_agenda_agenda_gestalt_purge_traits_effect_desc
		every_owned_leader = {
			if = {
				# Remove Exp per Trait being removed.
				limit = {
					is_councilor = yes
					num_leader_traits = {
						value > 0
						negative = yes						# Tiers don't matter in this check.
					}
				}
				hidden_effect = {
					add_experience = value:purge_traits_exp_change_total					# -300 per trait, not counting tiers.
					# Restore pending traits and tiers, so they'll come back eventually. But a 10% chance to reduce their numbers.
					change_variable = {
						which = leader_pending_negative_traits_unmodified
						value = value:leader_num_negative_leader_trait_tiers
					}
					random = {
						chance = 10
						subtract_variable = {
							which = leader_pending_negative_traits_unmodified
							value = 1
						}
					}
				}
				remove_trait = all_negative
			}
		}
	}
}

agenda_time_soul = {
	agenda_cost = value:agenda_tradition_horror_time_cost|BASE|10000|MIN|1000|

	potential = {
		has_ascension_perk = ap_origin_time
		has_origin_time_ruler = yes
		NOT = { has_country_flag = agenda_time_soul_finished }
	}

	allow = {
		custom_tooltip = {
			fail_text = "agenda_time_soul_requires_trait_time"
			owner_main_species = {
				has_trait = trait_time
			}
		}
		if = {
			limit = {
				has_tradition = tr_synthetics_finish
			}
			if = {
				limit = {
					has_country_flag = synth_virtual
				}
				custom_tooltip = {
					fail_text = "agenda_time_soul_requires_virtual_species"
					owner_main_species = {
						has_virtual_species_trait = yes
					}
				}
			}
			else = {
				custom_tooltip = {
					fail_text = "agenda_time_soul_requires_robotic_species"
					owner_main_species = {
						is_robotic = yes
					}
				}
			}
		}
		else = {
			custom_tooltip = {
				fail_text = "agenda_time_soul_requires_organic_species"
				owner_main_species = {
					is_robotic = no
				}
			}
		}
		if = {
			limit = {
				is_auth_time_democratic = yes
			}
			event_target:origin_time_ruler = { is_councilor = yes }
		}
		else = {
			event_target:origin_time_ruler = { is_ruler = yes }
		}
	}

	effect = {
		switch = {
			trigger = has_tradition
			tr_synthetics_finish = {
				switch = {
					trigger = has_country_flag
					synth_virtual = {
						custom_tooltip = council_agenda_agenda_time_soul_effect_virtuality

					synth_physical = {

					}
					default = {
						custom_tooltip = council_agenda_agenda_time_soul_effect_synthetic
					}
				}
			}
			tr_cybernetics_finish = {
				custom_tooltip = council_agenda_agenda_time_soul_effect_cybernetic
			}
			tr_psionics_finish = {
				custom_tooltip = council_agenda_agenda_time_soul_effect_psionic
			}
			tr_psionics_shroud_finish = {
				custom_tooltip = council_agenda_agenda_time_soul_effect_psionic
			}
			default = {
				if = {
					limit = {
						OR = {
							has_tradition = tr_genetics_finish
							has_tradition = tr_purity_finish
							has_tradition = tr_cloning_finish
							has_tradition = tr_mutation_finish
						}
					}
					custom_tooltip = council_agenda_agenda_time_soul_effect_genetic
				}
			}
		}
		hidden_effect = {
			set_country_flag = agenda_time_soul_finished
			create_timeline_agenda_time_soul = yes
			switch = {
				trigger = has_tradition
				tr_synthetics_finish = {
					switch = {
						trigger = has_country_flag
						synth_virtual = {
							agenda_time_soul_effect = {
								TYPE = virtuality
								ROBOTIC = yes
							}
						}
						synth_physical = {
							agenda_time_soul_effect = {
								TYPE = synthetic
								ROBOTIC = yes
							}
						}
						default = {
							agenda_time_soul_effect = {
								TYPE = synthetic
								ROBOTIC = yes
							}
						}
					}
				}
				tr_cybernetics_finish = {
					agenda_time_soul_effect = {
						TYPE = cybernetic
						ROBOTIC = no
					}
				}
				tr_psionics_finish = {

						TYPE = psionic

					}
				}

								has_tradition = tr_cloning_finish

							}
