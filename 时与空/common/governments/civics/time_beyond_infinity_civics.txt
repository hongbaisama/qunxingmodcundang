civic_time_machine = {
	description = "civic_tooltip_time_machine_effects"
	negative_description = "civic_tooltip_beyond_infinity_negative_effects"
	ai_playable = { always = no }
	potential = {
		ethics = {
			NOT = { value = ethic_gestalt_consciousness }
		}
		authority = {
			NOT = { value = auth_corporate }
		}
		civics = {
			OR = {
				value = civic_time_machine
				value = civic_time_machine_corporate
				value = civic_time_machine_hive
				value = civic_time_machine_machine
			}
		}
	}
	possible = {
		is_nomadic = no
		civics = {
			NOR = {
				text = civic_tooltip_not_circle_nexus
				value = civic_circle_nexus
				value = civic_circle_nexus_corporate
				value = civic_circle_nexus_hive
				value = civic_circle_nexus_machine
			}
			NOR = {
				text = civic_tooltip_not_dimension_engine
				value = civic_dimension_engine
				value = civic_dimension_engine_corporate
				value = civic_dimension_engine_hive
				value = civic_dimension_engine_machine
			}
			NOR = {
				text = civic_tooltip_not_genocidal
				value = civic_fanatic_purifiers
				value = civic_hive_devouring_swarm
				value = civic_machine_terminator
				value = civic_scorched_earth
				value = civic_hive_scorched_earth
			}
		}
	}
	modification = {
		add = {
			hidden_trigger = {
				has_civic = civic_time_machine_corporate
			}
		}
		remove = {
			always = no
		}
	}
	alternate_civic_version = civic_time_machine_corporate
	random_weight = { base = 0 }
	ai_weight = { base = 0 }
