declare_war_to_galaxy = {
	# 保持独立
	if = {
		limit = {
			is_subject = yes
		}
		set_subject_of = { who = none }
	}
	# 释放附庸
	if = {
		limit = {
			is_overlord = yes
		}
		every_subject = {
			set_subject_of = { who = none }
		}
	}
	# 退出联邦
	if = {
		limit = {
			has_federation = yes
		}
		leave_alliance = { override_requirements = yes }
	}
	# 退出共同体
	if = {
		limit = {
			is_galactic_community_member = yes
		}
		remove_from_galactic_community = yes
	}
	################################################################
	# 宣战全银河
	################################################################
	save_event_target_as = declarer
	# 无法向已经与自身交战的国家发起新的战争，必须先退出战争
	every_war = {
		remove_war_participant = event_target:declarer
	}
	# 抽取宣战目标
	if = {
		limit = {
			has_galactic_emperor = yes
			galactic_emperor = {
				NOT = { is_same_value = event_target:declarer }
			}
		}
		galactic_emperor = {
			establish_communications_no_message = event_target:declarer
			save_event_target_as = defender
		}
	}
	else_if = {
		limit = {
			has_galactic_custodian = yes
			galactic_custodian = {
				NOT = { is_same_value = event_target:declarer }
			}
		}
		galactic_custodian = {
			establish_communications_no_message = event_target:declarer
			save_event_target_as = defender
		}
	}
	else = {
		random_country = {
			limit = {
				OR = {
					is_default_or_player = yes
					is_fallen_empire = yes
				}
				NOT = { is_same_value = event_target:declarer }
			}
			establish_communications_no_message = event_target:declarer
			save_event_target_as = defender
		}
	}
	# 宣战
	declare_war = {
		target = event_target:defender
		attacker_war_goal = wg_time_become_end_crisis
		effect = { save_event_target_as = time_become_end_crisis_war }
	}
	# 其它国家加入战争
	every_country = {
		limit = {
			OR = {
				is_default_or_player = yes
				is_fallen_empire = yes
			}
			NOR = {
				is_same_value = event_target:declarer
				is_same_value = event_target:defender
			}
		}
		establish_communications_no_message = event_target:declarer
		join_war_on_side = {
			war = event_target:time_become_end_crisis_war
			side = defenders
		}
	}
	# 国家无法支持与自身交战的国家的战争，对漏网之鱼单独宣战
	every_country = {
		limit = {
			OR = {
				is_default_or_player = yes
				is_fallen_empire = yes
			}
			NOR = {
				is_same_value = event_target:declarer
				is_at_war_with = event_target:declarer
			}
		}
		establish_communications_no_message = event_target:declarer
		event_target:declarer = {
			declare_war = {
				target = prev
				attacker_war_goal = wg_time_become_end_crisis
			}
		}
	}
}

generate_time_mod_advanced_bonus = {
	set_country_flag = gain_time_mod_advanced_bonus
	
	give_physics_tech_tier_1_effect_no_message = yes
	give_physics_tech_tier_2_effect_no_message = yes
	give_society_tech_tier_1_effect_no_message = yes
	give_society_tech_tier_2_effect_no_message = yes
	give_engineering_tech_tier_1_effect_no_message = yes
	give_engineering_tech_tier_2_effect_no_message = yes
	
	while = {
		count = $COUNT|1$
		add_resource = {
			energy = 10000
			minerals = 10000
			food = 10000
			alloys = 5000
			consumer_goods = 5000
			volatile_motes = 500
			rare_crystals = 500
			exotic_gases = 500
			unity = 25000
			influence = 250
		}
		if = {
			limit = {
				is_ai = no
			}
			every_owned_leader = {
				add_skill_without_trait_selection = 3
			}
			every_pool_leader = {
				add_skill_without_trait_selection = 3
			}
		}
		else = {
			every_owned_leader = {
				add_skill = 3
			}
			every_pool_leader = {
				add_skill = 3
			}
		}
	}
	
	add_modifier = {
		modifier = new_country_others_modifier
		days = 1800
	}
	
	generate_time_mod_advanced_system_bonus = yes
	
	every_owned_colony = {
		# 障碍
		every_deposit = {
			limit = {
				is_blocker = yes
				has_deposit_category = deposit_cat_blockers_natural
				NOR = {
					has_deposit_flag = keep_deposit_on_terraform
					is_deposit_type = d_time_blocker_0
					is_deposit_type = d_time_blocker_1
					is_deposit_type = d_time_blocker_2
					is_deposit_type = d_time_blocker_3
				}
			}
			clear_blocker = yes
		}
		# 区划
		if = {
			limit = {
				num_free_districts = {
					type = district_city
					value > 0
				}
			}
			while = {
				count = 3
				add_district_and_planet_size_if_needed_effect = {
					district = district_city
				}
			}
		}
		else_if = {
			limit = {
				num_free_districts = {
					type = district_hive
					value > 0
				}
			}
			while = {
				count = 3
				add_district_and_planet_size_if_needed_effect = {
					district = district_hive
				}
			}
		}
		else_if = {
			limit = {
				num_free_districts = {
					type = district_nexus
					value > 0
				}
			}
			while = {
				count = 3
				add_district_and_planet_size_if_needed_effect = {
					district = district_nexus
				}
			}
		}
		# 人口
		while = {
			count = 25
			create_pop_group = {
				species = owner.species
			}
		}
	}
}
