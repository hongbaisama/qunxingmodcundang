randomly_add_horror_time_tech_option = {
	if = {
		limit = {
			NOR = {
				has_tech_option = $TECH$
				has_technology = $TECH$
			}
		}
		random = {
			chance = value:horror_time_tech_weight|LEVEL|$LEVEL$|TYPE|$TYPE$|
			add_research_option = $TECH$
		}
	}
}

active_horror_tech = {
	if = {
		limit = {
			last_increased_tech = tech_horror_modifier
		}
		capital_scope = {
			set_carrier_flag = time_gain_horror
			remove_modifier = horror_study_trophy
			remove_modifier = horror_deify_trophy
			remove_modifier = horror_study_trophy2
			remove_modifier = horror_deify_trophy2
			remove_modifier = horror_too_late
			remove_modifier = horror_stabilized
			add_modifier = { modifier = horror_stabilized }
		}
	}
	else_if = {
		limit = {
			last_increased_tech = tech_horror_dimension
		}
		if = {
			limit = {
				capital_scope = {
					NOT = { has_deposit = d_portal_research_zone }
				}
			}
			capital_scope = {
				add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_portal_research_zone }
				set_carrier_flag = tech_horror_dimension_planet
				save_global_event_target_as = tech_horror_dimension_planet_of@root
			}
		}
	}
	else_if = {
		limit = {
			last_increased_tech = tech_horror_5_a_1
		}
		event_target:tech_horror_dimension_planet_of@root = {
			set_carrier_flag = dportal_energy
		}
	}
	else_if = {
		limit = {
			last_increased_tech = tech_horror_5_a_2
		}
		event_target:tech_horror_dimension_planet_of@root = {
			set_carrier_flag = dportal_trader
			add_modifier = {
				modifier = "dportal_trade_crisis"
				days = -1
			}
		}
	}
	else_if = {
		limit = {
			last_increased_tech = tech_horror_5_b_1
		}
		event_target:tech_horror_dimension_planet_of@root = {
			set_carrier_flag = dportal_hell #负面效果
		}
		add_tech_option_or_research_effect = {
			TECH = tech_psionic_theory
			PROGRESS = 0.5
			CATEGORY = society_research
		}
		while = {
			count = value:add_random_psionic_tech_option_count
			random_list = {
				1 = {
					inline_script = {
						script = weights/add_random_psionic_tech_option
						TECH = tech_precognition_interface
					}
				}
				1 = {
					inline_script = {
						script = weights/add_random_psionic_tech_option
						TECH = tech_psi_jump_drive_1
					}
				}
				1 = {
					inline_script = {
						script = weights/add_random_psionic_tech_option
						TECH = tech_psionic_shield
					}
				}
				1 = {
					inline_script = {
						script = weights/add_random_psionic_tech_option
						TECH = tech_psionic_bombers
					}
				}
				1 = {
					inline_script = {
						script = weights/add_random_psionic_tech_option
						TECH = tech_psionic_lightning
					}
				}
				1 = {
					inline_script = {
						script = weights/add_random_psionic_tech_option
						TECH = tech_psionic_disruptor
					}
				}
				1 = {
					inline_script = {
						script = weights/add_random_psionic_tech_option
						TECH = tech_zro_launcher
					}
				}
			}
		}
	}
	else_if = {
		limit = {
			last_increased_tech = tech_horror_5_b_2
		}
		event_target:tech_horror_dimension_planet_of@root = {
			set_carrier_flag = dportal_void
			set_carrier_flag = dportal_void2
		}
	}
	else_if = {
		limit = {
			last_increased_tech = tech_horror_trait
		}
		event_target:tech_horror_dimension_planet_of@root = {
			every_owned_pop_group = {
				make_notofthisworld = yes
			}
		}
	}
}

active_time_tech = {
	if = {
		limit = {
			last_increased_tech = tech_time_modifier
		}
		set_country_flag = horrific_celebration # 禁用恐怖负相物质游行
		capital_scope = {
			set_carrier_flag = time_gain_horror
			set_carrier_flag = exists_d_time_horror
			remove_modifier = horror_study_trophy
			remove_modifier = horror_deify_trophy
			remove_modifier = horror_study_trophy2
			remove_modifier = horror_deify_trophy2
			remove_modifier = horror_too_late
			remove_modifier = horror_stabilized
			add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_time_horror }
		}
		if = {
			limit = {
				is_spiritualist = no
			}
			every_owned_colony = {
				limit = {
					NOT = { has_carrier_flag = time_gain_horror }
					has_horror_trophy = no
					is_under_colonization = no
					carrier_is_type = planet # 不会在方舟舰上扩散
				}
				set_carrier_flag = time_gain_horror
				add_modifier = { modifier = horror_study_trophy }
				carrier_event = { id = situation.1085 days = 2340 random = 280 }
			}
		}
		else = {
			every_owned_colony = {
				limit = {
					NOT = { has_carrier_flag = time_gain_horror }
					has_horror_trophy = no
					is_under_colonization = no
					carrier_is_type = planet # 不会在方舟舰上扩散
				}
				set_carrier_flag = time_gain_horror
				add_modifier = { modifier = horror_deify_trophy }
				carrier_event = { id = situation.1085 days = 2340 random = 280 }
			}
		}
	}
}
