transfer_akx_waiting_world_pop_to_akx_messenger = {
	set_variable = {
		which = akx_waiting_world_pop_transfer_percentage
		value = value:akx_waiting_world_pop_transfer_percentage|PERCENTAGE|$PERCENTAGE$|
	}
	every_owned_species = {
		limit = {
			has_akx_waiting_world = yes
			has_akx_messenger = no
		}
		save_event_target_as = old_akx_waiting_world_species
		modify_species = {
			species = this
			remove_trait = trait_akx_waiting_world
			remove_trait = trait_akx_waiting_world_robot
			add_trait = trait_akx_messenger
			change_scoped_species = no
			effect = {
				save_event_target_as = new_akx_messenger_species
			}
		}
		prev = {
			create_pop_group = {
				species = event_target:new_akx_messenger_species
				size = 0
				effect = {
					save_event_target_as = new_akx_messenger_pop_group
				}
			}
			every_owned_pop_group = {
				limit = {
					species = { is_same_value = event_target:old_akx_waiting_world_species }
				}
				transfer_pop_amount = {
					source = this
					target = event_target:new_akx_messenger_pop_group
					percentage = prev.akx_waiting_world_pop_transfer_percentage
				}
			}
		}
	}
	clear_variable = akx_waiting_world_pop_transfer_percentage
}

# 星系（清除戴森云、戴森球、奇迹球并返还建造次数）
worm_clear_dyson_sphere = {
	every_system_megastructure = {
		limit = {
			if = {
				limit = {
					OR = {
						is_time_dyson_swarm = yes
						is_time_dyson_sphere = yes
						is_time_dyson_sphere_ruined = yes
						is_time_dyson_sphere_disco = yes
					}
				}
				if = {
					limit = {
						exists = planet
					}
					planet = { is_time_black_hole_or_white_hole_planet_class = yes }
				}
			}
			else = {
				always = no
			}
		}
		if = {
			limit = {
				is_time_dyson_swarm = yes
			}
			if = {
				limit = { exists = owner }
				owner = {
					if = {
						limit = {
							check_variable = {
								which = dyson_swarm_counter
								value > 0
							}
						}
						subtract_variable = {
							which = dyson_swarm_counter
							value = 1
						}
					}
				}
			}
		}
		else_if = {
			limit = {
				is_time_dyson_sphere = yes
			}
			if = {
				limit = { exists = owner }
				owner = { remove_country_flag = built_dyson_sphere }
			}
			solar_system = { remove_star_flag = dyson_sphere_built }
		}
		else_if = {
			limit = {
				is_time_dyson_sphere_ruined = yes
			}
			solar_system = { remove_star_flag = dyson_sphere_built }
		}
		else_if = {
			limit = {
				is_time_dyson_sphere_disco = yes
			}
			if = {
				limit = { exists = owner }
				owner = { set_country_flag = can_build_disco_sphere }
			}
		}
		planet = { remove_carrier_flag = has_megastructure }
		delete_megastructure = this
	}
	if = {
		limit = {
			any_system_megastructure = {
				is_time_dyson_swarm = yes
			}
		}

	}

				which = r_time_count
				value >= 6
			}
		}
		add_r_time_modifier = { LEVEL = 6 }
	}

		limit = {
			check_variable = {
				which = r_time_count
				value >= 5
			}
		}
		add_r_time_modifier = { LEVEL = 5 }
	}
	else_if = {
		limit = {
			check_variable = {
				which = r_time_count
				value >= 4
			}
		}
		add_r_time_modifier = { LEVEL = 4 }
	}
	else_if = {
		limit = {
			check_variable = {
				which = r_time_count
				value >= 3
			}
		}
		add_r_time_modifier = { LEVEL = 3 }
	}
	else_if = {
		limit = {
			check_variable = {
				which = r_time_count
				value >= 2
			}
		}
		add_r_time_modifier = { LEVEL = 2 }
	}
	else = {
		add_r_time_modifier = { LEVEL = 1 }
	}
	custom_tooltip = r_time_effect
	if = {
		limit = {
			has_edict = akx_shorten_r_time_cooldown

				which = r_time_count
				value = 1
			}

		}

			}

				modifier = relic_activation_cooldown

			}
		}
		else = {
			add_modifier = {
				modifier = relic_activation_cooldown
				days = 3600
			}
		}
	}
