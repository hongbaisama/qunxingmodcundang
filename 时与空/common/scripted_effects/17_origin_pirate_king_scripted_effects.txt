# 国家
gain_origin_pirate_king_reward = {
	# 赏金
	if = {
		limit = {
			OR = {
				NOT = { exists = event_target:pirate_king_situation }
				event_target:pirate_king_situation = {
					current_situation_approach = approach_origin_pirate_king_default
				}
			}
		}
		add_resource = {
			energy = $energy|0$
			minerals = $minerals|0$
			food = $food|0$
			alloys = $alloys|0$
			consumer_goods = $consumer_goods|0$
			physics_research = $physics_research|0$
			society_research = $society_research|0$
			engineering_research = $engineering_research|0$
			unity = $unity|0$
			sr_dark_matter = $sr_dark_matter|0$
			astral_threads = $astral_threads|0$
			nanites = $nanites|0$
			mult = value:pirate_king_situation_resource_value
		}
	}
	# 恶名
	if = {
		limit = {
			exists = event_target:pirate_king_situation
		}
		event_target:pirate_king_situation = {
			if = {
				limit = {
					current_situation_approach = approach_origin_pirate_king_default
				}
				add_situation_progress = value:pirate_king_situation_progress_value|PROGRESS|$progress$|
			}
			else = {
				add_situation_progress = value:pirate_king_situation_progress_value|PROGRESS|$progress$|MULT|2|
			}
		}
	}
}

gain_leader_trait_origin_pirate = {
	switch = {
		trigger = leader_class
		official = {
			add_trait = {
				trait = leader_trait_origin_pirate_king_official
				show_message = no
			}
		}
		scientist = {
			add_trait = {
				trait = leader_trait_origin_pirate_king_scientist
				show_message = no
			}
		}
		commander = {
			add_trait = {
				trait = leader_trait_origin_pirate_king_commander
				show_message = no
			}
		}
	}
}

# 局势
choose_pirate_king_hostile_fleet_location = {
	# 优先选择“黑珍珠号”所在星系
	if = {
		limit = {
			exists = event_target:pirate_king_arkship
			event_target:pirate_king_arkship = { exists = solar_system }
		}
		event_target:pirate_king_arkship = {
			solar_system = {
				save_event_target_as = pirate_king_hostile_fleet_system
				random_system_planet = {
					save_event_target_as = pirate_king_hostile_fleet_location
				}
			}
		}
	}
	# “黑珍珠号”不存在星系域时（航道中/失踪）的保底位置
	else = {
		owner = {
			random_system_within_border = {
				save_event_target_as = pirate_king_hostile_fleet_system
				random_system_planet = {
					save_event_target_as = pirate_king_hostile_fleet_location
				}
			}
		}
	}
}

create_pirate_king_end_species = {
	if = {
		limit = {
			root.owner = {
				has_origin_pirate_king = yes
				owner_main_species = { is_archetype = MACHINE }
			}
		}
		create_species = {
			name = "NAME_Pirate_King_Species"
			class = MACHINE
			portrait = nec_machine
			traits = {
				ideal_planet_class = pc_ark
				trait = trait_machine_unit
				trait = trait_origin_pirate_king_end
				trait = trait_robot_history_warbot
				trait = trait_robot_enhanced_memory
				trait = trait_robot_integrated_weaponry
				trait = trait_robot_quarrelsome
				trait = trait_robot_high_maintenance
			}
			immortal = yes
			effect = {
				save_global_event_target_as = pirate_king_end_species
			}
		}
	}
	else = {
		create_species = {
			name = "NAME_Pirate_King_Species"
			class = NECROID
			portrait = random
			traits = {
				ideal_planet_class = pc_ark
				trait = trait_organic
				trait = trait_origin_pirate_king_end
				trait = trait_genetic_memory
				trait = trait_spare_organs
				trait = trait_acidic_vascularity
				trait = trait_quarrelsome
				trait = trait_brittle
			}
			immortal = yes
			effect = {
				save_global_event_target_as = pirate_king_end_species
			}
		}
	}
	if = {
		limit = {
			root.owner = { has_origin_pirate_king = yes }
		}
		root.owner = {
			owner_main_species = {
				event_target:pirate_king_end_species = {
					set_species_identity = prev
				}
			}
		}
	}
}
