leader_finish_ap_horror_project = {
	# 奥秘项目标识
	set_leader_flag = leader_finish_ap_horror_project_$TYPE$
	# 添加奥秘特质
	if = {
		limit = {
			is_dimension_traveler = yes
			owner = { has_country_flag = dimension_traveler_end }
		}
		# 无事发生
	}
	else_if = {
		limit = {
			is_dimension_traveler = yes
			has_finished_all_ap_horror_project = yes
		}
		add_trait = {
			trait = leader_trait_dimension_traveler_horror_secret
		}
	}
	else = {
		# 提示玩家获得新特质
		remove_trait = leader_trait_horror_secret
		add_trait = {
			trait = leader_trait_horror_secret
		}
	}
	# 领袖升级
	add_skill_without_trait_selection = 1
	remove_all_negative_traits = yes
	# 界旅者额外
	if = {
		limit = {
			is_dimension_traveler = yes
		}
		owner = {
			set_country_flag = dimension_traveler_trait_$TYPE$_secret
		}
	}
}

}

# 领袖

	remove_leader_flag = has_tried_to_promote_to_renowned_leader
	remove_leader_flag = has_tried_to_promote_to_legendary_leader
	if = {
		limit = {
			OR = {
				is_leader_tier = leader_tier_default
				is_leader_tier = leader_tier_renowned
			}
		}
		set_leader_tier = leader_tier_legendary
	}
	gain_leader_trait_inborn_legendary = yes
}
