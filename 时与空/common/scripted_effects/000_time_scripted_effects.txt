make_time_mod_origin_unique = {
	if = {
		# 玩家优先
		limit = {
			any_playable_country = {
				has_origin = $ORIGIN$
				is_ai = no
			}
		}
		random_playable_country = {
			limit = {
				has_origin = $ORIGIN$
				is_ai = no
			}
			country_event = { id = $EVENT$ }
			set_country_flag = true_$ORIGIN$_country
		}
	}
	else_if = {
		# 仅AI时
		limit = {
			any_playable_country = {
				has_origin = $ORIGIN$
			}
		}
		random_playable_country = {
			limit = {
				has_origin = $ORIGIN$
			}
			country_event = { id = $EVENT$ }
			set_country_flag = true_$ORIGIN$_country
		}
	}
	# 其它同起源改繁荣一统
	every_playable_country = {
		limit = {
			has_origin = $ORIGIN$
			NOT = { has_country_flag = true_$ORIGIN$_country }
		}
		set_origin_default_effect = yes
	}
}

		limit = {
			is_time_black_hole_or_white_hole_star_class = no

		set_star_class = sc_black_hole
	}
	if = {
		limit = {
			NOT = { is_same_value = event_target:dimensional_horror_crisis_destiny_system }
		}
		set_name = NAME_Akx_Time_Erasured
	}

		modifier = akx_time_erasure_system_modifier

	}
	if = {
		limit = {

		}
		set_asteroid_belt = {
			radius = 0
		}
	}
	every_system_planet = {
		limit = {
			NOR = {
				is_origin_time_planet_class = yes
				is_time_white_hole_planet_class = yes
				has_carrier_flag = dimensional_horror_crisis_changed
			}
		}
		# 肃正
		if = {
			limit = {
				is_planet_class = pc_ai
			}
			if = {
				limit = {
					NOT = { has_carrier_flag = machine_lair }
				}
				destroy_machine_world = yes
			}
			else = {
				event_target:akx_time_eraser = { save_event_target_as = final_machine_world_destroyer }
				stop_crisis_sound = yes
				carrier_event = { id = crisis.2046 }
			}
		}

			limit = { is_colony = yes }
			if = {
				limit = { exists = orbital_defence }
				destroy_fleet = orbital_defence
			}
		}
		set_dimensional_horrorling_planet = yes
		clear_deposits_and_orbital_station = yes
		clear_planet_modifiers = yes
		prevent_anomaly = yes
		set_carrier_flag = shroud_storm_changed
		set_carrier_flag = cannot_be_hit_by_aow_starfall_spear
		set_name = NAME_Akx_Time_Erasured
		add_modifier = {
			modifier = akx_time_erasure_planet_modifier
			days = -1
		}
	}

		limit = {
			NOT = { is_megastructure_type = pass_black_hole }
		}
		if = {
			limit = {
				is_megastructure_type = grand_archive_0
			}
			solar_system = {
				system_event = { id = grand_archive.4501 }
			}
		}
		remove_megastructure = this
	}
	every_system_ambient_object = {
		limit = {
			NOT = { has_ambient_object_flag = space_storm_object }
		}
		destroy_ambient_object = this
	}
	if = {
		limit = {
			NOT = { exists = event_target:akx_time_eraser_ship }
		}

			if = {

						is_owned_by = event_target:akx_time_eraser
					}

						save_event_target_as = akx_time_eraser_ship
