	create_species = {
		name = NAME_Dimension_Traveler
		class = $CLASS$
		namelist = TIME0
		portrait = $PORTRAIT$
		homeworld = event_target:dimension_traveler_planet
		traits = {
			trait = trait_pc_nuked_preference
			trait = trait_dimension_traveler
			trait = trait_spatial_mastery
		}
		gender = female
		immortal = yes
		can_be_modified = no
		effect = {
			save_global_event_target_as = dimension_traveler_species
			set_species_identity = prev.owner_main_species
		}
	}
}

check_dimension_traveler_level = {
	owner = {
		if = {
			limit = {
				# 等级只增不降
				check_variable = {
					which = dimension_traveler_level
					value <= prev.trigger:has_base_skill
				}
			}
			set_variable = {
				which = dimension_traveler_level
				value = prev.trigger:has_base_skill
			}
			set_variable = {
				which = dimension_traveler_experience
				value = prev.trigger:has_experience
			}
		}
	}
}

set_dimension_traveler_level_and_experience = {
	if = {
		limit = {
			owner = { is_variable_set = dimension_traveler_level }
		}
		set_skill = owner.dimension_traveler_level
	}
	if = {
		limit = {
			owner = { is_variable_set = dimension_traveler_experience }
		}
		add_experience = 10
		set_variable = {
			which = dimension_traveler_experience #假定原经验为100/0
			value = owner.dimension_traveler_experience
		}
		set_variable = {
			which = dimension_traveler_experience_gain #假定实际获得经验20
			value = trigger:has_experience
		}
		subtract_variable = {
			which = dimension_traveler_experience #100/0变为80/-20
			value = dimension_traveler_experience_gain #20
		}
		if = {
			limit = {
				check_variable = {
					which = dimension_traveler_experience
					value > 0
				}
			}
			divide_variable = {
				which = dimension_traveler_experience_gain #20变为2
				value = 10
			}
			divide_variable = {
				which = dimension_traveler_experience #80变为40
				value = dimension_traveler_experience_gain #2
			}
		}
		add_experience = dimension_traveler_experience #40x2+20=100/-20+20=0
		clear_variable = dimension_traveler_experience
		clear_variable = dimension_traveler_experience_gain
	}
}

set_dimension_traveler_location = {
	if = {
		limit = {
			exists = event_target:dimension_traveler_leader
		}
		event_target:dimension_traveler_leader = {
			if = {
				limit = {
					exists = colony
				}
				colony = { save_event_target_as = dimension_traveler_location }
			}
			else_if = {
				limit = {
					exists = planet
				}
				planet = { save_event_target_as = dimension_traveler_location }
			}
			else_if = {
				limit = {
					exists = fleet
				}
				fleet = { save_event_target_as = dimension_traveler_location }
			}
			else_if = {
				limit = {
					is_assigned_to_galcom = yes
				}
				if = {
					limit = {
						has_galactic_emperor = yes
					}
					galactic_emperor = { save_event_target_as = dimension_traveler_location }
				}
				else_if = {
					limit = {
						has_galactic_custodian = yes
					}
					galactic_custodian = { save_event_target_as = dimension_traveler_location }
				}
				else = {
					prev = { save_event_target_as = dimension_traveler_location }
				}
			}
			else_if = {
				limit = {
					is_assigned_to_federation = yes
				}
				prev.federation.federation_leader = { save_event_target_as = dimension_traveler_location }
			}
			else = {
				prev = { save_event_target_as = dimension_traveler_location }
			}
		}
	}
	else = {
		save_event_target_as = dimension_traveler_location
	}
}

set_dimension_traveler_event_portrait = {
	if = {
		limit = {
			exists = event_target:dimension_traveler_leader
			NOT = { has_country_flag = create_new_dimension_traveler_portrait_leader } # 创建新的领袖作为立绘领袖，用于界旅者本体即将不存在的情况
		}
		if = {
			limit = {
				exists = event_target:dimension_traveler_portrait_leader
				event_target:dimension_traveler_portrait_leader = {
					is_dimension_traveler = no
				}
			}
			event_target:dimension_traveler_portrait_leader = {
				kill_leader = { show_notification = no }
			}
		}
		event_target:dimension_traveler_leader = {
			save_global_event_target_as = dimension_traveler_portrait_leader
		}
	}
	else = {
		remove_country_flag = create_new_dimension_traveler_portrait_leader
		if = {
			limit = {
				exists = event_target:dimension_traveler_portrait_leader
				event_target:dimension_traveler_portrait_leader = {
					is_dimension_traveler = no # 适用于上一步跳过了的情况
				}
			}
			event_target:dimension_traveler_portrait_leader = {
				kill_leader = { show_notification = no }
			}
		}
		create_leader = {
			class = scientist
			species = event_target:dimension_traveler_species
			name = NAME_Dimension_Traveler
			gender = female
			immortal = yes
			randomize_traits = no
			event_leader = yes
			skill = 1
			set_age = 18
			background_planet = event_target:dimension_traveler_planet
			custom_catch_phrase = dimension_traveler_catch_phrase
			custom_description = dimension_traveler_backstory
			traits = {
				trait = subclass_scientist_dimension_traveler
			}
			effect = {
				owner = {
					switch = {
						trigger = has_country_flag
						dimension_traveler_official_active = {
							prev = { change_leader_class = official }
						}
						dimension_traveler_scientist_active = {
							prev = { change_leader_class = scientist }
						}
						dimension_traveler_commander_active = {
							prev = { change_leader_class = commander }
						}
						default = {
						}
					}
				}
				save_global_event_target_as = dimension_traveler_portrait_leader
				block_vanilla_leader_traits = yes
				set_leader_flag = immune_to_negative_traits
				set_leader_flag = has_gained_level_trait
				set_leader_flag = leader_death_events_blocked
				set_leader_flag = memory_vault_contributed
				set_dimension_traveler_level_and_experience = yes
				set_dimension_traveler_background = yes
				give_dimension_traveler_trait = yes
				dimension_traveler_wg_effect = yes
				exile_leader_as = dimension_traveler_portrait_leader
			}
		}
	}
	# 不同职业切换肖像
	# 美少女界旅者
	if = {
		limit = {
			active_dimension_traveler_special_portrait = yes
		}
		switch = {
			trigger = has_country_flag
			dimension_traveler_official_active = {
				event_target:dimension_traveler_portrait_leader = {
					change_leader_portrait = dimension_traveler_official_portrait
				}
			}
			dimension_traveler_scientist_active = {
				event_target:dimension_traveler_portrait_leader = {
					change_leader_portrait = dimension_traveler_scientist_portrait
				}
			}
			dimension_traveler_commander_active = {
				event_target:dimension_traveler_portrait_leader = {
					change_leader_portrait = dimension_traveler_commander_portrait
				}
			}
			default = {
				event_target:dimension_traveler_portrait_leader = {
					change_leader_portrait = event_target:dimension_traveler_species
				}
			}
		}
	}
	# 小狐狸界旅者
	else_if = {
		limit = {
			NOR = {
				has_country_flag = dimension_traveler_official_active
				has_country_flag = dimension_traveler_scientist_active
				has_country_flag = dimension_traveler_commander_active
			}
		}
		event_target:dimension_traveler_portrait_leader = {
			set_leader_flag = default_dimension_traveler_clothes
		}
	}
	else = {
		event_target:dimension_traveler_portrait_leader = {
			remove_leader_flag = default_dimension_traveler_clothes
		}
	}
}
