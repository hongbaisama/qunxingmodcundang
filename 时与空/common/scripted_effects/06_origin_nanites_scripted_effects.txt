# 国家
refresh_faction_hostility_with_graygoo_country = {
	if = {
		limit = {
			exists = event_target:graygoo_country
		}
		# 以防万一，恢复友好
		event_target:graygoo_country = {
			set_faction_hostility = {
				target = prev
				set_hostile = no
				set_neutral = no
				set_friendly = yes
			}
		}
	}
}
