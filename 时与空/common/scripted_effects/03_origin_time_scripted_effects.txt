finish_pc_endless_time = {
	# 星球设置
	while = {
		limit = {
			num_districts = {
				type = district_time
				value > 1
			}
		}
		remove_district = district_time
	}
	while = {
		limit = {
			num_districts = {
				type = district_time_physics
				value > 1
			}
		}
		remove_district = district_time_physics
	}
	while = {
		limit = {
			num_districts = {
				type = district_time_society
				value > 1
			}
		}
		remove_district = district_time_society
	}
	while = {
		limit = {
			num_districts = {
				type = district_time_engineering
				value > 1
			}
		}
		remove_district = district_time_engineering
	}
	remove_deposit = d_time
	add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_endless_time }
	change_pc = pc_endless_time
	remove_modifier = timesingularity
	set_planet_size = 0
	validate_planet_buildings_and_districts = yes
	save_global_event_target_as = time_dimensional_horror_planet
	if = {
		limit = {
			is_capital = yes
		}
		set_colony_type = col_endless_time_capital
		set_colony_type = none # 更新星球规划的同时改用自动规划
	}
	every_owned_pop_group = {
		limit = {
			is_origin_time_species = yes
		}
		if = {
			limit = {
				is_robot_pop_group = no
			}
			modify_species = {
				species = this
				ideal_planet_class = pc_endless_time
			}
		}
		else = {
			modify_species = {
				species = this
				remove_trait = trait_dry_planet_preference
				remove_trait = trait_wet_planet_preference
				remove_trait = trait_cold_planet_preference
				remove_trait = trait_pc_time_preference
				add_trait = trait_pc_endless_time_preference
			}
		}
	}
	carrier_event = { id = time.50 days = 1080 random = 1440 }
	# 国家设置
	owner = {
		save_global_event_target_as = time_dimensional_horror_owner
		set_country_flag = situation_origin_time_finished
		create_timeline_pc_endless_time = yes
		if = {
			limit = {
				has_origin_time_ruler = yes
			}
			switch = {
				trigger = has_country_flag
				time_soul_virtuality_finished = { set_origin = origin_endless_time_virtuality }
				time_soul_synthetic_finished = { set_origin = origin_endless_time_synthetic }
				time_soul_cybernetic_finished = { set_origin = origin_endless_time_cybernetic }
				time_soul_genetic_finished = { set_origin = origin_endless_time_genetic }
				time_soul_psionic_finished = { set_origin = origin_endless_time_psionic }
			}
			if = {
				limit = {
					is_normal_origin_time_authority = yes
				}
				set_country_flag = unlock_auth_endless_time
				advanced_authority_refresh = yes
			}
		}
	}
	# 时之螶设置
	create_country = {
		name = "NAME_The_Loop"
		type = faction
		auto_delete = no
		use_hostilities_from = event_target:time_dimensional_horror_owner
		flag = {
			icon= {
				category = "special"
				file = "extradimensional_02.dds"
			}
			background= {
				category = "backgrounds"
				file = "circle.dds"
			}
			colors={
				"black"
				"red"
				"null"
				"null"
			}
		}
		effect = {
			add_modifier = {
				modifier = time_dimensional_horror_modifier
				days = -1
			}
			establish_communications_no_message = event_target:time_dimensional_horror_owner
			establish_contact = { who = event_target:time_dimensional_horror_owner location = root }
			set_faction_hostility = {
				target = event_target:time_dimensional_horror_owner
				set_hostile = no
				set_neutral = no
				set_friendly = yes
			}
			save_global_event_target_as = time_dimensional_horror_country
			create_time_dimensional_horror_end_fleet = yes
		}
	}
	# 星系设置
	solar_system = {
		if = {
			limit = {
				has_star_flag = time_core_system
			}
			set_star_class = sc_endless_time_core
		}
		else = {
			set_star_flag = lcluster
			set_star_class = sc_endless_time
		}
		time_remove_neighbor_hyperlane = yes
	}
}

				AND = {
					has_situation_flag = materialist_is_winner

				}
			}

		}

			has_situation_flag = situation_time_death

				has_situation_flag = xenophile_is_winner
				owner = { is_megacorp = yes }
			}
		}
		set_situation_flag = auth_time_corporate

				has_situation_flag = spiritualist_is_winner

				has_situation_flag = xenophobe_is_winner
			}
		}
		set_situation_flag = auth_imperial
	}

		limit = {
			has_situation_flag = egalitarian_is_winner
			OR = {
				has_situation_flag = materialist_is_winner
				has_situation_flag = pacifist_is_winner

		limit = {
			owner = { is_megacorp = yes }
		}
		set_situation_flag = auth_corporate
	}
	else_if = {
		limit = {
			has_situation_flag = authoritarian_is_winner
		}
		set_situation_flag = auth_dictatorial
	}
	else_if = {
		limit = {

			remove_invalid_civics = yes
		}
		hidden_effect = {
			# 预防普政巨企间切换失败（还原理念）
			if = {
				limit = {
					prev = {
						NOR = {
							has_situation_flag = auth_time_corporate
							has_situation_flag = auth_corporate
						}
					}
				}
				change_time_government_give_civic = { CIVIC = civic_time_machine }
				change_time_government_give_civic = { CIVIC = civic_circle_nexus }
				change_time_government_give_civic = { CIVIC = civic_dimension_engine }
				change_time_government_give_civic = { CIVIC = civic_dark_consortium }
				change_time_government_give_civic = { CIVIC = civic_eager_explorers }
				change_time_government_give_civic = { CIVIC = civic_sovereign_guardianship }
				change_time_government_give_civic = { CIVIC = civic_storm_callers }
				change_time_government_give_civic = { CIVIC = civic_galactic_curators }
			}
			else = {
				change_time_government_give_civic = { CIVIC = civic_time_machine_corporate }
				change_time_government_give_civic = { CIVIC = civic_circle_nexus_corporate }
				change_time_government_give_civic = { CIVIC = civic_dimension_engine_corporate }
				change_time_government_give_civic = { CIVIC = civic_corporate_dark_consortium }
				change_time_government_give_civic = { CIVIC = civic_privatized_exploration }
				change_time_government_give_civic = { CIVIC = civic_corporate_sovereign_guardianship }
				change_time_government_give_civic = { CIVIC = civic_storm_callers_megacorp }
				change_time_government_give_civic = { CIVIC = civic_corporate_galactic_curators }
			}
		}
		if = {
			limit = {
				NOT = { has_civic = $CIVIC_1$ }
			}
			force_add_civic = $CIVIC_1$
		}
		if = {
			limit = {
				NOT = { has_civic = $CIVIC_2$ }
			}
			force_add_civic = $CIVIC_2$
		}
	}
}

change_time_government_check_civic = {
	if = {
		limit = {
			has_civic = $CIVIC$
		}
		set_country_flag = time_government_switch_to_$SWITCH$
		force_remove_civic = $CIVIC$
	}
}

# 国家
change_time_government_give_civic = {
	if = {
		limit = {
			has_country_flag = time_government_switch_to_$CIVIC$
		}
		remove_country_flag = time_government_switch_to_$CIVIC$
		force_add_civic = $CIVIC$
	}
}
