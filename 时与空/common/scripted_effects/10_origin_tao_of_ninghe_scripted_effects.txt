# 国家
refresh_tao_of_ninghe_country_value = {
	set_variable = {
		which = num_yin_leader
		value = 0
	}
	set_variable = {
		which = num_yang_leader
		value = 0
	}
	set_variable = {
		which = num_yin_planet
		value = value:num_yin_planet
	}
	set_variable = {
		which = num_yang_planet
		value = value:num_yang_planet
	}
	every_owned_leader = {
		limit = {
			is_yin_leader = yes
		}
		owner = {
			change_variable = {
				which = num_yin_leader
				value = prev.value:num_yin_leader
			}
		}
	}
	every_owned_leader = {
		limit = {
			is_yang_leader = yes
		}
		owner = {
			change_variable = {
				which = num_yang_leader
				value = prev.value:num_yang_leader
			}
		}
	}
}

# 国家
tao_of_ninghe_ai_country_bonus = {
	# 防止反复触发
	set_timed_country_flag = {
		flag = has_tao_of_ninghe_ai_country_bonus
		days = 1
	}
	if = {
		limit = {
			is_yin_country = yes
		}
		every_owned_leader = {
			limit = {
				is_yang_leader = yes
			}
			add_experience = 500
		}
	}
	else_if = {
		limit = {
			is_yang_country = yes
		}
		every_owned_leader = {
			limit = {
				is_yin_leader = yes
			}
			add_experience = 500
		}
	}
}
