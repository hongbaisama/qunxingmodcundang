origin_reckoning_arrived_basic_starting_effect = {
	save_global_event_target_as = origin_reckoning_arrived_country
	
	give_physics_tech_tier_1_effect_no_message = yes
	give_physics_tech_tier_2_effect_no_message = yes
	give_society_tech_tier_1_effect_no_message = yes
	give_society_tech_tier_2_effect_no_message = yes
	give_engineering_tech_tier_1_effect_no_message = yes
	give_engineering_tech_tier_2_effect_no_message = yes
	give_technology = { tech = "tech_psionic_theory" message = no }
	give_technology = { tech = "tech_telepathy" message = no }
	
	add_tradition = tr_adaptability_adopt
	add_tradition = tr_adaptability_recycling
	add_tradition = tr_adaptability_survival_fittest
	add_tradition = tr_adaptability_environmental_diversification
	add_tradition = tr_adaptability_adaptive_ecology
	add_tradition = tr_adaptability_appropriation
	add_ascension_perk = ap_one_vision
	
	add_tradition = tr_harmony_adopt
	add_tradition = tr_harmony_mind_and_body
	add_tradition = tr_harmony_kinship
	add_tradition = tr_harmony_the_greater_good
	add_tradition = tr_harmony_harmonious_directives
	add_tradition = tr_harmony_utopian_dream
	add_ascension_perk = ap_mind_over_matter
	
	add_tradition = tr_psionics_shroud_adopt
	add_tradition = tr_psionics_shroud_psi_corps
	add_tradition = tr_psionics_shroud_great_awakening
	add_tradition = tr_psionics_shroud_telekinesis
	add_tradition = tr_psionics_shroud_clairvoyance
	add_tradition = tr_psionics_shroud_psychometry
	add_ascension_perk = ap_voidborn
	
	owner_main_species = {
		set_species_homeworld = event_target:origin_reckoning_arrived_creatures_planet
	}
	capital_scope = {
		solar_system = { save_event_target_as = homeworld_system_name }
	}
	event_target:origin_reckoning_arrived_creatures_planet = {
		if = {
			limit = {
				NOT = { has_planet_flag = planet_earth }
			}
			set_name = NAME_Origin_Reckoning_Arrived_Creatures_Planet
		}
		solar_system = {
			root = { set_visited = prev }
			if = {
				limit = {
					NOT = { has_star_flag = sol_system }
				}
				set_name = event_target:homeworld_system_name
				star = { set_name = event_target:homeworld_system_name }
				every_system_planet = {
					limit = {
						is_primary_star = no
						NOT = { is_same_value = event_target:origin_reckoning_arrived_creatures_planet }
					}
					set_name = NAME_Origin_Reckoning_Arrived_Creatures_Other_Planet
				}
			}
		}
	}
	capital_scope = {
		solar_system = {
			set_name = NAME_Origin_Reckoning_Arrived_Starting_System
			star = { set_name = NAME_Origin_Reckoning_Arrived_Starting_Star_A }
			random_system_planet = {
				limit = {
					is_star = yes
					is_primary_star = no
				}
				set_name = NAME_Origin_Reckoning_Arrived_Starting_Star_B
			}
		}
		set_name = NAME_Origin_Reckoning_Arrived_Starting_Ship
		fleet = { set_name = NAME_Origin_Reckoning_Arrived_Starting_Fleet }
		ship = {
			starbase = {
				set_starbase_building = {
					slot = 3
					building = arkship_listening_post
				}
				set_starbase_building = {
					slot = 4
					building = arkship_afterburner
				}
			}
		}
		add_building = building_psi_corps
		# 初始人口
		every_owned_pop_group = {
			limit = {
				is_same_species = root.owner_main_species
			}
			kill_pop_group = {
				pop_group = this
				percentage = 0.5
			}
		}
		# 初始人口取整
		while = {
			limit = {
				check_variable_arithmetic = {
					which = value:num_modulo_of_time_pop_amount_value|MODULO|10|
					value > 0
				}
			}
			if = {
				limit = {
					check_variable_arithmetic = {
						which = value:num_modulo_of_time_pop_amount_value|MODULO|10|
						value >= 8
					}
				}
				create_pop_group = {
					species = owner_main_species
					size = 1
				}
			}
			else = {
				weighted_random_owned_pop_group = {
					kill_pop_group = {
						pop_group = this
						amount = 1
					}
				}
			}
		}
	}
}

			}
		}
