refresh_dimensional_horror_crisis_storm_original_level = {
	event_target:global_event_country = {
		set_variable = {
			which = dimensional_horror_crisis_storm_original_level
			value = value:dimensional_horror_crisis_storm_original_level
		}
	}
	every_country = {
		limit = {
			has_event_chain = dimensional_horror_crisis_chain
		}
		reset_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = dimensional_horror_crisis_storm_original_level
		}
		add_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = dimensional_horror_crisis_storm_original_level
			amount = event_target:global_event_country.dimensional_horror_crisis_storm_original_level
		}
	}
}

create_dimensional_horror_crisis_chain = {
	# 创建事件链
	begin_event_chain = {
		event_chain = dimensional_horror_crisis_chain
		target = this
	}
	# 更新初始计数
	if = {
		limit = {
			event_target:global_event_country = {
				is_variable_set = dimensional_horror_crisis_storm_original_level
			}
		}
		add_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = dimensional_horror_crisis_storm_original_level
			amount = event_target:global_event_country.dimensional_horror_crisis_storm_original_level
		}
	}
	if = {
		limit = {
			event_target:global_event_country = {
				is_variable_set = black_hole_horror_count
			}
		}
		add_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = black_hole_horror_count
			amount = event_target:global_event_country.black_hole_horror_count
		}
	}
	if = {
		limit = {
			event_target:global_event_country = {
				is_variable_set = dimensional_horror_crisis_affected_system
			}
		}
		add_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = dimensional_horror_crisis_affected_system
			amount = event_target:global_event_country.dimensional_horror_crisis_affected_system
		}
	}
	if = {
		limit = {
			event_target:global_event_country = {
				is_variable_set = dimensional_horror_crisis_erasured_system
			}
		}
		add_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = dimensional_horror_crisis_erasured_system
			amount = event_target:global_event_country.dimensional_horror_crisis_erasured_system
		}
	}
	if = {
		limit = {
			event_target:global_event_country = {
				is_variable_set = dimensional_horror_crisis_sealed_system
			}
		}
		add_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = dimensional_horror_crisis_sealed_system
			amount = event_target:global_event_country.dimensional_horror_crisis_sealed_system
		}
	}
	if = {
		limit = {
			event_target:global_event_country = {
				is_variable_set = dimensional_horror_crisis_kills
			}
		}
		add_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = dimensional_horror_crisis_kills_others
			amount = event_target:global_event_country.dimensional_horror_crisis_kills
		}
	}
	if = {
		limit = {
			event_target:global_event_country = {
				is_variable_set = dimensional_horror_crisis_victims
			}
		}
		add_event_chain_counter = {
			event_chain = dimensional_horror_crisis_chain
			counter = dimensional_horror_crisis_victims
			amount = event_target:global_event_country.dimensional_horror_crisis_victims
		}
	}
	# 开启使命
	enable_dimensional_horror_crisis_mission = { MISSION = MISSION_DIMENSIONAL_HORROR_CRISIS_SYSTEM_1 }
	enable_dimensional_horror_crisis_mission = { MISSION = MISSION_DIMENSIONAL_HORROR_CRISIS_SHIP_1 }
	enable_dimensional_horror_crisis_mission = { MISSION = MISSION_DIMENSIONAL_HORROR_CRISIS_TECHNOLOGY_1 }
	# 与天灾国家建立通讯
	event_target:dimensional_horror_crisis_country = {
		establish_communications_no_message = prev
	}
}

dimensional_horror_crisis_storm_add = {
	set_star_flag = storm_system
	set_global_flag = cosmic_storm_has_occurred
	if = {
		limit = {
			NOT = { has_global_flag = dimensional_horror_crisis_storm_has_occurred }
		}
		dimensional_horror_crisis_setup = yes
	}
	if = {
		limit = {
			exists = owner
			owner = { has_event_chain = dimensional_horror_crisis_chain }
		}
		owner = {
			add_dimensional_horror_crisis_mission_counter = {
				MISSION = SYSTEM
				COUNTER = system
			}
		}
	}
	every_playable_country = {
		limit = {
			# 游牧的方舟/驿站可以积累计数，并且一个星系同时只计一次数
			is_nomadic = yes
			prev = {
				NOT = { is_owned_by = prev }
				any_fleet_in_system = {
					is_controlled_by = prevprev
					OR = {
						is_arkship_fleet = yes
						is_waystation_ship = yes
					}
				}
			}
		}
		add_dimensional_horror_crisis_mission_counter = {
			MISSION = SYSTEM
			COUNTER = system
		}
	}
	if = {
		limit = {
			is_dimensional_horror_crisis_inactive_system = no
		}
		star = {
			create_ambient_object = {
				type = dimensional_horror_crisis_storm
				location = this
				use_3d_location = yes
			}
			last_created_ambient_object = {
				set_ambient_object_flag = space_storm_object
				set_ambient_object_flag = dimensional_horror_crisis_storm_object
			}
		}
	}
	every_system_planet = {
		limit = {
			OR = {
				has_modifier = dimensional_horror_crisis_storm_aftermath_modifier_severity_1
				has_modifier = dimensional_horror_crisis_storm_aftermath_modifier_severity_2
				has_modifier = dimensional_horror_crisis_storm_aftermath_modifier_severity_3
			}
		}
		remove_modifier = dimensional_horror_crisis_storm_aftermath_modifier_severity_1
		remove_modifier = dimensional_horror_crisis_storm_aftermath_modifier_severity_2
		remove_modifier = dimensional_horror_crisis_storm_aftermath_modifier_severity_3
	}
}

create_dimensional_horror_crisis_in_system = {
	if = {
		limit = {
			is_dimensional_horror_crisis_inactive_system = no
		}
		# 数据计算
		if = {
			limit = {
				is_dimensional_horror_crisis_affected_system = no
			}
			set_star_flag = dimensional_horror_crisis_affected
			event_target:global_event_country = {
				change_variable = {
					which = dimensional_horror_crisis_affected_system
					value = 1
				}
			}
			add_dimensional_horror_crisis_chain_counter = { COUNTER = dimensional_horror_crisis_affected_system }
		}
		# 终焉之魇
		if = {
			limit = {
				exists = star
				star = {
					NOR = {
						is_origin_time_planet_class = yes
						is_time_white_hole_planet_class = yes
					}
				}
				NOT = {
					any_fleet_in_system = {
						is_special_dimensional_horror = yes
					}
				}
			}
			star = {
				save_event_target_as = dimensional_horror_crisis_star
			}
			# 星系改造
			if = {
				limit = {
					NOR = {
						has_star_flag = dimensional_horror_crisis_changed
						star = {
							OR = {
								has_carrier_flag = dimensional_horror_crisis_changed
								can_not_be_changed_by_akx_worm = yes # 设定上与时虫无关，单纯把这条script_trigger拿来用了
								AND = {
									is_same_value = event_target:akx_star
									is_time_black_hole_planet_class = yes
								}
							}
						}
					}
				}
				star = {
					time_destroy_colony = yes
					set_planet_size = 30
					change_pc = pc_black_hole
					reset_planet = yes
					clear_deposits_and_orbital_station = yes
					clear_planet_modifiers = yes
					prevent_anomaly = yes
					set_name = "NAME_Akx_Time_Erasured"
					set_carrier_flag = dimensional_horror_crisis_changed
				}
				if = {
					limit = {
						is_time_black_hole_or_white_hole_star_class = no
					}
					set_star_class = sc_black_hole
					# 清除戴森云、戴森球、奇迹球并返还建造次数
					worm_clear_dyson_sphere = yes
				}
				# 避免因为修改星系类型导致星球图片还原
				star = {
					set_carrier_flag = exist_dimensional_horrorling_object
					set_planet_entity = {
						picture = pc_horror_black_hole
					}
				}
				set_star_flag = dimensional_horror_crisis_changed
			}
			# 生成舰船
			if = {
				limit = {
					exists = event_target:dimensional_horror_crisis_star
					event_target:dimensional_horror_crisis_star = { is_time_black_hole_planet_class = yes }
				}
				create_fleet = {
					name = NAME_Dimensional_Horror_Crisis_$TYPE$
					settings = {
						can_upgrade = no
						can_change_composition = no
						uses_naval_capacity = no
						can_disband = no
						spawn_debris = no
						is_boss = yes
					}
					effect = {
						set_owner = event_target:dimensional_horror_crisis_country
						create_ship = {
							name = NAME_Dimensional_Horror_Crisis_$TYPE$
							design = NAME_Dimensional_Horror_Crisis_Design_$TYPE$
							prefix = no
							upgradable = no
						}
						set_location = {
							target = event_target:dimensional_horror_crisis_star
							distance = 0
							direction = in_system
						}
						if = {
							limit = {
								is_time_erasure_system = no
								can_dimensional_horror_crisis_erasured_system = yes
							}
							fleet_event = { id = time_dimensional_horror_crisis.200 days = 150 random = 60 }
						}
					}
				}
			}
		}
	}
}

		which = dimensional_horror_crisis_storm_pop_killed_percentage

	multiply_variable = {
		which = dimensional_horror_crisis_storm_pop_killed_percentage

	}
	owner = {
		create_message = {
			type = DIMENSIONAL_HORROR_CRISIS_STORM_POP_KILLED

			variable = {
				type = name

				scope = prev
			}
			variable = {
				type = variable
				varname = dimensional_horror_crisis_storm_pop_killed_percentage
