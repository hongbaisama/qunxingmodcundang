# 国家
refresh_dimension_traveler_skill_point = {
	if = {
		limit = {
			NOT = { is_variable_set = used_dimension_traveler_skill_point }
		}
		set_variable = {
			which = used_dimension_traveler_skill_point
			value = 0
		}
	}
	if = {
		limit = {
			NOT = { is_variable_set = used_dimension_traveler_skill_point_official }
		}
		set_variable = {
			which = used_dimension_traveler_skill_point_official
			value = 0
		}
	}
	if = {
		limit = {
			NOT = { is_variable_set = used_dimension_traveler_skill_point_scientist }
		}
		set_variable = {
			which = used_dimension_traveler_skill_point_scientist
			value = 0
		}
	}
	if = {
		limit = {
			NOT = { is_variable_set = used_dimension_traveler_skill_point_commander }
		}
		set_variable = {
			which = used_dimension_traveler_skill_point_commander
			value = 0
		}
	}
	set_variable = {
		which = all_dimension_traveler_skill_point
		value = value:all_dimension_traveler_skill_point
	}
	set_variable = {
		which = available_dimension_traveler_skill_point
		value = value:available_dimension_traveler_skill_point
	}
	set_variable = {
		which = used_dimension_traveler_skill_point_official_councilor
		value = value:used_dimension_traveler_skill_point_class_councilor|CLASS|official|
	}
	set_variable = {
		which = used_dimension_traveler_skill_point_scientist_councilor
		value = value:used_dimension_traveler_skill_point_class_councilor|CLASS|scientist|
	}
	set_variable = {
		which = used_dimension_traveler_skill_point_commander_councilor
		value = value:used_dimension_traveler_skill_point_class_councilor|CLASS|commander|
	}
	set_variable = {
		which = used_dimension_traveler_skill_point_official_normal
		value = value:used_dimension_traveler_skill_point_class_normal|CLASS|official|
	}
	set_variable = {
		which = used_dimension_traveler_skill_point_scientist_normal
		value = value:used_dimension_traveler_skill_point_class_normal|CLASS|scientist|
	}
	set_variable = {
		which = used_dimension_traveler_skill_point_commander_normal
		value = value:used_dimension_traveler_skill_point_class_normal|CLASS|commander|
	}
}

refresh_dimension_traveler_skill_flag = {
	clear_dimension_traveler_skill_flag = yes
	switch = {
		trigger = has_country_flag
		dimension_traveler_official_active = {
			set_country_flag = show_dimension_traveler_official_skill
		}
		dimension_traveler_scientist_active = {
			set_country_flag = show_dimension_traveler_scientist_skill
		}
		dimension_traveler_commander_active = {
			set_country_flag = show_dimension_traveler_commander_skill
		}
	}
}

# 任意（未激活）（第1层函数）
# TIER为层数（1~9），NUM为每层序号（1~9）
choose_dimension_traveler_skill = {
	custom_tooltip = dimension_traveler_skill_$CLASS$_$TIER$_$NUM$_effect
	custom_tooltip = cost_dimension_traveler_skill_point_$COUNT$
	# 为了美观，不给玩家显示跳到国家域的过程
	tooltip = {
		add_resource = {
			sr_dark_matter = @dimension_traveler_skill_sr_dark_matter_cost_$COUNT$
			astral_threads = @dimension_traveler_skill_astral_threads_cost_$COUNT$
		}
	}
	hidden_effect = {
		event_target:dimension_traveler_owner = {
			change_variable = {
				which = used_dimension_traveler_skill_point
				value = $COUNT$
			}
			change_variable = {
				which = used_dimension_traveler_skill_point_$CLASS$
				value = $COUNT$
			}
			add_resource = {
				sr_dark_matter = @dimension_traveler_skill_sr_dark_matter_cost_$COUNT$
				astral_threads = @dimension_traveler_skill_astral_threads_cost_$COUNT$
			}
			set_country_flag = dimension_traveler_skill_$CLASS$_$TIER$_$NUM$
			refresh_dimension_traveler_skill_point = yes
			# 解锁新技能
			set_variable = {
				which = dimension_traveler_skill_tier
				value = $TIER$
			}
			set_variable = {
				which = dimension_traveler_skill_num
				value = $NUM$
			}
			if = {
				limit = {
					check_variable = {
						which = dimension_traveler_skill_tier
						value < 5
					}
				}
				change_variable = {
					which = dimension_traveler_skill_tier
					value = 1
				}
				change_variable = {
					which = dimension_traveler_skill_num
					value = 1
				}
				choose_dimension_traveler_skill_for_tier = { CLASS = $CLASS$ }
				subtract_variable = {
					which = dimension_traveler_skill_num
					value = 2
				}
				choose_dimension_traveler_skill_for_tier = { CLASS = $CLASS$ }
			}
			else_if = {
				limit = {
					check_variable = {
						which = dimension_traveler_skill_tier
						value > 5
					}
				}
				subtract_variable = {
					which = dimension_traveler_skill_tier
					value = 1
				}
				change_variable = {
					which = dimension_traveler_skill_num
					value = 1
				}
				choose_dimension_traveler_skill_for_tier = { CLASS = $CLASS$ }
				subtract_variable = {
					which = dimension_traveler_skill_num
					value = 2
				}
				choose_dimension_traveler_skill_for_tier = { CLASS = $CLASS$ }
			}
			clear_variable = dimension_traveler_skill_tier
			clear_variable = dimension_traveler_skill_num
		}
	}
}

	if = {
		limit = {
			root.from = { army_type = $TYPE$ }
		}
		if = {
			limit = {
				root.from = { exists = species }
			}
			root.fromfromfrom = {
				create_army = {
					owner = root
					type = $TYPE$
					species = root.from.species
				}
			}
		}
		else = {
			root.fromfromfrom = {
				create_army = {
					owner = root
					type = $TYPE$
				}
			}
		}
	}
}
