create_timeline_time_origin = {
	add_timeline_event = {
		type = timeline_origin
		override_tooltip = $ORIGIN$_desc
		override_text = { text:$ORIGIN$ }
		override_texture = { background:GFX_$ORIGIN$ }
		override_id = timeline_$ORIGIN$
	}
}

# 国家
create_timeline_wonder_country_time = {
	set_country_flag = created_timeline_wonder_country_time
	save_event_target_as = this_country
	ruler = { save_event_target_as = this_leader }
	if = {
		limit = {
			is_wonder_country_time_0 = yes
		}
		add_timeline_event = {
			type = timeline_origin_specific
			date = 2100.1.1
			override_id = timeline_wonder_country_time_0
			override_tooltip = "TIMELINE_EVENT_WONDER_COUNTRY_TIME_0_TOOLTIP"
			override_types = { country leader }
			override_text = { "text:TIMELINE_WONDER_COUNTRY_TIME" }
			override_texture = { "background:GFX_evt_chosen_leader" }
			targets = { event_target:this_country event_target:this_leader }
		}
	}
	else_if = {
		limit = {
			is_wonder_country_time_1 = yes
		}
		add_timeline_event = {
			type = timeline_origin_specific
			date = 2100.1.1
			override_id = timeline_wonder_country_time_1
			override_tooltip = "TIMELINE_EVENT_WONDER_COUNTRY_TIME_1_TOOLTIP"
			override_types = { country leader }
			override_text = { "text:TIMELINE_WONDER_COUNTRY_TIME" }
			override_texture = { "background:GFX_evt_chosen_leader" }
			targets = { event_target:this_country event_target:this_leader }
		}
	}
	else_if = {
		limit = {
			is_wonder_country_time_2 = yes
		}
		add_timeline_event = {
			type = timeline_origin_specific
			date = 2100.1.1
			override_id = timeline_wonder_country_time_2
			override_tooltip = "TIMELINE_EVENT_WONDER_COUNTRY_TIME_2_TOOLTIP"
			override_types = { country leader }
			override_text = { "text:TIMELINE_WONDER_COUNTRY_TIME" }
			override_texture = { "background:GFX_evt_chosen_leader" }
			targets = { event_target:this_country event_target:this_leader }
		}
	}
	add_timeline_event = {
		type = timeline_event_year
		date = 2100.1.1
	}
}
