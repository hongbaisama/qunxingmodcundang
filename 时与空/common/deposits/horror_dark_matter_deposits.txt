@ranger_large_production = 1
@evopred_large_blocker_progress = 40
@evopred_small_blocker_progress = 20

d_horror_dark_matter_blocker = {
	time = 720

	is_for_colonizable = yes
	category = deposit_cat_rare_blocker

	icon = d_liquid_dark_matter

	use_weights_for_blocker_swap_types = no
	
	can_be_cleared = {
		owner = { has_technology = tech_mine_dark_matter }
	}

	resources = {
		category = deposit_blockers
		cost = {
			sr_dark_matter = 100
		}
		produces = {
			sr_dark_matter = 2
		}
	}

	potential = {
		always = no
	}

	planet_modifier = {
		planet_max_districts_mult = -0.2
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = no }
		}
		pop_happiness = -0.1
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { is_gestalt = yes }
		}
		planet_amenities_mult = -0.1
	}
	
	on_cleared = {
		custom_tooltip = d_horror_dark_matter_blocker_tooltip
		hidden_effect = {
			d_horror_dark_matter_blocker_cleared_effect = yes
		}
	}

	drop_weight = {
		weight = 0
	}

	ai_weight = {
		weight = 1
	}
}
