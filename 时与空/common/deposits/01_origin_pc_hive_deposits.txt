@blocker_high = 10
@blocker_low = 5
@ranger_small_production = 0.5
@ranger_large_production = 1
@evopred_large_blocker_progress = 40
@evopred_small_blocker_progress = 20

d_pc_hive_blocker = {
	time = 720

	is_for_colonizable = yes
	category = deposit_cat_blockers_natural

	icon = d_infestation

	use_weights_for_blocker_swap_types = no
	
	can_be_cleared = {
		is_planet_class = pc_hive
		hidden_trigger = { exists = owner }
		owner = { has_origin = origin_pc_hive }
	}

	blocker_swap_types = {
		d_pc_hive
	}

	resources = {
		category = deposit_blockers_natural
		cost = {
			food = 1500
		}
	}

	potential = {
		always = no
	}

	planet_modifier = {
		planet_max_districts_mult = -0.5
		species_empire_size_mult = 0.2
		pop_amenities_usage_no_happiness_mult = 0.2
		planet_pops_upkeep_mult = 0.2
		bonus_pop_growth_mult = -0.2
		planet_jobs_produces_mult = -0.2
		planet_structures_upkeep_mult = 0.2
	}
	
	triggered_planet_modifier = {
		potential = {
			biologist_is_ranger_trigger = yes
		}
		planet_biologists_minerals_produces_add = @ranger_large_production
	}

	inline_script = deposits/environmental_architects_natural_blocker_modifier
	
	on_cleared = {
		hidden_effect = {
			owner = { set_country_flag = unlock_decision_origin_pc_hive }
			# 有机障碍
			if = {
				limit = {
					exists = owner
					owner = { has_civic = civic_hive_devouring_swarm }
				}
				owner = { add_resource = { food = 200 } }
			}
			give_bio_repurposing_reward_on_organic_deposit_cleared = yes
		}
	}

	drop_weight = {
		weight = 0
	}

	inline_script = deposits/ai_weight_natural_blocker
}

	is_for_colonizable = yes
	category = deposit_cat_rare
	
	icon = d_huge_tree
	
	custom_tooltip = d_pc_hive_effect
	custom_tooltip_short = d_pc_hive_effect_short
	
	resources = {
		category = planet_deposits
		upkeep = {
			trigger = {
				is_planet_class = pc_hive
				exists = owner
				owner = { has_origin = origin_pc_hive }
			}
			food = 10
			mult = value:d_pc_hive_value
		}
	}
	
	planet_modifier = {
		planet_max_districts_mult = -0.25

		}
		species_empire_size_mult = -0.025
