d_gas_blocker_2 = {
	time = 1080

	is_for_colonizable = yes
	category = deposit_cat_blockers_natural

	icon = d_organic_landfill

	resources = {
		category = deposit_blockers_natural
		cost = {
			energy = 3000
			volatile_motes = 30
			rare_crystals = 30
			exotic_gases = 30
		}
	}

	potential = {
		always = no
	}

	planet_modifier = {
		planet_max_districts_add = -6
	}
	
	triggered_planet_modifier = {
		potential = {
			biologist_is_ranger_trigger = yes
		}
		planet_biologists_minerals_produces_add = @ranger_large_production
	}

	inline_script = deposits/environmental_architects_natural_blocker_modifier

	on_cleared = {
		if = {
			limit = {
				exists = owner_or_space_owner
			}
			owner_or_space_owner = {
				if = {
					limit = {
						country_uses_food = yes
						country_uses_consumer_goods = yes
					}
					add_resource = {
						energy = 1000
						minerals = 1000
						food = 1000
						alloys = 500
						consumer_goods = 500
					}
				}
				else_if = {
					limit = {
						country_uses_food = yes
					}
					add_resource = {
						energy = 1500
						minerals = 1000
						food = 1500
						alloys = 500
					}
				}
				else_if = {
					limit = {
						country_uses_consumer_goods = yes
					}
					add_resource = {
						energy = 1500
						minerals = 1500
						alloys = 500
						consumer_goods = 500
					}
				}
				else = {
					add_resource = {
						energy = 2000
						minerals = 2000
						alloys = 500
					}
				}
			}
		}
		else = {
			tooltip = {
				add_resource = {
					energy = 1000
					minerals = 1000
					food = 1000
					alloys = 500
					consumer_goods = 500
				}
			}
		}
	}

	drop_weight = {
		weight = 0
	}

	inline_script = deposits/ai_weight_natural_blocker
	
	prerequisites = { "tech_gas_blocker_2" }
}
