	is_for_colonizable = yes
	
	category = deposit_cat_rare
	
	custom_tooltip = d_time_horror_effect
	
	resources = {
		category = planet_deposits
	}
	
	triggered_planet_modifier = {
		potential = {
			is_capital = yes
		}
		country_unity_produces_mult = 0.5
		country_physics_research_produces_mult = 0.5
		country_society_research_produces_mult = 0.5
		country_engineering_research_produces_mult = 0.5
		pop_amenities_usage_mult = 0.5
		mult = value:d_time_horror_value
	}
	
	triggered_planet_modifier = {
		potential = {
			is_capital = no
		}
		planet_jobs_unity_produces_mult = 0.5
		planet_jobs_physics_research_produces_mult = 0.5
		planet_jobs_society_research_produces_mult = 0.5
		planet_jobs_engineering_research_produces_mult = 0.5
	}
	
	potential = {
		always = no
	}
	
	drop_weight = {
		weight = 0
	}
}

			}
			volatile_motes = 2.5
			rare_crystals = 2.5
			exotic_gases = 2.5
		}
		upkeep = {
			trigger = {
				OR = {
					has_carrier_flag = end_messenger_3
					has_carrier_flag = future_messenger_3
				}
			}
			volatile_motes = 5
			rare_crystals = 5
			exotic_gases = 5

			has_carrier_flag = end_messenger_1
			exists = owner
			owner = { is_gestalt = no }
		}
		job_end_messenger_per_pop = @deposit_end_messenger_job
		mult = value:num_akx_messenger_job_per_pop_value|TYPE|end|

	# 未来信使
	triggered_planet_modifier = {
		potential = {
			has_carrier_flag = future_messenger_1

		}
		job_future_messenger_per_pop = @deposit_future_messenger_job

	}

	}

	}
