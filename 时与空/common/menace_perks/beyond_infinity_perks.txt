menp_beyond_infinity_crisis = {
	portrait = "GFX_menp_beyond_infinity_crisis"

	on_unlock = {
		if = {
			limit = {
				OR = {
					has_global_flag = dimensional_horror_crisis_storm_has_occurred
					has_beyond_infinity_level = { LEVEL = 4 }
				}
			}
			custom_tooltip = menp_beyond_infinity_crisis_effect
		}
		else = {
			custom_tooltip = menp_beyond_infinity_crisis_warning
		}
		if = {
			limit = {
				has_beyond_infinity_level = { LEVEL = 3 }
			}
			if = {
				limit = {
					is_galactic_emperor = yes
				}
				custom_tooltip = unlock_resolution_beyond_infinity_emperor
			}
			else_if = {
				limit = {
					is_galactic_custodian = yes
				}
				custom_tooltip = unlock_resolution_beyond_infinity_custodian
			}
			else_if = {
				limit = {
					is_galactic_community_formed = yes
				}
				custom_tooltip = unlock_resolution_beyond_infinity
			}
		}
		hidden_effect = {
			country_event = { id = time_dimensional_horror_crisis.1 days = 360 random = 360 }
		}
	}
}
