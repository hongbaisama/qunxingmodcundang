horror_titan_limit = {
	ship_types = {
		horror_titan
	}
	
	base = 40
	
	show = {
		is_scope_valid = yes
		has_country_flag = gain_time_titan
		NOR = {
			has_country_flag = upgrade_to_time_titan_finished
			has_country_flag = upgrade_to_end_titan_finished
		}
	}
}

time_titan_limit = {
	ship_types = {
		time_titan
	}
	
	base = 80
	
	show = {
		is_scope_valid = yes
		has_country_flag = gain_time_titan
		has_country_flag = upgrade_to_time_titan_finished
		NOT = { has_country_flag = upgrade_to_end_titan_finished }
	}
}
