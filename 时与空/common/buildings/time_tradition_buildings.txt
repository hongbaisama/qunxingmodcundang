# 研究所

building_horror_research = {
	base_buildtime = @b2_time
	planet_limit = 1
	
	category = research
	
	building_sets = {
		government
		research
		engineering
	}

	potential = {
		is_special_colony_type = no
		owner = {
			OR = {
				has_tradition = tr_horror_3
				has_country_flag = unlock_tech_horror_3_a_1
			}
		}
	}

	allow = {
		has_upgraded_capital = yes
	}
	
	inline_script = {
		script = buildings/building_akx_research_modifiers
		LEVEL = 1
	}

	resources = {
		category = planet_buildings
		cost = {
			sr_dark_matter = 25
			volatile_motes = 25
		}
		upkeep = {
			sr_dark_matter = 0.05
			volatile_motes = 0.05
			mult = value:num_time_normal_resources_value
		}
	}
	
	upgrades = {
		"building_time_research"
	}
	
	prerequisites = {
		"tech_horror_3_a_1"
	}
	
	triggered_desc = {
		text = building_horror_research_effect
	}
	
	on_built = {
		owner = { set_country_flag = horror_focus_11_a_finished }
	}
}
