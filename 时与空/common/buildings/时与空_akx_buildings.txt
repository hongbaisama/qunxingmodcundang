building_akx_worm_1 = {
	base_buildtime = @b4_time
	planet_limit = 1
	empire_limit = { base = 1 }
	can_demolish = no
	can_be_ruined = no
	can_be_disabled = no
	category = resource
	
	building_sets = { akx_worm }

	potential = {
		is_capital = yes
		is_special_colony_type = no
		owner = {
			is_ai = no
			has_country_flag = horizonsignal_trigger
		}
	}
	
	destroy_trigger = {
		OR = {
			is_capital = no
			is_special_colony_type = yes
			num_zones = {
				type = zone_akx_worm
				value <= 0
			}
			owner = {
				NOT = { has_country_flag = horizonsignal_trigger }
			}
		}
	}
	
	allow = {
		custom_tooltip = {
			fail_text = NEED_ZONE_AKX_WORM
			num_zones = {
				type = zone_akx_worm
				value > 0
			}
		}
	}
	
	triggered_country_modifier = {
		country_energy_produces_mult = 0.15
		mult = owner.value:num_akx_ap_value
	}
	
	triggered_planet_modifier = {
		planet_technician_energy_produces_add = 3
		generator_district_housing_add = 250
		mult = owner.value:num_akx_ap_value
	}
	
	prerequisites = {
		"tech_akx_worm_1"
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = volatile_motes
			COST = @b4_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		cost = {
			volatile_motes = @b4_rare_cost
		}
		upkeep = {
			volatile_motes = 5
		}
		produces = {
			energy = 1
			mult = value:building_akx_worm_value
		}
	}
	
	triggered_desc = {
		trigger = {
			exists = owner
			owner = {
				NOR = {
					has_ascension_perk = ap_horror
					has_ascension_perk = ap_time
				}
			}
		}
		text = building_akx_worm_1_effect
	}
	
	triggered_desc = {
		trigger = {
			exists = owner
			owner = {
				has_ascension_perk = ap_horror
				NOT = { has_ascension_perk = ap_time }
			}
		}
		text = building_akx_worm_1_effect_ap_horror
	}
	
	triggered_desc = {
		trigger = {
			exists = owner
			owner = {
				has_ascension_perk = ap_time
			}
		}
		text = building_akx_worm_1_effect_ap_time
	}
}

building_akx_worm_3 = {
	base_buildtime = @b4_time

	can_be_ruined = no
	can_be_disabled = no
	category = research
	
	building_sets = { akx_worm }

	potential = {
		is_capital = yes
		is_special_colony_type = no
		owner = {
			is_ai = no
			has_country_flag = horizonsignal_trigger
		}
	}
	
	destroy_trigger = {
		OR = {
			is_capital = no
			is_special_colony_type = yes
			num_zones = {
				type = zone_akx_worm
				value <= 0
			}
			owner = {
				NOT = { has_country_flag = horizonsignal_trigger }
			}
		}
	}
	
	allow = {
		custom_tooltip = {
			fail_text = NEED_ZONE_AKX_WORM
			num_zones = {
				type = zone_akx_worm
				value > 0
			}
		}
		if = {
			limit = {
				has_global_flag = has_wsg_mod
			}
			# 舰R也有一个圆环学院
			custom_tooltip = {
				fail_text = NEED_AKX_WORM_2_FOR_WSG
				has_building = building_akx_worm_2
			}
		}
		else = {
			custom_tooltip = {
				fail_text = NEED_AKX_WORM_2
				has_building = building_akx_worm_2
			}
		}
		custom_tooltip = {
			fail_text = REQUIREMENT_OF_ORIGIN_TIME
			owner = {
				if = {
					limit = {
						# 可触发“时之螶：交流”
						has_origin_time = yes
						NOT = { has_country_flag = interrupt_time_black_hole_chain }
						ruler = { is_origin_time_ruler = yes }
					}
					OR = {
						has_tech_option = tech_time_blocker_3
						has_technology = tech_time_blocker_3
					}
				}
				NOT = { has_event_chain = TIME_BLACK_HOLE_CHAIN }
			}
		}
	}
	
	triggered_country_modifier = {
		country_physics_research_produces_mult = 0.15
		country_engineering_research_produces_mult = 0.15
		mult = owner.value:num_akx_ap_value
	}
	
	triggered_planet_modifier = {
		planet_physicists_physics_research_produces_add = 1.5
		planet_engineers_engineering_research_produces_add = 1.5
		mult = owner.value:num_akx_ap_value
	}

	prerequisites = {
		"tech_akx_worm_3"
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher

			NOMADIC_RESOURCE = exotic_gases
			COST = @b4_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		cost = {
			exotic_gases = @b4_rare_cost
		}
		upkeep = {
			exotic_gases = 5
		}
		produces = {
			physics_research = 0.5
			engineering_research = 0.5
			mult = value:building_akx_worm_value
		}
	}
	
	triggered_desc = {
		trigger = {
			exists = owner
			owner = {
				NOR = {
