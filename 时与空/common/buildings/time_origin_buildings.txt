# 机械神教
building_adeptus_mechanicus_1 = {
	icon = building_organic_sanctuary
	base_buildtime = @b1_time

	category = unity
	
	building_sets = {
		adeptus_mechanicus
		urban
		unity
	}

	potential = {
		OR = {
			is_special_colony_type = no
			has_modifier = resort_colony
		}
		exists = owner
		owner = { has_origin = origin_adeptus_mechanicus }
	}

	destroy_trigger = {
		exists = owner
		OR = {
			owner = {
				NOT = { has_origin = origin_adeptus_mechanicus }
			}
			AND = {
				owner = { is_ai = yes }
				count_owned_pop_amount = {
					limit = {
						has_citizenship_type = {
							type = citizenship_adeptus_mechanicus
							country = owner
						}
					}
					count <= 2000
				}
				num_buildings = { type = building_adeptus_mechanicus_1 value > 3 }
			}
			AND = {
				owner = { is_ai = yes }
				count_owned_pop_amount = {
					limit = {
						has_citizenship_type = {
							type = citizenship_adeptus_mechanicus
							country = owner
						}
					}
					count <= 4000
				}
				num_buildings = { type = building_adeptus_mechanicus_1 value > 5 }
			}
		}
	}

	convert_to = {
		building_luxury_residence
		building_communal_housing
		building_organic_sanctuary
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = rare_crystals
			COST = @b1_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		upkeep = {
			energy = @b1_upkeep
		}
	}

	planet_modifier = {
		job_adeptus_mechanicus_add = 1500
	}

	upgrades = {
		"building_adeptus_mechanicus_2"
	}

	# We need to cheat here and tell AI that this will probably give some complex output since AI can not evaluate what this building does without the pops which it cant get without the building - chicken and egg
	ai_resource_production = {
		unity = 10
		physics_research = 10
		alloys = 5

		trigger = {
			NOT = { has_building = building_adeptus_mechanicus_1 }
		}
	}

	ai_weight_coefficient = 1.2

	# Here we try to adress the second issue that the AI cant evaluate the pop job output of the bio trophy given the happiness trigger that was put inplace to guard against players exploiting bio trophies

	#science
	ai_resource_production = {
		unity = 10
		physics_research = 10

		trigger = {
			any_owned_pop_group = {
				is_pop_category = adeptus_mechanicus_unemployment
			}

			planet_resource_compare = {
				resource = physics_research
				value >= 100
				type = produces
			}
		}
	}

	#alloys
	ai_resource_production = {
		unity = 10
		alloys = 10

		trigger = {
			any_owned_pop_group = {
				is_pop_category = adeptus_mechanicus_unemployment
			}

			planet_resource_compare = {
				resource = alloys
				value >= 100
				type = produces
			}
		}
	}

	#cgs
	ai_resource_production = {
		unity = 10
		consumer_goods = 10

		trigger = {
			any_owned_pop_group = {
				is_pop_category = adeptus_mechanicus_unemployment
			}

			planet_resource_compare = {
				resource = consumer_goods
				value >= 100
				type = produces
			}
		}
	}
}
