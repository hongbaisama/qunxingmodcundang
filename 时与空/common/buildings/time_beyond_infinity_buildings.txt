	}

		always = no
	}
	
	inline_script = {
		script = buildings/building_beyond_infinity_end_modifiers
		TYPE = time_machine
		JOB = time_traveler
		LEVEL = 1
	}
	
	upgrades = {
		building_infinity_time_machine
	}

	resources = {
		category = planet_buildings
		cost = {
			sr_dark_matter = 1000
		}
		upkeep = {
			sr_dark_matter = 10
		}
	}
	
	prerequisites = {
		"tech_akx_worm_3"
	}
	
	show_tech_unlock_if = {
		unlock_building_time_machine = yes
	}
	
	triggered_desc = {
		trigger = {
			has_global_flag = time_machine_is_defective
		}
		text = repair_defective_time_machine_effect
	}
	
	on_built = {
		set_variable = {
			which = exists_building_time_machine
			value = 3
		}
		if = {
			limit = {
				has_global_flag = time_machine_is_defective
			}
			carrier_event = { id = beyond_infinity.2 }
		}
	}
}

building_tower_of_infinity = {
	base_buildtime = 0

	category = amenity
	
	building_sets = {
		government
		urban
		industrial
		foundry
		factory
		fortress
		trade
		entertainment
		resort
	}
	
	potential = {
		is_special_colony_type = no
		owner = { has_ascension_perk = ap_beyond_infinity }
	}
	
	destroy_trigger = {
		OR = {
			is_special_colony_type = yes
			owner = {
				NOT = { has_ascension_perk = ap_beyond_infinity }
			}
		}
	}
	
	allow = {
		has_major_upgraded_capital = yes
	}
	
	triggered_planet_modifier = {
		infinity_spectrum = 0.1
	}
	
	triggered_planet_modifier = {
		planet_stability_add = 2.5
		planet_amenities_add = 500
		planet_housing_add = 500
		planet_crime_add = -5
		mult = value:num_time_modifier|BASE|0.9|MODIFIER|infinity_spectrum|MIN|1| # 不包括本身
	}
	
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = { has_crisis_perk = menp_beyond_infinity_building_4 }
		}
		planet_stability_mult = 0.025
		planet_amenities_mult = 0.05
		planet_housing_mult = 0.05
		planet_crime_mult = -0.05
		mult = value:num_time_modifier|BASE|0.9|MODIFIER|infinity_spectrum|MIN|1| # 不包括本身
