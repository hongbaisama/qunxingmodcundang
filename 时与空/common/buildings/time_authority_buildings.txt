	icon = building_sanctuary_of_toil
	base_buildtime = 1800
	planet_limit = 1
	empire_limit = { base = 1 }
	position_priority = 5

	category = government
	
	building_sets = {
		zone_time_physics
		zone_time_society
	}

		building_hope_institution
	}
	
	country_modifier = {
		country_official_cap_add = 2
		official_skill_add = 2

			}
		}
		country_official_cap_add = 1
		official_skill_add = 1
		official_exp_gain = 0.25
	}
	
	inline_script = {
		script = buildings/building_origin_time_authority_modifiers

		ETHIC_1 = egalitarian
		ETHIC_2 = xenophile
		JOB_1 = hope_guardian
		JOB_2 = hope_disseminator
	}
	
	triggered_desc = {
		text = building_hope_parliament_effect
	}
}
