	from = pc_gas_giant
	to = pc_gas_colony
	
	resources = {
		category = terraforming

		cost = {
			volatile_motes = 1000
			rare_crystals = 1000
			exotic_gases = 1000
		}
	}
	
	duration = 7200
	
	condition = {
		has_technology = tech_climate_restoration
	}

	potential = {
		has_origin = origin_gas_colony
		has_ascension_perk = ap_origin_gas_colony
		from = {
			NOT = { has_carrier_flag = prevent_gas_colony_change }
		}
	}

	effect = {
		from = {
			carrier_event = { id = gas.100 }
		}
	}
	
	ai_weight = {
		weight = 0
	}
}
