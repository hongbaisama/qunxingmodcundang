# The Shroud
country_event = {
	id = utopia.2655
	title = "utopia.2655.name"
	desc = {
		text = utopia.2655.desc
		trigger = {
			NOT = { has_origin = origin_psionic }
		}
	}
	desc = {
		text = utopia.2655.desc.psionic
		trigger = {
			has_origin = origin_psionic
		}
	}
	picture = GFX_evt_psionics
	show_sound = {
		trigger = {
			has_shroud_dlc = yes
			is_robot_empire = yes
		}
		sound = event_psionic_robot
	}
	show_sound = {
		trigger = {
			OR = {
				has_shroud_dlc = no
				is_robot_empire = no
			}
		}
		sound = event_psionic
	}
	trackable = yes

	is_triggered_only = yes

	trigger = {
		has_tradition = tr_psionics_breach_shroud
	}

	option = {
		name = "utopia.2655.a"
		trigger = {
			NOT = { has_origin = origin_psionic }
		}
		hidden_effect = { save_event_target_as = breach_shroud_owner }
		capital_scope = {
			enable_special_project = {
				name = "BREACHING_SHROUD_PROJECT"
				location = this
				owner = root
			}
		}
	}
	
	option = {
		name = "utopia.2655.psionic"
		trigger = {
			has_origin = origin_psionic
		}
		hidden_effect = { save_event_target_as = breach_shroud_owner }
		capital_scope = {
			enable_special_project = {
				name = "BREACHING_SHROUD_PSIONIC_PROJECT"
				location = this
				owner = root
			}
		}
	}
}
