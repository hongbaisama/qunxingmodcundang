		}

			always = no
		}

		owner = {

		}

		name = dimension_traveler.1.a

			add_monthly_resource_mult = {
				resource = physics_research
				value = @tier3researchreward
				min = @tier3researchmin
				max = @tier3researchmax
			}
		}
	}
}

			owner = {
				is_ai = no

				NOT = { has_country_flag = first_meet_dimension_traveler }

			enable_special_project = {
				name = DIMENSION_TRAVELER_PROJECT_1
				location = this
				owner = root.owner
			}
		}
		owner = {
			create_timeline_dimension_traveler_begin = yes
			if = {
				limit = {
					has_country_flag = first_meet_white_hole_show_desc
				}
				add_monthly_resource_mult = {
					resource = physics_research
					value = @tier3researchreward
					min = @tier3researchmin
					max = @tier3researchmax
				}
			}
		}
	}
	
	option = {
		name = dimension_traveler.2.b
		trigger = {
			owner = {
				NOT = { has_origin = origin_white_hole }
			}
		}
		owner = {
			create_timeline_dimension_traveler_not_begin = yes
			add_monthly_resource_mult = {
				resource = influence
