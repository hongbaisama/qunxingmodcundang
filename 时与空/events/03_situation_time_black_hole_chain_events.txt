fleet_event = {
	id = time.201
	title = time.201.name
	desc = time.201.desc
	picture = GFX_evt_black_hole
	show_sound = event_alien_signal
	location = event_target:time_black_hole
	trackable = yes

	is_triggered_only = yes

	trigger = {
		if = {
			limit = {
				# 船上有科学家
				exists = leader
				leader = { leader_class = scientist }
			}
			if = {
				limit = {
					exists = owner
					owner = {
						is_ai = no
						has_origin_time = yes # 神秘奇点起源
						NOT = { has_country_flag = time_201_triggered } # 不可重复触发
					}
				}
				if = {
					limit = {
						owner = {
							has_country_flag = worm_events_available # 可触发时虫事件
							ruler = { is_origin_time_ruler = yes } # 统治者是圆环之子
						}
					}
					# 是符合条件的黑洞星系
					from = {
						is_time_black_hole_star_class = yes
						star = {
							is_time_black_hole_planet_class = yes
							is_origin_time_planet_class = no # 恒星是神秘奇点则不会触发
						}
						NOR = {
							has_star_flag = horizonsignal_system
							is_same_value = event_target:dimension_traveler_system
						}
					}
				}
				else = {
					always = no
				}
			}
			else = {
				always = no
			}
		}
		else = {
			always = no
		}
	}

	immediate = {
		owner = { set_country_flag = time_201_triggered }
		from = {
			set_star_flag = time_black_hole_system
			save_global_event_target_as = time_black_hole_system
			star = { save_global_event_target_as = time_black_hole }
		}
	}

	option = {
		name = time.201.a
		begin_event_chain = {
			event_chain = TIME_BLACK_HOLE_CHAIN
			target = root.owner
		}
		event_target:time_black_hole = {
			enable_special_project = {
				name = TIME_BLACK_HOLE_PROJECT
				location = this
				owner = root.owner
			}
		}
	}

	option = {
		name = time.201.b
		owner = {
			add_monthly_resource_mult = {
				resource = influence
				value = @tier2influencereward
				min = @tier2influencemin
				max = @tier2influencemax
			}
			hidden_effect = {
				set_country_flag = interrupt_time_black_hole_chain
			}
		}
	}
}
