
namespace = gas

country_event = {
	id = gas.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_origin = origin_gas_colony
	}
	
	immediate = {
		give_technology = { tech = "tech_mine_volatile_motes" message = no }
		give_technology = { tech = "tech_mine_exotic_gases" message = no }
		give_technology = { tech = "tech_mine_rare_crystals" message = no }
		give_technology = { tech = "tech_genome_mapping" message = no }
		give_technology = { tech = "tech_cloning" message = no }
		give_technology = { tech = "tech_epigenetic_triggers" message = no }
		if = {
			limit = {
				is_ai = yes
			}
			set_variable = {
				which = ai_gas_colony_cheat
				value = 0
			}
			country_event = { id = gas.1000 days = 1800 random = 1800 }
		}
		capital_scope = {
			# 设置区划、区域与建筑
			inline_script = "game_start/rural_districts_initializer"
			inline_script = "game_start/shipsets"
			inline_script = "game_start/species_lithoids"
			while = {
				limit = {
					has_district = district_city
				}
				remove_district = district_city
			}
			while = {
				limit = {
					has_district = district_hive
				}
				remove_district = district_hive
			}
			if = {
				limit = {
					has_district = district_gas_city
				}
				while = {
					count = 2
					add_district = district_gas_city
				}
			}
			else_if = {
				limit = {
					has_district = district_gas_hive
				}
				while = {
					count = 2
					add_district = district_gas_hive
				}
			}
			while = {
				count = energy_districts
				add_district = district_gas_generator
			}
			while = {
				count = mining_districts
				add_district = district_gas_mining
			}
			while = {
				count = farming_districts
				add_district = district_gas_farming
			}
			clear_variable = energy_districts
			clear_variable = mining_districts
			clear_variable = farming_districts
			if = {
				limit = {
					owner = { is_regular_empire = yes }
				}
				add_zone = {
					district = district_gas_city
					zone = zone_research_unity
					zone_slot = 1
				}
				add_zone = {
					district = district_gas_city
					zone = zone_industrial
					zone_slot = 2
				}
				add_building = {
					building = building_factory_1
					zone = zone_industrial
				}
				## 只有唯物和唯心限定了区域
				inline_script = "game_start/ethic_spiritualist"
				inline_script = "game_start/ethic_materialist"
			}
			else = {
				add_zone = {
					district = district_gas_hive
					zone = zone_research_unity
					zone_slot = 1
				}
				add_zone = {
					district = district_gas_hive
					zone = zone_foundry
					zone_slot = 2
				}
			}
			validate_and_repair_planet_buildings_and_districts = yes
		}
	}
}

# 在empire_init_capital_colony与on_game_start_country间会进行一次修正更新，必须在这里添加地块
colony_event = {
	id = gas.5
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		is_planet_class = pc_gas_colony
		owner = { has_origin = origin_gas_colony }
	}
	
	immediate = {
		set_planet_size = 40
		set_carrier_flag = original_gas_colony
		# 设置地块
		reroll_pc_gas_colony_deposits = yes
		add_blocker = {
			type = d_gas_blocker_1
			blocked_deposit = d_gas_generator
		}
		add_blocker = {
			type = d_gas_blocker_1
			blocked_deposit = d_gas_mining
		}
		add_blocker = {
			type = d_gas_blocker_1
			blocked_deposit = d_gas_farming
		}
		while = {
			count = trigger:num_guaranteed_colonies
			random_list = {
				1 = {
					add_deposit = d_gas_generator
					add_deposit = d_gas_mining
				}
				1 = {
					add_deposit = d_gas_mining
					add_deposit = d_gas_farming
				}
				1 = {
					add_deposit = d_gas_farming
					add_deposit = d_gas_generator
				}
			}
		}
	}
}

	}
	
	immediate = {
		from = {
			set_carrier_flag = gas_9_triggered_by@root.owner

				id = gas.10
				days = 1
				scopes = { from = from }
			}
		}
	}
}

country_event = {
	id = gas.1000
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		has_origin = origin_gas_colony
		is_ai = yes
	}
	
	immediate = {
		country_event = { id = gas.1000 days = 1800 random = 1800 }
		if = {
			limit = {
				has_ascension_perk = ap_origin_gas_colony
				has_technology = tech_climate_restoration
			}
			if = {
				limit = {
					any_planet_within_border = {
						is_planet_class = pc_gas_giant
						is_terraforming = no
						has_anomaly = no
					}
					has_country_resource = {
						type = volatile_motes
						amount >= 1000
					}
					has_country_resource = {
						type = rare_crystals
						amount >= 1000
					}
					has_country_resource = {
						type = exotic_gases
						amount >= 1000
					}
				}
				random_planet_within_border = {
					limit = {
						is_planet_class = pc_gas_giant
						is_terraforming = no
						has_anomaly = no
					}
					change_pc = pc_gas_colony
					carrier_event = { id = gas.100 }
					create_colony = {
						owner = root
						species = root.owner_main_species
					}
				}
				add_resource = {
					volatile_motes = -1000
					rare_crystals = -1000
					exotic_gases = -1000
				}
			}
		}
		if = {
			limit = {
				check_variable = {
					which = ai_gas_colony_cheat
					value < 5
				}
			}
			if = {
				limit = {
					any_planet_within_border = {
						is_planet_class = pc_gas_giant
						is_terraforming = no
						has_anomaly = no
					}
					country_has_energy_or_minerals = {
						TYPE = energy
						AMOUNT = 500
					}
				}
				add_resource = { energy = -500 }
				random_list = {
					3 = {
						change_variable = {
							which = ai_gas_colony_cheat
							value = 1
						}
						random_planet_within_border = {
							limit = {
								is_planet_class = pc_gas_giant
								is_terraforming = no
								has_anomaly = no
							}
							change_pc = pc_gas_colony
							carrier_event = { id = gas.100 }
							create_colony = {
								owner = root
								species = root.owner_main_species
							}
						}
					}
					1 = {
					}
				}
			}
		}
	}
