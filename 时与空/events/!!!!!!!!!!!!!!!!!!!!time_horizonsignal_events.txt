# 触发时虫事件链（允许格式塔）
# triggered by on_entering_system_fleet
# chance to trigger the first time a player enters a black hole system
fleet_event = {
	id = akx.9000
	title = akx.9000.name
	desc = akx.9000.desc
	picture = GFX_evt_black_hole
	show_sound = event_alien_signal
	location = event_target:horizonsignal_black_hole
	trackable = yes

	is_triggered_only = yes

	fire_only_once = yes

	trigger = {
		if = {
			limit = {
				NOT = { has_global_flag = akx.9000.fired }
			}
			if = {
				limit = {
					from = { # solar system scope
						is_time_black_hole_star_class = yes
						has_star_flag = horizonsignal_spawn
					}
				}
				exists = leader
				leader = { leader_class = scientist }
				exists = owner
				owner = {
					is_ai = no
					is_nomadic = no
					is_homicidal = no
					is_wilderness_empire = no
					NOT = { has_origin = origin_endbringers }
					# 多人游戏下，部分国家无法抢夺时虫事件
					if = {
						limit = {
							is_multiplayer = yes
						}
						NOR = {
							has_origin = origin_blessed_by_god
							has_origin = origin_equinox_seal
						}
					}
				}
			}
			else = {
				always = no
			}
		}
		else = {
			always = no
		}
	}

	immediate = {
		set_global_flag = akx.9000.fired
		from = {
			set_star_flag = horizonsignal_system
			random_system_planet = {
				limit = {
					is_star = yes
				}
				save_global_event_target_as = horizonsignal_black_hole
			}
			save_global_event_target_as = horizonsignal_system
		}
		owner = {
			observer_event = { id = observer.70 }
			set_country_flag = horizonsignal_trigger # 神眷之族被抛弃后、彼岸封印完成封印后仍可触发
			set_country_flag = forbid_embark
		}
		if = {
			limit = {
				any_playable_country = {
					has_event_chain = PC_TIME_END_CHAIN
				}
			}
			every_playable_country = {
				limit = {
					has_event_chain = PC_TIME_END_CHAIN
				}
				end_event_chain = PC_TIME_END_CHAIN
			}
		}
	}
	
	# 常规
	option = {
		name = akx.9000.normal
		trigger = {
			owner = {
				NOR = {
					has_origin_time = yes
					has_origin = origin_blessed_by_god
					has_origin = origin_equinox_seal
				}
			}
		}
		event_target:horizonsignal_black_hole = {
			enable_special_project = {
				name = HORIZON_SIGNAL_PROJECT
				location = this
				owner = root.owner
			}
		}
	}
	
	# 神秘奇点
	option = {
		name = akx.9000.origin_time
		trigger = {
			owner = { has_origin_time = yes }
		}
		event_target:horizonsignal_black_hole = {
			enable_special_project = {
				name = HORIZON_SIGNAL_PROJECT
				location = this
				owner = root.owner
			}
		}
		owner = {
			give_situation_pc_time_reward = { LEVEL = 3 }
		}
	}
	
	# 神眷之族
	option = {
		name = akx.9000.blessed_by_god
		trigger = {
			owner = { has_origin = origin_blessed_by_god }
		}
		custom_tooltip = akx.9000.blessed_by_god.tooltip
		owner = {
			add_monthly_resource_mult = {
				resource = unity
				value = @tier4unityreward
				min = @tier4unitymin
				max = @tier4unitymax
			}
		}
	}
	
	# 彼岸封印
	option = {
		name = {
			text = akx.9000.equinox_seal.a
			trigger = {
				owner = {
					NOT = { has_country_flag = unseal_equinox_seal }
				}
			}
		}
		name = {
			text = akx.9000.equinox_seal.b
			trigger = {
				owner = {
					has_country_flag = unseal_equinox_seal
				}
			}
		}
		trigger = {
			owner = { has_origin = origin_equinox_seal }
		}
		owner = {
			if = {
				limit = {
					has_country_flag = unseal_equinox_seal
				}
				add_monthly_resource_mult = {
					resource = physics_research
					value = @tier5researchreward
					min = @tier5researchmin
					max = @tier5researchmax
				}
			}
			else = {
				add_monthly_resource_mult = {
					resource = physics_research
					value = @tier4researchreward
					min = @tier4researchmin
					max = @tier4researchmax
				}
			}
		}
	}
