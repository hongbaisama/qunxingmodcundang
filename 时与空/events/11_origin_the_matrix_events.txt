
namespace = origin_the_matrix

country_event = {
	id = origin_the_matrix.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_origin = origin_the_matrix
	}
	
	immediate = {
		give_technology = { tech = "tech_encryption_1" message = no }
		give_technology = { tech = "tech_decryption_1" message = no }
		add_research_option = tech_encryption_2
		add_research_option = tech_decryption_2
		capital_scope = {
			set_planet_size = 3
			planet = {
				set_planet_entity = {
					entity = "neural_calculator_phase_1_entity"
				}
			}
			remove_all_districts = yes
			remove_all_buildings = yes
			clear_deposits = yes
			set_carrier_flag = megastructure
			add_building = building_origin_the_matrix_lathe_capital
			add_building = building_origin_the_matrix_lathe_overclocker
			add_building = building_origin_the_matrix_lathe_preserver
			add_building = building_origin_the_matrix_lathe_reactor
			validate_and_repair_planet_buildings_and_districts = yes
			while = {
				count = 20
				time_mod_kill_single_pop = yes
				add_deposit = d_the_matrix_blocker
			}
			if = {
				limit = {
					owner_main_species = { is_lithoid = yes }
				}
				set_carrier_flag = d_the_matrix_blocker_for_lithoid
			}
		}
	}
}
