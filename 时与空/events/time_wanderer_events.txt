# 抵达星系
system_event = {
	id = time_wanderer.2
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = { has_global_flag = time_wanderer_end }
	}
	
	immediate = {
		set_timed_star_flag = {
			flag = recent_time_wanderer_system
			years = 10
		}
		add_modifier = { modifier = time_wanderer_system_modifier }
		save_global_event_target_as = time_wanderer_system
		# 循环
		system_event = { id = time_wanderer.4 days = 180 random = 180 }
		# 选择星球
		if = {
			limit = {
				count_system_planet = {
					count > 0
				}
			}
			if = {
				limit = {
					any_system_planet = {
						is_time_black_hole_planet_class = yes
					}
				}
				random_system_planet = {
					limit = {
						is_time_black_hole_planet_class = yes
					}
					choose_as_time_wanderer_planet = yes
				}
			}
			else = {
				random_system_planet = {
					choose_as_time_wanderer_planet = yes
				}
			}
		}
		else = {
			# 新星系内无星球，也要移除旧的位置
			event_target:time_wanderer_planet = {
				remove_modifier = time_wanderer_planet_modifier
			}
			clear_global_event_target = time_wanderer_planet
		}
		# 星系拥有者获得提示
		if = {
			limit = {
				exists = owner
			}
			owner = {
				if = {
					limit = {
						is_dimension_traveler_owner = yes
						NOT = { has_country_flag = time_wanderer_10_triggered }
					}
					country_event = { id = time_wanderer.10 }
				}
				else = {
					country_event = { id = time_wanderer.6 }
				}
			}
		}
	}
}

# 随机行动
system_event = {
	id = time_wanderer.4
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = { has_global_flag = time_wanderer_end }
	}
	
	immediate = {
		# 刷新
		if = {
			limit = {
				NOT = {
					is_same_value = event_target:time_wanderer_system
				}
			}
			event_target:time_wanderer_system = {
				remove_modifier = time_wanderer_system_modifier
			}
			add_modifier = { modifier = time_wanderer_system_modifier }
			save_global_event_target_as = time_wanderer_system
		}
		# 随机行动
		random_list = {
			# 移动至相邻星系
			2 = {
				if = {
					limit = {
						count_neighbor_system = {
							count > 0
							limit = {
								NOT = { has_star_flag = recent_time_wanderer_system }
							}
						}
					}
					random_neighbor_system = {
						limit = {
							NOT = { has_star_flag = recent_time_wanderer_system }
						}
						choose_as_time_wanderer_system = yes
					}
				}
				else_if = {
					limit = {
						count_neighbor_system = {
							count > 0
						}
					}
					random_neighbor_system = {
						choose_as_time_wanderer_system = yes
					}
				}
				else = {
					refresh_time_wanderer = yes
				}
			}
			# 移动至随机星球
			0 = {
				modifier = {
					add = 3
					any_system_planet = {
						NOT = { is_same_value = event_target:time_wanderer_planet }
					}
				}
				random_system_planet = {
					limit = {
						NOT = { is_same_value = event_target:time_wanderer_planet }
					}
					choose_as_time_wanderer_planet = yes
				}
				# 循环
				system_event = { id = time_wanderer.4 days = 180 random = 180 }
			}
			# 吞噬星球
			1 = {
				modifier = {
					factor = 0
					NOT = { exists = event_target:time_wanderer_planet }
				}
				modifier = {
					factor = 0
					event_target:time_wanderer_planet = {
						OR = {
							# 不吃殖民地
							is_colony = yes
							# 不吃人工星球
							is_artificial = yes
							# 不吃正在殖民的星球
							is_under_colonization = yes
							# 不吃黑洞与白洞
							is_time_black_hole_or_white_hole_planet_class = yes
							# 不吃异常
							has_anomaly = yes
							# 不吃项目
							# （好像没有对应的trigger）
							# 不吃局势
							count_targeting_situation = {
								count > 0
							}
							# 不吃考古
							exists = archaeological_site
							# 不吃巨构依附星球
							has_any_megastructure = yes
							# 不吃MOD特殊星球
							can_not_be_changed_by_akx_worm = yes
						}
					}
				}
				event_target:time_wanderer_planet = {
					if = {
						limit = {
							is_primary_star = yes
							solar_system = { can_become_sc_black_hole_by_akx_worm = yes }
						}
						solar_system = {
							set_star_class = sc_black_hole
							# 清除戴森云、戴森球、奇迹球并返还建造次数
							worm_clear_dyson_sphere = yes
						}
					}
					set_dimensional_horrorling_planet = yes
					add_modifier = { modifier = time_wanderer_black_hole_modifier }
					# 黑洞穿行可选星系
					solar_system = { set_star_flag = has_black_hole_by_horror }
				}
				# 星系拥有者获得提示
				if = {
					limit = {
						exists = owner
					}
					owner = {
						country_event = { id = time_wanderer.7 }
					}
				}
				# 循环
				system_event = { id = time_wanderer.4 days = 180 random = 180 }
			}
		}
	}
}
