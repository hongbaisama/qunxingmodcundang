country_event = {
	id = beyond_infinity.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
	}
	
	immediate = {
		if = {
			limit = {
				has_time_machine_civic = yes
			}
			date_distortion = yes
			set_global_flag = time_machine_is_defective
			capital_scope = {
				add_building = building_defective_time_machine
				validate_and_repair_planet_buildings_and_districts = yes
				set_variable = {
					which = exists_building_time_machine
					value = 2
				}
				carrier_event = { id = beyond_infinity.2 days = 7200 random = 3600 }
			}
			every_playable_country = {
				country_event = { id = beyond_infinity.5 }
			}
			every_country = {
				limit = {
					NOT = { is_same_value = root }
				}
				add_modifier = {
					modifier = time_machine_is_defective
					days = -1
				}
			}
		}
		if = {
			limit = {
				has_circle_nexus_civic = yes
			}
			set_global_flag = circle_nexus_is_defective

				add_building = building_defective_circle_nexus

carrier_event = {
	id = beyond_infinity.2
	title = beyond_infinity.2.name
	desc = {
		text = beyond_infinity.2.desc.ruin
		trigger = {
			has_building = building_ruin_time_machine
		}
	}
	desc = {
		text = beyond_infinity.2.desc.repair
		trigger = {
			NOT = { has_building = building_ruin_time_machine }
		}
	}
	picture = GFX_evt_clocks
	show_sound = ap_rift_event_15_time_crystal
	location = root
	
	is_triggered_only = yes
	
	trigger = {
		has_global_flag = time_machine_is_defective
		has_building_time_machine = yes
	}
	
	immediate = {
		owner = { create_timeline_time_machine_end = yes }
		date_distortion = no
		remove_global_flag = time_machine_is_defective
		remove_global_flag = enable_defective_time_machine_edict
		if = {
			limit = { has_building = building_defective_time_machine }
			remove_building = building_defective_time_machine
			add_building = building_ruin_time_machine
			validate_and_repair_planet_buildings_and_districts = yes
			set_variable = {
				which = exists_building_time_machine
				value = 1
			}
		}
		every_playable_country = {
			limit = {
				NOT = { is_same_value = root.owner }
			}
			country_event = { id = beyond_infinity.8 }
		}
		every_country = {
			limit = {
				has_modifier = time_machine_is_defective
			}
			remove_modifier = time_machine_is_defective
		}
	}
	
	option = {
		name = {
			text = beyond_infinity.2.a.ruin
			trigger = {
				has_building = building_ruin_time_machine
			}
		}
		name = {
			text = beyond_infinity.2.a.repair
			trigger = {
				NOT = { has_building = building_ruin_time_machine }
			}
		}
		custom_tooltip = beyond_infinity.2.tooltip
		if = {
			limit = {
				has_building = building_ruin_time_machine
			}
			owner = {
				add_monthly_resource_mult = {
					resource = unity
					value = @tier3unityreward
					min = @tier3unitymin
					max = @tier3unitymax
				}
			}
		}
		else = {
			owner = {
				add_monthly_resource_mult = {
					resource = unity
					value = @tier5unityreward
					min = @tier5unitymin
					max = @tier5unitymax
				}
			}
		}
	}
}

		if = {
			limit = {
				is_variable_set = exists_building_time_machine
			}
			owner = { unlock_building_time_machine = yes }
			NOR = {
				has_building = building_ruin_time_machine
				has_building = building_defective_time_machine
				has_building = building_time_machine
				has_building = building_infinity_time_machine
				has_building_construction = building_time_machine
				has_building_construction = building_infinity_time_machine
			}
		}
		else = {
			always = no
		}
	}
	
	immediate = {
		remove_building = building_ruin_time_machine
		remove_building = building_defective_time_machine
		remove_building = building_time_machine
		remove_building = building_infinity_time_machine
		if = {
			limit = {
				check_variable = {
					which = exists_building_time_machine
					value = 4
				}
			}
			add_building = building_infinity_time_machine
		}
		else_if = {
			limit = {
				check_variable = {
					which = exists_building_time_machine
					value = 3
				}
			}
			add_building = building_time_machine

country_event = {
	id = beyond_infinity.100
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		if = {
			limit = {
				has_ascension_perk = ap_beyond_infinity
			}
			OR = {
				last_increased_tech = tech_beyond_infinity_physics
				last_increased_tech = tech_beyond_infinity_society
				last_increased_tech = tech_beyond_infinity_engineering
			}
		}
		else = {
			always = no
		}
	}
	
	immediate = {
		if = {
			limit = {
				last_increased_tech = tech_beyond_infinity_physics
			}
			add_beyond_infinity_tech_modifier = {
				VARIABLE = beyond_infinity_physics_variable
				MODIFIER = beyond_infinity_physics_modifier
			}
		}
		else_if = {
			limit = {
				last_increased_tech = tech_beyond_infinity_society
			}
			add_beyond_infinity_tech_modifier = {
				VARIABLE = beyond_infinity_society_variable
				MODIFIER = beyond_infinity_society_modifier
			}
		}
		else = {
			add_beyond_infinity_tech_modifier = {
				VARIABLE = beyond_infinity_engineering_variable
				MODIFIER = beyond_infinity_engineering_modifier
