country_event = {
	id = origin_pirate_king.0
	hide_window = yes
	is_triggered_only = yes

	# This = owner of ship 1 (combatant)
	# From = owner of ship 2 (destroyed)
	# FromFrom = ship 1
	# FromFromFrom = ship 2

	trigger = {
		if = {
			limit = {
				has_origin_pirate_king = yes
			}
			From = {
				OR = {
					is_default_or_player = yes
					is_fallen_empire = yes
					is_country_type = pirate
					is_country_type = dormant_marauders
					is_country_type = ruined_marauders

					is_country_type = origin_pirate_king_end

			}
		}
		else = {
			always = no
		}
	}

	immediate = {
		if = {
			limit = {
				any_situation = {
					is_situation_type = situation_origin_pirate_king
				}
			}
			random_situation = {
				limit = {
					is_situation_type = situation_origin_pirate_king
				}
				save_event_target_as = pirate_king_situation
			}
		}
		# 工程船
		if = {
			limit = {
				FromFromFrom = {
					is_ship_class = shipclass_constructor
				}
			}
			gain_origin_pirate_king_reward = {
				energy = 100
				minerals = 50
				progress = 1
			}
		}
		# 载人科研船
		else_if = {
			limit = {
				FromFromFrom = {
					is_ship_class = shipclass_science_ship

				}
			}
			gain_origin_pirate_king_reward = {
				energy = 150
				food = 50
				unity = 25
				progress = 2
			}
		}
		# 无人科研船
		else_if = {
			limit = {
				FromFromFrom = {
					is_ship_class = shipclass_science_ship
				}
			}
			gain_origin_pirate_king_reward = {
				energy = 100
