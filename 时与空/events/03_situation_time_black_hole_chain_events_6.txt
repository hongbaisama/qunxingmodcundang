situation_event = {
	id = time.261
	title = time.261.name
	desc = time.261.desc
	picture = GFX_evt_friendly_infiltration
	show_sound = event_factions
	location = event_target:time_home_planet
	situation = event_target:situation_time_militarist_vs_pacifist
	
	is_triggered_only = yes
	
	trigger = {
		exists = event_target:situation_time_militarist_vs_pacifist
	}
	
	immediate = {
		set_timed_situation_flag = {
			flag = time_261_triggered
			days = @time_ethic_random_situation_event_cooldown_middle
		}
	}
	
	abort_trigger = {
		has_situation_flag = abort_time_ethic_random_situation_event
	}
	
	abort_effect = {
		target = {
			shift_random_pop_time_ethic = {
				ETHIC = pacifist
				PERCENTAGE = @time_ethic_shift_percentage_middle
			}
		}
		refresh_time_ethic_power_variable = yes
	}
	
	option = {
		name = time.211.a
		target = {
			custom_tooltip = time_change_pop_ethic_pacifist
			hidden_effect = {
				shift_random_pop_time_ethic = {
					ETHIC = pacifist
					PERCENTAGE = @time_ethic_shift_percentage_middle
				}
			}
		}
		hidden_effect = {
			refresh_time_ethic_power_variable = yes
		}
	}
	
	option = {
		name = time.211.b
		allow = {
			owner = {
				has_resource = {
					type = influence
					amount >= 75
				}
			}
		}
		owner = {
			add_resource = { influence = -75 }
		}
	}
}
