country_event = {
	id = time_help.850
	title = time_help.850.name
	desc = time_help.850.desc

	diplomatic = yes
	force_open = yes

	picture_event_data = {
		room = no_video_feed_room
	}

	is_triggered_only = yes
	
	trigger = {
	}
	
	immediate = {
		random_owned_species = {
			limit = {
				is_same_species = root.owner_main_species
				root = { is_default_species = prev }
			}
			save_event_target_as = target_dominant_species
		}
	}
	
	option = {
		name = time_help.850.a
		allow = {
			if = {
				limit = {
					exists = event_target:target_dominant_species
				}
				custom_tooltip = {
					fail_text = "time_help.850.a.allow.1"
					event_target:target_dominant_species = {
						NOT = { is_same_value = root.owner_main_species }
					}
				}
				custom_tooltip = {
					fail_text = "time_help.850.a.allow.2"
					event_target:target_dominant_species = {
						is_same_species = root.owner_main_species
						is_same_species_class = root.owner_main_species
					}
				}
				custom_tooltip = {
					fail_text = "time_help.850.a.allow.3"
					check_variable_arithmetic = {
						which = value:time_help_850_pop_amount_value
						value >= 0
					}
				}
			}
			else = {
				custom_tooltip = {
					fail_text = "time_help.850.a.allow.0"
					always = no
				}
			}
		}
		custom_tooltip = time_help.850.a.tooltip
		change_dominant_species = {
			species = event_target:target_dominant_species
		}
		hidden_effect = {
			time_change_leader_species_for_country = yes
			country_event = { id = time_help.850 }
		}
	}
	
	option = {
		name = time_help.850.b
		custom_tooltip = time_help.850.b.tooltip
		hidden_effect = {
			time_change_leader_species_for_country = yes
			country_event = { id = time_help.850 }
		}
	}
	
	option = {
		name = time_help.850.c
		trigger = {
			has_technology = tech_end_dimension
		}
		custom_tooltip = time_help.850.c.tooltip
		hidden_effect = {
			create_pass_black_hole = yes
			country_event = { id = time_help.850 }
		}
	}
	
	option = {
		name = time_help.850.d
		trigger = {
			has_event_chain = beyond_infinity_seal_chain
			NOT = { has_global_flag = all_erasured_system_is_sealed }
		}
		custom_tooltip = time_help.850.d.tooltip
		hidden_effect = {
			every_system = {
				limit = {
					OR = {
						is_dimensional_horror_crisis_erasured_system = yes
						has_star_flag = exists_beyond_infinity_seal_project_for@root
					}
					is_dimensional_horror_crisis_sealed_system = no # 未被封印
				}
				# 刷新旧有项目
				if = {
					limit = {
						has_star_flag = exists_beyond_infinity_seal_project_for@root # 已有封印项目
					}
					star = {
						save_event_target_as = beyond_infinity_seal_project_old_target
					}
					root = {
						abort_special_project = {
							type = BEYOND_INFINITY_SEAL_PROJECT
							location = event_target:beyond_infinity_seal_project_old_target
						}
					}
				}
				# 激活封印项目
				set_star_flag = exists_beyond_infinity_seal_project_for@root
				save_event_target_as = beyond_infinity_seal_project_target
				star = {
					enable_special_project = {
						name = BEYOND_INFINITY_SEAL_PROJECT
						location = this
						owner = root
					}
				}
			}
			country_event = { id = time_help.850 }
		}
	}
	
	option = {
		name = time_help.back
