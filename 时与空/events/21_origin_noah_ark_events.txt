country_event = {
	id = origin_noah_ark.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_origin = origin_noah_ark
	}
	
	immediate = {
		capital_scope = {
			solar_system = {
				random_system_planet = {
					limit = {
						has_carrier_flag = noah_ark_home_planet
					}
					save_event_target_as = noah_ark_home_planet
				}
				if = {
					limit = {
						count_neighbor_system = { count > 0 }
					}
					system_event = { id = origin_noah_ark.2 }
				}
			}
			add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_noah_ark }
			create_archaeological_site = noah_ark_1_digsite
			save_global_event_target_as = noah_ark_starting_planet_of@root
		}
		owner_species = {
			set_species_homeworld = event_target:noah_ark_home_planet
		}
		# 针对高级AI
		every_owned_colony = {
			limit = {
				is_capital = no
			}
			time_destroy_colony = yes
		}
		every_owned_starbase = {
			limit = {
				solar_system = {
					is_capital_system = no
				}
			}
			fleet = {
				delete_fleet = this
			}
		}
	}
}

				has_modifier = noah_ark_home_planet_modifier

			always = no
		}

		add_modifier = {

		}

		if = {
			limit = {
				exists = owner
			}
			carrier_event = { id = origin_noah_ark.11 }
		}
	}
}
