country_event = {
	id = time.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_origin_time = yes
		is_default_or_player = yes #原始文明不触发
	}
	
	immediate = {
		# 银心神秘奇点
		if = {
			limit = {
				capital_scope = {
					solar_system = { has_star_flag = time_core_system }
				}
			}
			set_country_flag = time_core_country
			capital_scope = {
				solar_system = {
					save_global_event_target_as = time_core_system
					closest_system = {
						use_bypasses = no
						min_steps = 1
						max_steps = 10
						save_event_target_as = time_core_system_closest_system
					}
					# 以防万一
					if = {
						limit = {
							NOT = { exists = event_target:time_core_system_closest_system }
						}
						random_system = {
							limit = {
								NOT = { is_same_value = event_target:time_core_system }
							}
							save_event_target_as = time_core_system_closest_system
						}
					}
				}
			}
			
			set_spawn_system_batch = begin
			
			no_scope = {
				spawn_system = {
					min_distance = 0
					max_distance = 0
					min_orientation_angle = 0
					max_orientation_angle = 0
					initializer = fake_time_core_init
					hyperlane = no
					authorize_spawn_on_galactic_core = yes
					effect = {
						save_event_target_as = fake_time_core_system
					}
				}
			}
			
			event_target:time_core_system = {
				move_system = {
					target = event_target:fake_time_core_system
					min_jumps = 0
					max_jumps = 0
					min_distance = 0
					max_distance = 0
					hyperlane = no
					reset_terra_incognita = yes
				}
			}
			
			event_target:fake_time_core_system = {
				move_system = {
					target = event_target:time_core_system_closest_system
					min_distance >= 1
					max_distance <= 10
					max_jumps = 0
					hyperlane = yes
					reset_terra_incognita = yes
				}
				every_neighbor_system_euclidean = {
					limit = {
						NOR = {
							has_star_flag = surveillance_supercomputer_system
							has_special_star_flag_trigger = yes
							has_natural_wormhole = yes
							any_system_planet = {
								is_colonizable = yes
							}
						}
						NOT = { has_hyperlane_to = prev }
						distance = {
							source = prev
							max_distance < 30
							type = euclidean
						}
					}
					add_hyperlane = {
						from = prev
						to = this
					}
				}
			}
			
			set_spawn_system_batch = end
		}
		
		# 通用
		set_country_flag = origin_time_empire
		begin_event_chain = {
			event_chain = PC_TIME_CHAIN
			target = root
		}
		start_situation = {
			type = situation_pc_time
			target = capital_scope
			effect = {
				save_global_event_target_as = situation_pc_time
			}
		}
		# 一些基础资源储量（恰好够造一个区划）
		add_resource = {
			energy = 300
			minerals = 300
			food = 300
			consumer_goods = 150
			alloys = 50
			volatile_motes = 50
			rare_crystals = 50
			exotic_gases = 50
			sr_dark_matter = 25
		}
		if = {
			limit = {
				is_ai = yes
			}
			country_event = { id = time.5 days = 1080 random = 1440 }
		}
		capital_scope = {
			set_carrier_flag = time_home_planet_of@root
			set_planet_size = 30
			planet = { change_colony_foundation_date = -725888 } # 0.1.1
			# 初始人口
			every_owned_pop_group = {
				limit = {
					is_same_species = root.owner_main_species
				}
				kill_pop_group = {
					pop_group = this
					percentage = 0.9
				}
			}
			# 初始人口取整
			while = {
				limit = {
					check_variable_arithmetic = {
						which = value:num_modulo_of_time_pop_amount_value|MODULO|10|
						value > 0
					}
				}
				if = {
					limit = {
						check_variable_arithmetic = {
							which = value:num_modulo_of_time_pop_amount_value|MODULO|10|
							value >= 8
						}
					}
					create_pop_group = {
						species = owner_main_species
						size = 1
					}
				}
				else = {
					weighted_random_owned_pop_group = {
						kill_pop_group = {
							pop_group = this
							amount = 1
						}
					}
				}
			}
			# 设置地块
			reroll_pc_time_deposits = yes
			owner = { set_country_flag = hide_time_blocker_cleared_message }
			while = {
				count = trigger:num_guaranteed_colonies
				d_time_blocker_0_cleared_effect = yes
				add_deposit = d_time_blocker_1
				random_deposit = {
					limit = {
						is_deposit_type = d_time_blocker_1
					}
					clear_blocker = yes
				}
			}
			owner = { remove_country_flag = hide_time_blocker_cleared_message }
			# 设置区划、区域与建筑
			remove_all_districts = yes
			add_district = district_time
			add_district = district_time_physics
			add_district = district_time_society
			add_district = district_time_engineering
			add_zone = {
				district = district_time
				zone = zone_time
				zone_slot = 1
			}
			add_building = {
				building = building_research_lab_1
				zone = zone_time
			}
			add_building = {
				building = building_foundry_1
				zone = zone_time
			}
			add_building = {
				building = building_factory_1
				zone = zone_time
			}
			validate_and_repair_planet_buildings_and_districts = yes
			
			set_colony_type = col_time_capital
			set_colony_type = none # 更新星球规划的同时改用自动规划
			if = {
				limit = {
					root = { has_country_flag = generate_time_0_bonus }
				}
				generate_time_0_bonus = yes
			}
		}
		ruler = {
			save_global_event_target_as = origin_time_ruler
			add_trait = {
				trait = leader_trait_ruler_time
				show_message = no
			}
		}
	}
}

	picture = GFX_evt_dimensional_horror
	show_sound = event_activating_unknown_technology
	location = event_target:akx_star
	event_chain = AP_HORROR_CHAIN
	
	is_triggered_only = yes

	trigger = {
	}
	
	immediate = {
		begin_event_chain = {
			event_chain = AP_HORROR_CHAIN
			target = root
		}
		event_target:akx_star = {
			if = {
				limit = {
					is_colonizable_or_colony = yes
				}
				remove_deposit = d_akx_dark_matter
				remove_deposit = d_orbital_akx_dark_matter
				add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_akx_dark_matter_special }
			}
			else = {
				remove_deposit = d_akx_dark_matter
				remove_deposit = d_orbital_akx_dark_matter
				add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_orbital_akx_dark_matter_special }
			}
			solar_system = {
				every_system_planet = {
					limit = {
						is_primary_star = no
						is_planet_class = pc_black_hole
					}
					remove_deposit = d_dark_matter_deposit_5
					add_deposit = d_dark_matter_deposit_10
				}
				starbase = {
					remove_starbase_component = {
						component = HORROR_AURA_1
					}
					add_starbase_component = {
						component = HORROR_AURA_2
					}
				}
			}
		}
		country_event = { id = time.500 }
	}
	
	after = {
		hidden_effect = {
			event_target:astral_rift_akx_horror = {
				clear_deposits_and_orbital_station = yes
				destroy_astral_rift = this
			}
			event_target:akx_star.solar_system = {
				spawn_astral_rift = {
					id = akx_time
					in_place_of = event_target:akx_star
					graphics_entity_name = astral_rift_akx_time_entity
					init_effect = {
						save_global_event_target_as = astral_rift_akx_time
						# 裂隙自带2星界丝缕野矿
						add_deposit = d_astral_threads_deposit_2
						add_deposit = d_astral_threads_deposit_3
						add_deposit = d_astral_threads_deposit_3
					}
				}
			}
		}
	}
	
	option = {
		name = time.10.a
		custom_tooltip = time.10.a.tooltip
		tooltip = {
			begin_event_chain = {
				event_chain = AP_HORROR_CHAIN
				target = root
			}
		}
	}
}

	id = time.15
	title = time.15.name
	desc = time.15.desc
	picture = GFX_evt_dimensional_horror
	show_sound = event_activating_unknown_technology
	location = root
	
	is_triggered_only = yes

	pre_triggers = {
		has_owner = yes
	}

	trigger = {
		if = {
			limit = {
				owner = { has_technology = tech_time_modifier }
			}
			NOT = { has_carrier_flag = time_gain_horror }
			has_horror_trophy = no
			is_under_colonization = no
			is_scope_type = planet # 不会在方舟舰上扩散
		}
		else = {
			always = no
		}
	}
	
	immediate = {
		set_carrier_flag = time_gain_horror
		if = {
			limit = {
