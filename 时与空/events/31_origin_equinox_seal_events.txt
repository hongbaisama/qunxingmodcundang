system_event = {
	id = origin_equinox_seal.8
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		NOT = { has_global_flag = finish_equinox_seal }
	}
	
	immediate = {
		set_global_flag = finish_equinox_seal
		every_playable_country = {
			limit = {
				OR = {
					has_origin = origin_equinox_seal
					has_origin = origin_equinox_glimpse
				}
			}
			country_event = { id = origin_equinox_seal.5 }
		}
		if = {
			limit = {
				exists = owner
				owner = {
					NOR = {
						has_origin = origin_equinox_seal
						has_origin = origin_equinox_glimpse
					}
				}
			}
			owner = {
				country_event = { id = origin_equinox_seal.6 }
			}
		}
		event_target:astral_rift_equinox_seal = {
			if = {
				limit = {
					exists = leader
				}
				leader = {
					kill_leader = { show_notification = no }
				}
				finish_astral_rift = yes
			}
			clear_deposits_and_orbital_station = yes
			destroy_astral_rift = this # 封印松动，被位面之魇暴力摧毁而彻底消失；主动解除封印则不会使封印彻底消失
		}
		starbase = {
			fleet = {
				delete_fleet = this
			}
		}
		star = {
			remove_deposit = d_dark_matter_deposit_3
			add_deposit = d_dark_matter_deposit_5
			create_ambient_object = {
				type = "destroyed_system"
				location = this
			}
			last_created_ambient_object = {
				set_ambient_object_flag = destroyed_system_effect
				set_location = {
					target = prev
					distance = 0
					angle = random
				}
			}
		}
		every_system_planet = {
			limit = {
				NOR = {
					is_primary_star = yes
					is_asteroid = yes
					# 黑洞白洞
					is_time_black_hole_or_white_hole_planet_class = yes
					# 星界裂隙
					is_astral_scar = yes
					# 虚空
					is_planet_class = pc_giga_empty_space
					is_planet_class = pc_giga_empty_space_asteroid
					is_planet_class = pc_giga_asteroid_artillery
					is_planet_class = pc_giga_orbital_site
					is_planet_class = pc_rtg_empty_space
				}
			}
			if = {
				limit = {
					is_capital = yes
				}
				owner = {
					ruler = {
						kill_leader = { show_notification = no }
					}
				}
			}
			if = {
				limit = {
					is_colony = yes
				}
				time_destroy_colony = yes
			}
			clear_planet_modifiers = yes
			clear_deposits_and_orbital_station = yes
			if = {
				limit = {
					is_ringworld = yes
				}
				spawn_ringworld_cracker_effect = yes
				change_pc = pc_ringworld_habitable_damaged
			}
			else_if = {
				limit = {
					is_planet_class = pc_ringworld_tech
				}
				spawn_ringworld_cracker_effect = yes
				change_pc = pc_ringworld_tech_damaged
			}
			else_if = {
				limit = {
					is_planet_class = pc_ringworld_seam
				}
				spawn_ringworld_cracker_effect = yes
				change_pc = pc_ringworld_seam_damaged
			}
			else = {
				change_pc = pc_black_hole
				random_list = {
					3 = {
						add_deposit = d_dark_matter_deposit_1
					}
					2 = {
						add_deposit = d_dark_matter_deposit_2
					}
					1 = {
						add_deposit = d_dark_matter_deposit_3
					}
				}
			}
		}
		every_system_planet = {
			limit = {
				exists = orbital_station
			}
			orbital_station = { dismantle = yes }
		}
		every_system_planet = {
			limit = { is_asteroid = yes }
			clear_deposits_and_orbital_station = yes
			clear_planet_modifiers = yes
		}
		every_fleet_in_system = {
			destroy_fleet = this
		}
		every_system_megastructure = {
			set_ruined_megastructure = yes
		}
		# 创建清醒沉眠者
		create_country = {
			name = "NAME_equinox_seal_sleeper"
			type = guardian_horror
			auto_delete = no
			flag={
				icon={
					category="spherical"
					file="flag_spherical_18.dds"
				}
				background={
					category="backgrounds"
					file="circle.dds"
				}
				colors={
					"shadow_purple"
					"black"
					"null"
					"null"
				}
			}
			effect = {
				save_global_event_target_as = equinox_seal_sleeper_country
			}
		}
		create_fleet = {
			name = "NAME_equinox_seal_sleeper"
			settings = {
				can_upgrade = no
				can_change_composition = no
				can_change_leader = no
				uses_naval_capacity = no
				spawn_debris = no
				is_boss = yes
			}
			effect = {
				set_owner = event_target:equinox_seal_sleeper_country
				create_ship = {
					name = "NAME_equinox_seal_sleeper"
					design = "NAME_equinox_seal_sleeper"
					prefix = no
					upgradable = no
				}
				set_location = {
					target = event_target:astral_rift_equinox_seal
					distance = 0
					direction = in_system
				}
				save_global_event_target_as = equinox_seal_sleeper_fleet
			}
		}
	}
}
