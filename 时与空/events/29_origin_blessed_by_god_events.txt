country_event = {
	id = origin_blessed_by_god.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_origin = origin_blessed_by_god
	}
	
	immediate = {
		give_technology = { tech = tech_mine_zro message = no }
		set_variable = {
			which = blessed_by_god_tradition_count
			value = 0
		}
		set_variable = {
			which = blessed_by_god_decision_count
			value = 0
		}
		add_resource = {
			unity = 801 # 你问我这个1哪里来的？我不到啊
			sr_zro = 10
		}
		random_list = {
			2 = {
			}
			1 = {
				set_country_flag = will_be_abandoned_by_god
			}
		}
		capital_scope = {
			set_carrier_flag = exists_building_blessed_by_god
			add_building = building_blessed_by_god
			add_zone = {
				district = district_city
				zone = zone_unity_spiritualist
				zone_slot = 1
				replace = yes
			}
			validate_and_repair_planet_buildings_and_districts = yes
			solar_system = {
				every_system_planet = {
					limit = {
						has_carrier_flag = blessed_by_god_planet
					}
					create_research_station = { owner = root }
				}
			}
		}
		if = {
			limit = {
				is_ai = yes
			}
			country_event = { id = origin_blessed_by_god.100 }
		}
	}
}

carrier_event = {
	id = origin_blessed_by_god.3
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		owner = {
			has_origin = origin_blessed_by_god
			OR = {
				NOT = { has_country_flag = recently_blessed_by_god }
				# 允许一定程度上的误差，以确保正常自动进行大祭祀
				timed_flag_days_left = {
					flag = recently_blessed_by_god
					value <= 5
				}
			}
		}
	}
	
	immediate = {
		owner = {
			remove_country_flag = recently_blessed_by_god
			create_message = {
				type = CAN_BLESSED_BY_GOD
				localization = CAN_BLESSED_BY_GOD_DESC
				days = @toast_important_message_days
				target = root
				variable = {
					type = name
					localization = PLANET
					scope = root
				}
				custom_message_text = can_blessed_by_god_custom_desc
			}
			if = {
				limit = {
					can_select_decision_blessed_by_god = yes
					root = { has_active_building = building_blessed_by_god }
				}
				root = {
					select_decision = {
						name = decision_blessed_by_god
					}
				}
			}
		}
	}
}

country_event = {

# AI自动进行大祭祀
country_event = {
	id = origin_blessed_by_god.100
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_origin = origin_blessed_by_god
		is_ai = yes
	}
	
	immediate = {
		if = {
			limit = {
				can_select_decision_blessed_by_god = yes
				any_owned_colony = {
					has_active_building = building_blessed_by_god
				}
			}
			random_owned_colony = {
				limit = {
					has_active_building = building_blessed_by_god
				}
				select_decision = {
					name = decision_blessed_by_god
				}
			}
			country_event = { id = origin_blessed_by_god.100 days = 360 }
		}
		else = {
			# AI会积极地尝试进行大祭祀
			country_event = { id = origin_blessed_by_god.100 days = 30 }
		}
	}
}
