	}

	id = origin_corrupt_brain.2

	trigger = {
		OR = {
			has_origin = origin_corrupt_brain
			has_origin = origin_corrupt_brain_end
			has_origin = origin_pure_brain
			has_origin = origin_pure_brain_true
		}
	}
	
	immediate = {
		from = {
			leader_event = { id = origin_corrupt_brain.3 days = 1 } #格式塔节点领袖在被创建时还不位于节点位置，等一天再执行效果
		}
	}

leader_event = {
	id = origin_corrupt_brain.3
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		OR = {
			is_ruler = yes
			is_gestalt_node = yes
		}
	}
	
	immediate = {
		if = {
			limit = {
				owner = {
					OR = {
						has_origin = origin_pure_brain
						has_origin = origin_pure_brain_true
					}
				}
			}
			if = {
				limit = {
					is_ruler = yes
				}
				add_trait = {
					trait = leader_trait_pure_brain
					show_message = no
				}
			}
			else = {
				add_trait = {
					trait = leader_trait_pure_node
					show_message = no
				}
			}
		}
		else = {
			if = {
				limit = {
					is_ruler = yes
				}
				add_trait = {
					trait = leader_trait_corrupt_brain
					show_message = no
				}
			}
			else = {
				add_trait = {
					trait = leader_trait_corrupt_node
					show_message = no
				}
			}
		}
		owner = {
			refresh_corrupt_brain_value = yes
		}
	}
}

			}

				trait = leader_trait_pure_node

			}
		}
		random_situation = {
			limit = {
				is_situation_type = situation_origin_corrupt_brain
			}

		}
	}
	
	option = {
		name = origin_corrupt_brain.8.a

			set_origin = origin_pure_brain_true
		}
	}
