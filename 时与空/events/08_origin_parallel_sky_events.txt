country_event = {
	id = parallel_sky.1
	title = parallel_sky.1.name
	desc = parallel_sky.1.desc
	picture = GFX_evt_kaleidoscopic_lights
	show_sound = event_mystic_reveal

	is_triggered_only = yes

	trigger = {
		has_origin = origin_parallel_sky
	}
	
	immediate = {
		capital_scope = {
			save_global_event_target_as = parallel_sky_planet
			set_planet_size = 30
			add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_parallel_sky }
			carrier_event = { id = parallel_sky.1000 days = 360 random = 720 }
		}
		set_country_flag = meet_parallel_sky_country
		random_list = {
			1 = {
				# 极端唯物平等
				create_parallel_sky_country = {
					AUTH = auth_democratic
					CIVIC = civic_parallel_sky_1
					ETHIC_1 = ethic_fanatic_materialist
					ETHIC_2 = ethic_egalitarian
					COLOR_1 = indigo
					COLOR_2 = dark_teal
					ETHIC_RULER = ethic_materialist
					ROOM = personality_erudite_explorers_room
				}
			}
			1 = {
				# 排外军国威权
				create_parallel_sky_country = {
					AUTH = auth_imperial
					CIVIC = civic_parallel_sky_2
					ETHIC_1 = ethic_militarist
					ETHIC_2 = ethic_authoritarian
					ETHIC_3 = ethic_xenophobe
					COLOR_1 = red
					COLOR_2 = black
					ETHIC_RULER = ethic_militarist
					ROOM = personality_slaving_despots_room
				}
			}
			1 = {
				# 极端唯心和平
				create_parallel_sky_country = {
					AUTH = auth_dictatorial
					CIVIC = civic_parallel_sky_3
					ETHIC_1 = ethic_fanatic_spiritualist
					ETHIC_2 = ethic_pacifist
					COLOR_1 = shadow_purple
					COLOR_2 = bright_yellow
					ETHIC_RULER = ethic_spiritualist
					ROOM = personality_spiritual_seekers_room
				}
			}
		}
	}

	option = {
		name = parallel_sky.1.a
		trigger = {
			has_origin = origin_parallel_sky
		}
		hidden_effect = {
			country_event = { id = parallel_sky.2 }
			# 防止事件效果被卡掉，导致母星未被存储为全局事件目标
			capital_scope = {
				if = {
					limit = {
						NOT = { exists = event_target:parallel_sky_planet }
					}
					save_global_event_target_as = parallel_sky_planet
				}
				if = {
					limit = {
						NOT = { planet_size = 30 }
					}
					set_planet_size = 30
				}
				if = {
					limit = {
						NOT = { has_deposit = d_parallel_sky }
					}
					add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_parallel_sky }
				}
			}
		}
	}
}
