country_event = {
	id = time_hotel.0
	title = time_hotel.0.name
	desc = time_hotel.0.desc
	picture = GFX_evt_board_meeting
	show_sound = event_alien_signal
	location = event_target:time_hotel_site
	
	mean_time_to_happen = {
		months = 12
	}

	trigger = {
		has_country_flag = worm_events_available
		NOT = { has_global_flag = time_hotel_triggered }
		any_planet_within_border = {
			is_asteroid = yes
			has_anomaly = no
			NOT = { exists = archaeological_site }
		}
	}
	
	immediate = {
		set_global_flag = time_hotel_triggered
		random_planet_within_border = {
			limit = {
				is_asteroid = yes
				has_anomaly = no
				NOT = { exists = archaeological_site }
			}
			save_global_event_target_as = time_hotel_site
		}
	}
	
	option = {
		name = time_hotel.0.a
		event_target:time_hotel_site = {
			add_anomaly = {
				category = time_hotel_categories
				target = root
			}
		}
	}
	
	option = {
		name = time_hotel.0.b
		allow = {
			is_materialist = no
		}
		add_resource = { influence = 100 }
	}
}

fleet_event = {
	id = time_hotel.7
	title = "time_hotel.7.name"
	desc = time_hotel.7.desc
	picture = GFX_evt_clocks
	show_sound = event_big_door_opening
	location = from

	archaeology = yes

	is_triggered_only = yes
	
	immediate = {
		from = { set_site_progress_locked = yes }
		leader = { unassign_leader = this }
		set_leader = event_target:time_hotel_leader
		owner = {
			if = {
				limit = {
					is_gestalt = no
				}
				event_target:time_hotel_leader = {
					add_experience = 500
					add_trait = {
						trait = leader_trait_time_hotel
					}
				}
			}
			else = {
				event_target:time_hotel_leader = {
					add_experience = 1000
				}
				random_owned_leader = {
					limit = {
						is_councilor_type = councilor_gestalt_cognitive
					}
					add_trait = {
						trait = leader_trait_time_hotel_gestalt
					}
				}
			}
		}
	}

	after = {
		hidden_effect = {
			fleet_event = { id = time_hotel.8 }
		}
		from = { expire_site_event = time_hotel.7 }
	}

	option = {
		name = time_hotel.7.a
		great_artifact_reward = yes
	}
}
