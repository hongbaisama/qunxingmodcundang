country_event = {
	id = new_country.0
	title = new_country.0.name
	desc = new_country.0.desc
	picture = GFX_evt_ship_in_orbit_3
	show_sound = event_pre_ftl_civilization

	is_triggered_only = yes

	trigger = {
	}
	
	option = {
		name = new_country.0.a
		hidden_effect = {
			random_playable_country = {
				limit = {
					has_origin = origin_new_country
					is_ai = yes
				}
				set_country_flag = allow_origin_new_country
				country_event = { id = new_country.1 }
				create_timeline_time_origin = { ORIGIN = origin_new_country }
			}
			every_playable_country = {
				limit = {
					has_origin = origin_new_country
					is_ai = yes
					NOT = { has_country_flag = allow_origin_new_country }
				}
				set_origin_default_effect = yes
				create_timeline_time_origin = { ORIGIN = origin_default }
			}
		}
	}
	
	option = {
		name = new_country.0.b
		hidden_effect = {
			every_playable_country = {
				limit = {
					has_origin = origin_new_country
					is_ai = yes
				}
				set_origin_default_effect = yes
				create_timeline_time_origin = { ORIGIN = origin_default }
			}
		}
	}
}

country_event = {
	id = new_country.1
	title = new_country.1.name
	desc = new_country.1.desc
	picture = GFX_evt_ship_in_orbit_3
	show_sound = event_pre_ftl_civilization

	is_triggered_only = yes

	trigger = {
		has_origin = origin_new_country
		OR = {
			is_ai = no
			has_country_flag = allow_origin_new_country
		}
	}
	
	immediate = {
		# 成立共同体
		if = {
			limit = {
				NOT = { has_global_flag = dark_forest }
			}
			set_global_flag = galcom_founding_begun
			set_global_flag = galactic_community_founded
			random_country = {
				limit = { is_country_type = global_event }
				country_event = { id = galcom.15 days = 20 }
			}
		}
		
		capital_scope = {
			solar_system = {
				save_event_target_as = origin_new_country_capital
			}
		}
		every_playable_country = {
			limit = {
				NOT = { has_origin = origin_new_country }
			}
			country_event = { id = new_country.2 }
		}
		create_timeline_origin_new_country_begin = yes
	}

	option = {
		name = new_country.1.a
		trigger = {
			has_origin = origin_new_country
		}
		add_modifier = {
			modifier = new_country_modifier
			days = 7200
		}
	}
}

country_event = {
	id = new_country.2
	title = new_country.2.name
	desc = new_country.2.desc
	picture = GFX_evt_ship_in_orbit_3
	show_sound = event_pre_ftl_civilization
	location = event_target:origin_new_country_capital

	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = new_country_2 }
	}
	
	immediate = {
		set_country_flag = new_country_2
		set_visited = event_target:origin_new_country_capital
		if = {
			limit = {
				NOR = {
					has_origin = origin_broken_shackles
					has_origin = origin_fourth_crisis
					has_country_flag = gain_time_mod_advanced_bonus
				}
			}
			generate_time_mod_advanced_bonus = { COUNT = 1 }
		}
		# 加入共同体
		if = {
			limit = {
				NOR = {
					has_country_flag = galcom_form_aye
					has_country_flag = galcom_form_nay
				}
			}
			if = {
				limit = {
					is_homicidal = no
					NOT = { has_origin = origin_fourth_crisis }
				}
				set_country_flag = galcom_form_aye
				set_country_flag = league_of_nations_achievement
				add_to_galactic_community_no_message = yes
			}
			else = {
				set_country_flag = galcom_form_nay
				set_country_flag = whatever_it_is_im_against_it_rejected_community
			}
		}
	}

	option = {
		name = new_country.2.a
	}
}
