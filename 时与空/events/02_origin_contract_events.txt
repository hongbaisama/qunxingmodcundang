country_event = {
	id = contract.1
	diplomatic_title = UNKNOWN
	title = contract.1.name
	desc = contract.1.desc
	
	is_triggered_only = yes
	diplomatic = yes
	force_open = yes
	
	picture_event_data = {
		portrait = shroud1
		room = shroud_room
	}
	show_sound = {
		trigger = { is_robot_empire = yes }
		sound = event_psionic_robot
	}
	show_sound = {
		trigger = { is_robot_empire = no }
		sound = event_psionic
	}

	trigger = {
		has_origin = origin_contract
	}

	immediate = {
		random_list = {
			1 = {
				set_country_flag = origin_contract_replace_the_eater_of_worlds
			}
			1 = {
				set_country_flag = origin_contract_replace_the_cradle_of_souls
			}
			1 = {
				set_country_flag = origin_contract_replace_the_instrument_of_desire
			}
			1 = {
				set_country_flag = origin_contract_replace_the_composer_of_strands
			}
		}
	}
	
	option = {
		name = contract.1.a
		hidden_effect = {
			country_event = { id = contract.2 }
		}
	}
}

	}
}

situation_event = {
	id = contract.100
	diplomatic_title = shroud.3.diplo_title
	title = contract.100.name
	desc = contract.100.desc
	
	is_triggered_only = yes
	diplomatic = yes
	force_open = yes
	
	picture_event_data = {
		portrait = shroud1
		room = shroud_room
	}
	show_sound = {
		trigger = {
			owner = { is_robot_empire = yes }
		}
		sound = event_psionic_robot
	}
	show_sound = {
		trigger = {
			owner = { is_robot_empire = no }
		}
		sound = event_psionic
	}

	trigger = {
		owner = { has_origin = origin_contract }
	}

	immediate = {
		owner = {
			origin_contract_whisperers_in_the_void_replace_patron_by_situation = {
				PATRON = the_eater_of_worlds
			}
		}
	}
	
	after = {
		owner = {
			clear_origin_contract_assimilation = yes
			add_modifier = {
				modifier = origin_contract_eater_assimilation
				days = -1
			}
			hidden_effect = {
				start_situation = {
					type = situation_origin_contract_eater_vs_cradle
					target = this
				}
			}
		}
		hidden_effect = {
			destroy_situation = this
		}
	}
	
	option = {
		name = contract.100.a
	}
}
