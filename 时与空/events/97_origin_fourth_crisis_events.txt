country_event = {
	id = fourth_crisis.1
	title = fourth_crisis.1.name
	desc = fourth_crisis.1.desc
	picture = GFX_evt_sapient_AI
	show_sound = event_death_cult

	is_triggered_only = yes

	trigger = {
		has_origin = origin_fourth_crisis
		OR = {
			is_ai = no
			has_country_flag = allow_origin_fourth_crisis
		}
	}
	
	immediate = {
		set_global_flag = fourth_crisis_happened
		if = {
			limit = {
				NOT = { has_country_flag = gain_time_mod_advanced_bonus }
			}
			generate_time_mod_advanced_bonus = { COUNT = 2 }
		}
		give_technology = { tech = "tech_ascension_theory" message = no }
		time_active_nemesis = yes
		if = {
			limit = {
				is_ai = yes
			}
			add_modifier = {
				modifier = fourth_crisis_ai_modifier
				days = -1
			}
		}
		
		# 成立共同体
		if = {
			limit = {
				NOT = { has_global_flag = dark_forest }
			}
			set_global_flag = galcom_founding_begun
			set_global_flag = galactic_community_founded
			random_country = {
				limit = { is_country_type = global_event }
				country_event = { id = galcom.15 days = 20 }
			}
		}

		set_global_flag = galactic_crisis_happened
		if = {
			limit = {
				NOT = { has_global_flag = origin_swarm_crisis_happened }
			}
			set_global_flag = origin_swarm_crisis_happened
			event_target:global_event_country = {
				country_event = { id = crisis.10 days = 10800 random = 7200 }
			}
		}
		if = {
			limit = {
				NOT = { has_global_flag = origin_ai_crisis_happened }
			}
			set_global_flag = origin_ai_crisis_happened
			event_target:global_event_country = {
				country_event = { id = crisis.2005 days = 10800 random = 7200 }
			}
		}
		if = {
			limit = {
				NOT = { has_global_flag = origin_exd_crisis_happened }
			}
			set_global_flag = origin_exd_crisis_happened
			event_target:global_event_country = {
				country_event = { id = crisis.1000 days = 10800 random = 7200 }
			}
		}
		if = {
			limit = {
				NOT = { has_global_flag = origin_synth_queen_crisis_happened }
			}
			set_global_flag = origin_synth_queen_crisis_happened
			event_target:global_event_country = {
				country_event = { id = crisis.8005 days = 10800 random = 7200 }
			}
		}
		every_playable_country = {
			limit = {
				NOT = { has_origin = origin_fourth_crisis }
			}
			establish_communications_no_message = root
			country_event = { id = fourth_crisis.2 }
		}
		create_timeline_origin_fourth_crisis_begin = yes
	}

	option = {
		name = fourth_crisis.1.a
		trigger = {
			has_origin = origin_fourth_crisis
		}
		tooltip = {
			add_ascension_perk = ap_become_the_crisis
		}
	}
}

country_event = {
	id = fourth_crisis.2
	title = fourth_crisis.2.name
	desc = fourth_crisis.2.desc
	picture = GFX_evt_sapient_AI
	show_sound = event_death_cult
	location = from

	is_triggered_only = yes

	trigger = {
		NOT = { has_country_flag = fourth_crisis_2 }
	}
	
	immediate = {
		set_country_flag = fourth_crisis_2
		# 加入共同体
		if = {
			limit = {
				NOR = {
					has_country_flag = galcom_form_aye
					has_country_flag = galcom_form_nay
				}
			}
			if = {
				limit = {
					is_homicidal = no
					NOT = { has_origin = origin_new_country }
				}
				set_country_flag = galcom_form_aye
				set_country_flag = league_of_nations_achievement
				add_to_galactic_community_no_message = yes
			}
			else = {
				set_country_flag = galcom_form_nay
				set_country_flag = whatever_it_is_im_against_it_rejected_community
			}
		}
	}

	option = {
		name = fourth_crisis.2.a
		if = {
			limit = {
				is_ai = no
			}
			add_modifier = {
				modifier = fourth_crisis_modifier
				days = -1
			}
		}
		# 使AI能更尽力抗击第四天灾
		else = {
			add_modifier = {
				modifier = fourth_crisis_modifier
				days = -1
				multiplier = 2
			}
		}
	}
}
