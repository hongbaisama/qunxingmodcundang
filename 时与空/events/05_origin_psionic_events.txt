country_event = {
	id = psionic.1
	title = psionic.1.name
	desc = psionic.1.desc
	picture = GFX_evt_psionics
	show_sound = event_mystic_reveal
	
	is_triggered_only = yes

	trigger = {
		has_origin = origin_psionic
	}
	
	immediate = {
		give_technology = { tech = "tech_psionic_theory" message = no }
		give_technology = { tech = "tech_mine_zro" message = no }
		give_technology = { tech = "tech_ascension_theory" message = no }
		if = {
			limit = {
				is_ai = yes
			}
			country_event = { id = psionic.1000 days = 1800 random = 1800 }
		}
		capital_scope = {
			# 设置区划、区域与建筑
			inline_script = "game_start/rural_districts_initializer"
			inline_script = "game_start/shipsets"
			inline_script = "game_start/species_lithoids"
			while = {
				limit = {
					has_district = district_city
				}
				remove_district = district_city
			}
			while = {
				count = 2
				add_district = district_psionic_city
			}
			while = {
				count = energy_districts
				add_district = district_psionic_generator
			}
			while = {
				count = mining_districts
				add_district = district_psionic_mining
			}
			while = {
				count = farming_districts
				add_district = district_psionic_farming
			}
			clear_variable = energy_districts
			clear_variable = mining_districts
			clear_variable = farming_districts
			add_zone = {
				district = district_psionic_city
				zone = zone_research_unity
				zone_slot = 1
			}
			add_zone = {
				district = district_psionic_city
				zone = zone_industrial
				zone_slot = 2
			}
			add_building = {
				building = building_factory_1
				zone = zone_industrial
			}
			## 只有唯物和唯心限定了区域
			inline_script = "game_start/ethic_spiritualist"
			validate_and_repair_planet_buildings_and_districts = yes
			
			set_colony_type = col_psionic_capital
			set_colony_type = none # 更新星球规划的同时改用自动规划
			create_ambient_object = {
				type = consecrated_planet_stage_2_object
				location = this

				use_3d_location = yes
				entity_scale_to_size = yes
			}
			last_created_ambient_object = {
				set_ambient_object_flag = psionic_origin_ambient_object
			}
		}
	}
	
	option = {
		name = "psionic.1.a"
		trigger = {
			has_origin = origin_psionic
		}
	}
}

	id = psionic.5
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		is_planet_class = pc_psionic
		owner = { has_origin = origin_psionic }
	}
	
	immediate = {
		if = {
			limit = {
				planet_size < 18
			}
			set_planet_size = 18
		}
		set_carrier_flag = original_psionic_planet
		add_modifier = {
			modifier = psionic_origin
			days = -1
		}
		# 设置地块
		reroll_pc_psionic_deposits = yes
		while = {
			count = trigger:num_guaranteed_colonies
			random_list = {
				1 = {
					add_deposit = d_psionic_generator
				}
				1 = {
					add_deposit = d_psionic_mining
				}
				1 = {
					add_deposit = d_psionic_farming
				}
			}
		}
	}
}

country_event = {
	id = psionic.1000
	hide_window = yes
	
	is_triggered_only = yes
	
	trigger = {
		has_origin = origin_psionic
		is_ai = yes
	}
	
	immediate = {
		country_event = { id = psionic.1000 days = 1800 random = 1800 }
		if = {
			limit = {
				has_technology = tech_terrestrial_sculpting
			}
			if = {
				limit = {
					any_planet_within_border = {
						is_planet_class = pc_gaia
						is_terraforming = no
					}
					has_country_resource = {
						type = sr_zro
						amount >= 250
					}
				}
				random_planet_within_border = {
					limit = {
						is_planet_class = pc_gaia
						is_terraforming = no
					}
					change_pc = pc_psionic
					carrier_event = { id = psionic.100 }
					create_colony = {
						owner = root
						species = root.owner_main_species
					}
				}
				add_resource = {
					sr_zro = -250
				}
			}
			else_if = {
				limit = {
					any_planet_within_border = {
						habitable_planet = yes
						is_terraforming = no
					}
					has_country_resource = {
						type = sr_zro
						amount >= 500
					}
				}
				random_planet_within_border = {
					limit = {
						habitable_planet = yes
						is_terraforming = no
					}
					change_pc = pc_psionic
					carrier_event = { id = psionic.100 }
					create_colony = {
						owner = root
						species = root.owner_main_species
					}
				}
				add_resource = {
					sr_zro = -500
				}
			}
		}
		else_if = {
			limit = {
				has_technology = tech_climate_restoration
			}
			if = {
				limit = {
					any_planet_within_border = {
						OR = {
							is_planet_class = pc_nuked
							is_planet_class = pc_hive
						}
						is_terraforming = no
					}
					has_country_resource = {
						type = sr_zro
						amount >= 1000
					}
				}
				random_planet_within_border = {
					limit = {
						OR = {
							is_planet_class = pc_nuked
							is_planet_class = pc_hive
						}
						is_terraforming = no
					}
					change_pc = pc_psionic
					carrier_event = { id = psionic.100 }
					create_colony = {
						owner = root
						species = root.owner_main_species
					}
				}
				add_resource = {
					sr_zro = -1000
				}
			}
		}
	}
}
