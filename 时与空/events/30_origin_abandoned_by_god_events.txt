namespace = origin_abandoned_by_god

country_event = {
	id = origin_abandoned_by_god.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		has_origin = origin_abandoned_by_god
	}
	
	immediate = {
		give_technology = { tech = tech_mine_dark_matter message = no }
		every_owned_leader = {
			add_trait = {
				trait = leader_trait_abandoned_by_god
				show_message = no
			}
			set_age = value:abandoned_by_god_initial_age
		}
		every_pool_leader = {
			add_trait = {
				trait = leader_trait_abandoned_by_god
				show_message = no
			}
			set_age = value:abandoned_by_god_initial_age
		}
		every_envoy = {
			add_trait = {
				trait = leader_trait_abandoned_by_god
				show_message = no
			}
			set_age = value:abandoned_by_god_initial_age
		}
		capital_scope = {
			set_carrier_flag = exists_building_abandoned_by_god
			add_building = building_abandoned_by_god
			add_zone = {
				district = district_city
				zone = zone_research
				zone_slot = 1
				replace = yes
			}
			validate_and_repair_planet_buildings_and_districts = yes
			solar_system = {
				every_system_planet = {
					limit = {
						has_carrier_flag = blessed_by_god_planet
					}
					create_research_station = { owner = root }
				}
				random_system_planet = {
					limit = {
						has_carrier_flag = abandoned_by_god_home_planet
					}
					root = {
						owner_species = {
							set_species_homeworld = prevprev
						}
					}
				}
			}
		}
	}
}

# 恢复神弃之族特殊地块
carrier_event = {
	id = origin_abandoned_by_god.11
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NAND = {
			has_deposit = d_abandoned_by_god_home_planet
			has_modifier = origin_abandoned_by_god_home_planet
		}
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { has_deposit = d_abandoned_by_god_home_planet }
			}
			add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_abandoned_by_god_home_planet }
		}
		if = {
			limit = {
				NOT = { has_modifier = origin_abandoned_by_god_home_planet }
			}
			add_modifier = {
				modifier = origin_abandoned_by_god_home_planet
				days = -1
			}
		}
	}
}
