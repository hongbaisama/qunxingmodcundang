fleet_event = {
	id = origin_reckoning_arrived.100
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		exists = event_target:origin_reckoning_arrived_country
	}
	
	immediate = {
		fleet_event = { id = origin_reckoning_arrived.100 days = 360 }
		event_target:origin_reckoning_arrived_country = {
			if = {
				limit = {
					capital_scope = { carrier_is_type = ship }
				}
				capital_scope = {
					ship = {
						fleet = { save_event_target_as = origin_reckoning_arrived_creatures_target }
					}
				}
			}
			else_if = {
				limit = {
					any_owned_colony = { carrier_is_type = ship }
				}
				random_owned_colony = {
					ship = {
						fleet = { save_event_target_as = origin_reckoning_arrived_creatures_target }
					}
				}
			}
			else_if = {
				limit = {
					capital_scope = { carrier_is_type = planet }
				}
				capital_scope = {
					planet = { save_event_target_as = origin_reckoning_arrived_creatures_target }
				}
			}
			else_if = {
				limit = {
					any_owned_colony = { carrier_is_type = planet }
				}
				random_owned_colony = {
					planet = { save_event_target_as = origin_reckoning_arrived_creatures_target }
				}
			}
			else = {
				random_owned_fleet = {
					save_event_target_as = origin_reckoning_arrived_creatures_target
				}
			}
		}
		if = {
			limit = {
				exists = event_target:origin_reckoning_arrived_creatures_target
			}
			if = {
				limit = {
					event_target:origin_reckoning_arrived_creatures_target = { is_scope_type = fleet }
				}
				auto_follow_fleet = {
					target = event_target:origin_reckoning_arrived_creatures_target
					attack_fleet = yes
				}
			}
			else = {
				auto_move_to_planet = {
					target = event_target:origin_reckoning_arrived_creatures_target
					clear_auto_move_on_arrival = yes
				}
			}
		}
	}
}
