country_event = {
	id = time_akx.5
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		if = {
			limit = {
				has_country_flag = coils_god_halted
			}
			event_target:coils_planet = {
				exists = owner
				owner = {
					is_country_type = primitive
				}
				solar_system = {
					any_system_planet = {
						NOT = { is_same_value = event_target:coils_planet }
						is_star = no
						is_asteroid = no
						is_artificial = no
						is_colonizable_or_colony = no
						can_not_be_changed_by_akx_worm = no
						NOT = { is_planet_class = pc_gas_giant }
					}
				}
			}
		}
		else = {
			event_target:coils_planet = {
				is_colonizable = yes
				NOT = { exists = owner }
			}
		}
	}
	
	immediate = {
		set_country_flag = time_akx_coils_god_triggered
		if = {
			limit = {
				has_country_flag = coils_god_halted
			}
			country_event = { id = time_akx.300 }
		}
		else_if = {
			limit = {
				has_country_flag = coils_god_end
			}
			country_event = { id = time_akx.310 }
		}
	}
}

	hide_window = yes

		is_colonizable_or_colony = yes
		if = {
			limit = {
				has_carrier_flag = exists_d_akx_messenger
			}
			NOR = {
				has_deposit = d_akx_messenger_closed
				has_deposit = d_akx_messenger
			}
		}
		else = {
			NOR = {
				has_deposit = d_akx_gestalt_messenger_closed
				has_deposit = d_akx_gestalt_messenger
			}
		}
	}
	
	immediate = {
		if = {
			limit = {
				has_carrier_flag = exists_d_akx_messenger
			}
			if = {
				limit = {
					OR = {
						has_carrier_flag = end_messenger_1
						has_carrier_flag = future_messenger_1
					}
				}
				add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_akx_messenger }
			}
			else = {
				add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_akx_messenger_closed }
			}
		}
		else = {
			if = {
				limit = {
					has_carrier_flag = gestalt_messenger_1
				}
				add_time_special_deposit_keep_on_terraform = { DEPOSIT = d_akx_gestalt_messenger }
			}

# 行动条件：战争中的敌对星球/自己境内的原始星球/自己拥有的星球
carrier_event = {
	id = time_akx.20030
	is_triggered_only = yes
	hide_window = yes
	
	immediate = {
		from = {
			remove_fleet_flag = in_time_titan_end_shelter_progress
			controller = { save_event_target_as = time_titan_controller }
			random_owned_ship = {
				limit = { is_time_titan_ship_size = yes }
				colony = { save_event_target_as = time_shelter_colony }
			}
		}
		# 清除敌对陆军
		every_planet_army = {
			limit = {
				NOT = { is_owned_by = event_target:time_titan_controller }
			}
			remove_army = yes
		}
		every_ground_combat_defender = {
			limit = {
				NOT = { is_owned_by = event_target:time_titan_controller }
			}
			remove_army = yes
		}
		every_ground_combat_attacker = {
			limit = {
				NOT = { is_owned_by = event_target:time_titan_controller }
			}
			remove_army = yes
		}
		# 占领敌对星球/掌控敌对星球（全战）
		if = {
			limit = {
				exists = owner
				owner = { is_at_war_with = event_target:time_titan_controller }
			}
			owner = {
				add_static_war_exhaustion = {
					attacker = event_target:time_titan_controller
					location = root
					value_for_planet_destruction = 1.0
				}
			}
			add_modifier = {
				modifier = time_titan_end_shelter_bad
				years = 5
			}
			if = {
				limit = {
					owner = {
						is_at_total_war_with = { TARGET = event_target:time_titan_controller }
					}
				}
				set_owner = event_target:time_titan_controller
				event_target:time_titan_controller = {
					create_time_titan_end_shelter_message = { TYPE = OWN }
				}
			}
			else = {
				set_controller = event_target:time_titan_controller
				event_target:time_titan_controller = {
					create_time_titan_end_shelter_message = { TYPE = CONTROL }
				}
			}
		}
		# 掌控原始星球
		else_if = {
			limit = {
				exists = owner
				owner = { is_country_type = primitive }
			}
			add_modifier = {
				modifier = time_titan_end_shelter_bad
				years = 5
			}
			add_modifier = {
				modifier = planet_culture_shock
				years = 10
			}
			set_owner = event_target:time_titan_controller
			event_target:time_titan_controller = {
				create_time_titan_end_shelter_message = { TYPE = OWN }
			}
		}
		# 庇护我方星球
		else = {
			add_modifier = {
				modifier = time_titan_end_shelter_good
				years = 5
			}
			if = {
				limit = {
					NOT = { is_controlled_by = event_target:time_titan_controller }
				}
				set_controller = event_target:time_titan_controller
			}
			event_target:time_titan_controller = {
				create_time_titan_end_shelter_message = { TYPE = SELF }
			}
		}
		# 转移人口
		every_owned_pop_group = {
			resettle_pop_group = {
				POP_GROUP = this
				PLANET = event_target:time_shelter_colony
				PERCENTAGE = 0.5
			}
		}
	}
}
