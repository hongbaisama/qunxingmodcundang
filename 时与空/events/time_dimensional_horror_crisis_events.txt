# 终焉宣告天灾出现
country_event = {
	id = time_dimensional_horror_crisis.1
	title = time_dimensional_horror_crisis.1.name
	desc = {
		text = time_dimensional_horror_crisis.1.desc.a
		trigger = {
			has_country_flag = discover_between_three_and_four
		}
	}
	desc = {
		text = time_dimensional_horror_crisis.1.desc.b
		trigger = {
			NOT = { has_country_flag = discover_between_three_and_four }
		}
	}
	picture = GFX_evt_astral_rift_library
	show_sound = ap_rift_event_15_time_crystal
	event_chain = beyond_infinity_chain
	
	is_triggered_only = yes

	trigger = {
		forbid_beyond_infinity = no
		NOT = { has_global_flag = dimensional_horror_crisis_happened }
	}
	
	immediate = {
		set_global_flag = dimensional_horror_crisis_happened
		create_country = {
			name = "NAME_Dimensional_Horror_Crisis_Country"
			type = dimensional_horror_crisis
			auto_delete = no
			flag = {
				icon= {
					category = "special"
					file = "extradimensional_02.dds"
				}
				background= {
					category = "backgrounds"
					file = "circle.dds"
				}
				colors={
					"dark_purple"
					"red"
					"null"
					"null"
				}
			}
			effect = {
				save_global_event_target_as = dimensional_horror_crisis_country
				country_event = { id = time_dimensional_horror_crisis.50 days = 720 }
			}
		}
		if = {
			limit = {
				event_target:akx_star = {
					solar_system = { is_owned_by = root }
				}
			}
			event_target:akx_star = {
				solar_system = { save_global_event_target_as = dimensional_horror_crisis_storm_start_system }
			}
		}
		else = {
			capital_scope = {
				solar_system = { save_global_event_target_as = dimensional_horror_crisis_storm_start_system }
			}
		}
		event_target:dimensional_horror_crisis_storm_start_system = {
			if = {
				limit = {
					is_inside_storm = yes
				}
				every_cosmic_storm = {
					limit = {
						any_system_within_storm = {
							is_same_value = event_target:dimensional_horror_crisis_storm_start_system
						}
					}
					destroy_cosmic_storm = yes
				}
			}
			random_create_dimensional_horror_crisis_storm = yes
		}
		# 以太相引擎会促进终焉宣告的形成
		every_system = {
			limit = {
				if = {
					limit = {
						any_system_megastructure = {
							is_crisis_sphere = yes
						}
					}
					NOR = {
						is_same_value = event_target:dimensional_horror_crisis_storm_start_system
						has_star_flag = random_create_dimensional_horror_crisis_storm_system
						is_inside_storm = yes
					}
				}
				else = {
					always = no
				}
			}
			set_timed_star_flag = {
				flag = random_create_dimensional_horror_crisis_storm_system
				days = 1
			}
			random_create_dimensional_horror_crisis_storm = yes
		}
		# 曾被时空抹除的星系也要随机出现终焉宣告
		while = {
			count = value:random_create_dimensional_horror_crisis_storm_system
			random_system = {
				limit = {
					if = {
						limit = {
							has_star_flag = time_erasure_system
						}
						NOR = {
							is_same_value = event_target:dimensional_horror_crisis_storm_start_system
							has_star_flag = random_create_dimensional_horror_crisis_storm_system
							is_inside_storm = yes
						}
					}
					else = {
						always = no
					}
				}
				set_timed_star_flag = {
					flag = random_create_dimensional_horror_crisis_storm_system
					days = 1
				}
				random_create_dimensional_horror_crisis_storm = yes
			}
		}
	}
	
	option = {
		name = time_dimensional_horror_crisis.1.a
		custom_tooltip = time_dimensional_horror_crisis.1.a.tooltip
	}
}

		}
		# 创造风暴

			limit = {

					is_star_class = sc_endless_time

			weights = {
				base = 1
				modifier = {
					is_time_black_hole_star_class = yes
					NOT = { has_star_flag = recent_dimensional_horror_crisis_storm_system }
					factor = 5
				}
				modifier = {
					has_star_flag = recent_dimensional_horror_crisis_storm_system

				}

				}

					is_storm_active_in_storm_sector = yes
					factor = 0.0001
				}

		}
	}

	id = time_dimensional_horror_crisis.200
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		is_dimensional_horror_crisis_inactive_system = no
		is_time_erasure_system = no
	}
	
	immediate = {
		if = {
			limit = {
				is_in_combat = no
				is_time_erasure_system = no
				can_dimensional_horror_crisis_erasured_system = yes
			}
			random_owned_ship = {
				save_event_target_as = akx_time_eraser_ship
			}
			owner = {
				save_event_target_as = akx_time_eraser
			}
			solar_system = {
				save_event_target_as = dimensional_horror_crisis_erasured_system
				set_star_flag = dimensional_horror_crisis_erasured
				if = {
					limit = {
						count_neighbor_system = {
							count < 1
						}
						NOT = { has_star_flag = lcluster }
					}
					akx_time_erasure_system = yes
					remove_star_flag = lcluster # 避免孤立星系无法进入
				}
				else = {
					akx_time_erasure_system = yes
				}
			}
			event_target:global_event_country = {
				change_variable = {
					which = dimensional_horror_crisis_erasured_system
					value = 1
				}
			}
			increase_beyond_infinity_seal_requirement = yes
			add_dimensional_horror_crisis_chain_counter = { COUNTER = dimensional_horror_crisis_erasured_system }
			if = {
				limit = {
					any_playable_country = { has_event_chain = beyond_infinity_seal_chain }
				}
				every_playable_country = {
					limit = {
						has_event_chain = beyond_infinity_seal_chain
					}
					event_target:dimensional_horror_crisis_erasured_system = {
						set_star_flag = exists_beyond_infinity_seal_project_for@prev

						star = {
							enable_special_project = {
								name = BEYOND_INFINITY_SEAL_PROJECT
								location = this
								owner = prevprev
							}
						}
					}
				}
			}
		}
		else = {
			fleet_event = { id = time_dimensional_horror_crisis.200 days = 90 }
		}
	}
}

			}
		}
	}
	desc = {
		text = time_dimensional_horror_crisis.500.desc.b
		trigger = {
			event_target:dimensional_horror_crisis_sealed_system = {
				NOT = { has_star_flag = last_sealed_erasured_system }
			}
		}
	}
	picture = GFX_dimensional_horror_crisis_storm
	show_sound = ap_rift_event_10_wormhole_sealed
	location = event_target:dimensional_horror_crisis_sealed_system
	event_chain = beyond_infinity_seal_chain
	
	is_triggered_only = yes

	trigger = {
		exists = event_target:dimensional_horror_crisis_sealed_system
		event_target:dimensional_horror_crisis_sealed_system = {
			is_dimensional_horror_crisis_sealed_system = no
		}
	}
	
	immediate = {
		event_target:dimensional_horror_crisis_sealed_system = {
			set_star_flag = dimensional_horror_crisis_sealed
			every_system_ambient_object = {
				limit = {
					has_ambient_object_flag = dimensional_horror_crisis_storm_object
				}
				destroy_ambient_object = this
			}
			every_fleet_in_system = {
				limit = {
					is_dimensional_horror_crisis_ship_size = yes
					NOT = { is_same_value = event_target:dimensional_horror_crisis_destiny_fleet }
				}
				destroy_fleet = this
			}
		}
		event_target:global_event_country = {
			change_variable = {
				which = dimensional_horror_crisis_sealed_system
				value = 1
			}
			if = {
				limit = {
					event_target:dimensional_horror_crisis_sealed_system = {
						is_dimensional_horror_crisis_erasured_system = yes
					}
				}
				change_variable = {
					which = dimensional_horror_crisis_sealed_erasured_system
					value = 1
				}
				meet_beyond_infinity_seal_requirement = yes
			}
		}
		owner = {
			add_event_chain_counter = {
				event_chain = dimensional_horror_crisis_chain
				counter = dimensional_horror_crisis_sealed_us
				amount = 1
			}
		}
		add_dimensional_horror_crisis_chain_counter = { COUNTER = dimensional_horror_crisis_sealed_system }
		# 移除其它国家对该星系的封印项目
		every_playable_country = {
			limit = {
				has_event_chain = beyond_infinity_seal_chain
				NOT = { is_same_value = root.owner }
			}
			abort_special_project = {
				type = BEYOND_INFINITY_SEAL_PROJECT
				location = event_target:dimensional_horror_crisis_sealed_star
			}
		}
		# 判断全部被抹除星系是否都已被封印
		if = {
			limit = {
				NOT = { has_global_flag = all_erasured_system_is_sealed }
			}
			if = {
				limit = {
					event_target:global_event_country = {
						check_variable = {
							which = dimensional_horror_crisis_sealed_erasured_system
							value >= dimensional_horror_crisis_erasured_system
						}
					}
				}
				set_global_flag = all_erasured_system_is_sealed
				# 用于文本差异判断
				event_target:dimensional_horror_crisis_sealed_system = {
					set_star_flag = last_sealed_erasured_system
				}
				# 观察者事件
				observer_event = { id = beyond_infinity.4650 }
				# 不是超越无限国家，立即触发下一个事件
				if = {
					limit = {
						owner = {
							NOT = { is_same_value = event_target:beyond_infinity_country }
						}
					}
					event_target:beyond_infinity_country = {
						country_event = { id = beyond_infinity.4500 }
					}
				}
			}
		}
	}
	
	after = {
		hidden_effect = {
			# 将该星系的封印项目从日志中移除
			owner = {
				abort_special_project = {
					type = BEYOND_INFINITY_SEAL_PROJECT
					location = event_target:dimensional_horror_crisis_sealed_star
				}
			}
		}
	}
	
	option = {
		name = time_dimensional_horror_crisis.500.a
		trigger = {
			event_target:dimensional_horror_crisis_sealed_system = {
				has_star_flag = last_sealed_erasured_system
			}
		}
		custom_tooltip = time_dimensional_horror_crisis.500.a.tooltip
		custom_tooltip = time_dimensional_horror_crisis.500.b.tooltip
		hidden_effect = {
			event_target:dimensional_horror_crisis_sealed_system = {
				remove_star_flag = last_sealed_erasured_system
			}
			# 是超越无限国家，事后触发下一个事件
			if = {
				limit = {
					owner = { is_same_value = event_target:beyond_infinity_country }
				}
				event_target:beyond_infinity_country = {
					country_event = { id = beyond_infinity.4500 }
				}
			}
		}
	}
	
	option = {
		name = time_dimensional_horror_crisis.500.b
		trigger = {
			event_target:dimensional_horror_crisis_sealed_system = {
				NOT = { has_star_flag = last_sealed_erasured_system }
			}
		}
		custom_tooltip = time_dimensional_horror_crisis.500.b.tooltip
	}
